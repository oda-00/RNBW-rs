//! Lane-implied 4-ary trie codec for sorted sets of unsigned integers.
//!
//! Each trie node is a four-lane mask. A set lane implies the corresponding
//! two key bits, so edges carry no key data. Leaves index unary bucket counts
//! and packed low suffixes. The serialized query form is one contiguous byte
//! image and does not require a heap directory to answer membership queries.

use std::error::Error;
use std::fmt;
use std::sync::OnceLock;

pub mod lane_lz;
mod rans;
pub mod store;
pub mod trnmap;

#[cfg(feature = "wgpu")]
pub use lane_lz::gpu::{LaneLzGpu, LaneLzRestarted};
pub use lane_lz::{LaneLz, LaneLzEncoder};
pub use trnmap::{TriWord, TrnMap, TrnMapError};

const MAGIC: &[u8; 4] = b"RNBW";
const VERSION: u16 = 2;
const FIXED_HEADER: usize = 40;
const LEVEL_DESCRIPTOR: usize = 16;
const TRAILER_LEN: usize = 32;
const SELECT_STRIDE: usize = 64;
const RANK_STRIDE: usize = 128;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    UnsortedInput { index: usize },
    ValueOutOfRange { index: usize, width: u8 },
    Truncated,
    InvalidFormat(&'static str),
    LengthOverflow,
    Gpu(String),
}

impl fmt::Display for CodecError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::UnsortedInput { index } => {
                write!(f, "input is not strictly increasing at index {index}")
            }
            Self::ValueOutOfRange { index, width } => {
                write!(f, "value at index {index} does not fit in {width} bits")
            }
            Self::Truncated => f.write_str("truncated RNBW byte stream"),
            Self::InvalidFormat(reason) => write!(f, "invalid RNBW byte stream: {reason}"),
            Self::LengthOverflow => f.write_str("RNBW stream exceeds the 32-bit format limits"),
            Self::Gpu(reason) => write!(f, "RNBW GPU operation failed: {reason}"),
        }
    }
}

impl Error for CodecError {}

/// Encoded set in the directly queryable lane-trie representation.
#[derive(Debug, Clone)]
pub struct Rnbw {
    image: Vec<u8>,
}

/// Validated zero-copy view of a directly queryable RNBW image.
#[derive(Debug, Clone, Copy)]
pub struct RnbwView<'a> {
    image: &'a [u8],
}

impl<'a> RnbwView<'a> {
    /// Internal view for bytes validated when their owning store record was built.
    #[inline]
    pub(crate) fn from_validated_bytes(image: &'a [u8]) -> Self {
        Self { image }
    }
}

#[derive(Clone, Copy, Debug, Default)]
struct LevelOffsets {
    mask: usize,
    rank: usize,
    rank_count: usize,
}

#[derive(Clone, Copy, Debug, Default)]
struct BucketOffsets {
    bits: usize,
    bit_len: usize,
    bytes: usize,
    select: usize,
    select_count: usize,
    low: usize,
    low_width: usize,
    low_bytes: usize,
}

#[derive(Debug)]
struct ImageOffsets {
    header_len: usize,
    levels: [LevelOffsets; 16],
    buckets: BucketOffsets,
    total_len: usize,
}

impl Rnbw {
    /// Encodes a strictly increasing, unique list of 32-bit keys.
    pub fn encode(sorted_values: &[u32]) -> Result<Self, CodecError> {
        Self::encode_width(sorted_values, 32)
    }

    /// Encodes keys using an even key width from 2 through 32 bits.
    ///
    /// This is used by the multi-level manager for a group or chunk whose
    /// directory already implies the higher bits.
    pub fn encode_width(sorted_values: &[u32], width: u8) -> Result<Self, CodecError> {
        validate_input(sorted_values, width)?;
        let max_depth = (width / 2) as usize;
        let mut node_counts = [0usize; 17];
        count_prefixes_by_level(
            sorted_values,
            width,
            max_depth,
            &mut node_counts[..=max_depth],
        );
        let nodes = &node_counts[..=max_depth];

        let mut best: Option<(usize, usize)> = None;
        for depth in 0..=max_depth {
            let leaves = nodes[depth];
            let offsets = image_offsets(width, depth, sorted_values.len(), leaves, nodes)?;
            let size = offsets.total_len;
            if best.is_none_or(|(best_size, best_depth)| {
                size < best_size || (size == best_size && depth > best_depth)
            }) {
                best = Some((size, depth));
            }
        }
        let (total_len, depth) = best.ok_or(CodecError::LengthOverflow)?;
        let leaves = nodes[depth];
        let offsets = image_offsets(width, depth, sorted_values.len(), leaves, nodes)?;
        if offsets.total_len != total_len {
            return Err(CodecError::InvalidFormat("unstable split cost"));
        }
        let mut image = vec![0u8; total_len];
        write_header(
            &mut image,
            width,
            depth,
            sorted_values.len(),
            leaves,
            nodes,
            &offsets,
        )?;

        let mut masks = Vec::new();
        for (level, &node_count) in nodes.iter().take(depth).enumerate() {
            build_level_masks(&mut masks, sorted_values, width, level, node_count);
            let offsets_for_level = offsets.levels[level];
            for (word_index, word) in masks.iter().enumerate() {
                let at = offsets_for_level.mask + word_index * 8;
                image[at..at + 8].copy_from_slice(&word.to_le_bytes());
            }
            let rank_bytes = offsets_for_level.rank_count * 4;
            build_rank_directory(
                &masks,
                node_count,
                &mut image[offsets_for_level.rank..offsets_for_level.rank + rank_bytes],
            );
        }

        build_bucket_stream(
            &mut image,
            sorted_values,
            width,
            depth,
            leaves,
            offsets.buckets,
        )?;
        build_low_stream(
            &mut image[offsets.buckets.low..offsets.buckets.low + offsets.buckets.low_bytes],
            sorted_values,
            offsets.buckets.low_width,
        );

        #[cfg(debug_assertions)]
        validate_image(&image)?;
        Ok(Self { image })
    }

    /// Sorts, deduplicates, then encodes arbitrary input.
    pub fn encode_unsorted(values: &[u32]) -> Result<Self, CodecError> {
        let mut sorted = values.to_vec();
        sorted.sort_unstable();
        sorted.dedup();
        Self::encode(&sorted)
    }

    /// Parses and validates a byte image, copying it into owned storage.
    pub fn from_bytes(bytes: &[u8]) -> Result<Self, CodecError> {
        validate_image(bytes)?;
        Ok(Self {
            image: bytes.to_vec(),
        })
    }

    /// Parses and validates a byte image while taking ownership of its buffer.
    pub fn from_bytes_owned(image: Vec<u8>) -> Result<Self, CodecError> {
        validate_image(&image)?;
        Ok(Self { image })
    }

    /// Number of keys in the set.
    #[inline]
    pub fn len(&self) -> usize {
        read_u64_unchecked(&self.image, 12) as usize
    }

    /// Whether the set is empty.
    #[inline]
    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    /// Serialized and directly queryable image.
    #[inline]
    pub fn as_bytes(&self) -> &[u8] {
        &self.image
    }

    /// Serializes the masks, unary bucket vector, and low bits with four-state
    /// order-0 rANS whenever the model and payload together reduce their size.
    pub fn to_compressed_bytes(&self) -> Result<Vec<u8>, CodecError> {
        rans::compress_image(&self.image)
    }

    /// Restores a validated directly queryable image from the entropy form.
    pub fn from_compressed_bytes(bytes: &[u8]) -> Result<Self, CodecError> {
        Ok(Self {
            image: rans::decompress_image(bytes)?,
        })
    }

    /// Query image has no separately allocated in-memory directory.
    #[inline]
    pub fn runtime_index_bytes(&self) -> usize {
        0
    }

    /// Tests membership without decoding or allocating.
    #[inline]
    pub fn contains(&self, value: u32) -> bool {
        self.view().contains(value)
    }

    /// Decodes all keys in increasing order.
    pub fn decode(&self) -> Vec<u32> {
        self.view().decode()
    }

    /// Borrows the validated queryable image.
    #[inline]
    pub fn view(&self) -> RnbwView<'_> {
        RnbwView { image: &self.image }
    }

    /// Number of trie levels selected for the key set.
    #[inline]
    pub fn depth(&self) -> usize {
        self.image[9] as usize
    }

    /// Selected CPU membership kernel; feature detection is cached process-wide.
    #[inline]
    pub fn simd_backend() -> &'static str {
        dispatch().name
    }
}

impl<'a> RnbwView<'a> {
    /// Parses and validates a borrowed queryable image without copying it.
    pub fn from_bytes(image: &'a [u8]) -> Result<Self, CodecError> {
        validate_image(image)?;
        Ok(Self { image })
    }

    #[inline]
    pub fn len(&self) -> usize {
        read_u64_unchecked(self.image, 12) as usize
    }

    #[inline]
    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    #[inline]
    pub fn as_bytes(&self) -> &'a [u8] {
        self.image
    }

    #[inline]
    pub fn runtime_index_bytes(&self) -> usize {
        0
    }

    #[inline]
    pub fn contains(&self, value: u32) -> bool {
        contains_image(self.image, value)
    }

    pub fn decode(&self) -> Vec<u32> {
        let mut output = Vec::with_capacity(self.len());
        let depth = self.image[9] as usize;
        let leaves = read_u32_unchecked(self.image, 20) as usize;
        if leaves == 0 {
            return output;
        }
        decode_node(self.image, 0, 0, 0, depth, &mut [0; 17], &mut output);
        output
    }

    #[inline]
    pub fn depth(&self) -> usize {
        self.image[9] as usize
    }
}

fn validate_input(values: &[u32], width: u8) -> Result<(), CodecError> {
    if !(2..=32).contains(&width) || !width.is_multiple_of(2) {
        return Err(CodecError::InvalidFormat(
            "key width must be even and between 2 and 32",
        ));
    }
    if values.len() > u32::MAX as usize {
        return Err(CodecError::LengthOverflow);
    }
    if width < 32 {
        let limit = 1u32 << width;
        if let Some((index, _)) = values
            .iter()
            .enumerate()
            .find(|(_, value)| **value >= limit)
        {
            return Err(CodecError::ValueOutOfRange { index, width });
        }
    }
    for (index, pair) in values.windows(2).enumerate() {
        if pair[0] >= pair[1] {
            return Err(CodecError::UnsortedInput { index: index + 1 });
        }
    }
    Ok(())
}

fn count_prefixes_by_level(values: &[u32], width: u8, max_depth: usize, counts: &mut [usize]) {
    counts.fill(0);
    if values.is_empty() {
        return;
    }
    counts[0] = 1;

    // Each adjacent pair contributes one new prefix at every depth below its
    // first differing 2-bit digit. A difference array counts all levels in a
    // single pass instead of rescanning the input once per level.
    let mut increments = [0usize; 17];
    let width = width as usize;
    for pair in values.windows(2) {
        let differing_bits = (u32::BITS - (pair[0] ^ pair[1]).leading_zeros()) as usize;
        let common_bits = width - differing_bits;
        let first_distinct_depth = common_bits / 2 + 1;
        if first_distinct_depth <= max_depth {
            increments[first_distinct_depth] += 1;
        }
    }
    let mut added = 0usize;
    for depth in 1..=max_depth {
        added += increments[depth];
        counts[depth] = added + 1;
    }
}

fn image_offsets(
    width: u8,
    depth: usize,
    key_count: usize,
    leaf_count: usize,
    node_counts: &[usize],
) -> Result<ImageOffsets, CodecError> {
    let header_len = FIXED_HEADER
        .checked_add(
            depth
                .checked_mul(LEVEL_DESCRIPTOR)
                .ok_or(CodecError::LengthOverflow)?,
        )
        .and_then(|value| value.checked_add(TRAILER_LEN))
        .ok_or(CodecError::LengthOverflow)?;
    if header_len > u16::MAX as usize {
        return Err(CodecError::LengthOverflow);
    }
    let bucket_bits = key_count
        .checked_add(leaf_count)
        .ok_or(CodecError::LengthOverflow)?;
    if bucket_bits > u32::MAX as usize || leaf_count > u32::MAX as usize {
        return Err(CodecError::LengthOverflow);
    }
    let mut position = header_len;
    let mut levels = [LevelOffsets::default(); 16];
    for (level, &node_count) in node_counts.iter().take(depth).enumerate() {
        position = align_up(position, 8).ok_or(CodecError::LengthOverflow)?;
        let mask_len = node_count
            .div_ceil(16)
            .checked_mul(8)
            .ok_or(CodecError::LengthOverflow)?;
        let mask = position;
        position = position
            .checked_add(mask_len)
            .ok_or(CodecError::LengthOverflow)?;
        position = align_up(position, 4).ok_or(CodecError::LengthOverflow)?;
        let rank = position;
        let rank_count = node_count.div_ceil(RANK_STRIDE);
        position = position
            .checked_add(
                rank_count
                    .checked_mul(4)
                    .ok_or(CodecError::LengthOverflow)?,
            )
            .ok_or(CodecError::LengthOverflow)?;
        levels[level] = LevelOffsets {
            mask,
            rank,
            rank_count,
        };
        position = align_up(position, 8).ok_or(CodecError::LengthOverflow)?;
    }
    position = align_up(position, 8).ok_or(CodecError::LengthOverflow)?;
    let bucket_offset = position;
    let bucket_bytes = bucket_bits.div_ceil(8);
    position = position
        .checked_add(bucket_bytes)
        .ok_or(CodecError::LengthOverflow)?;
    position = align_up(position, 4).ok_or(CodecError::LengthOverflow)?;
    let select_offset = position;
    let select_count = leaf_count.div_ceil(SELECT_STRIDE);
    position = position
        .checked_add(
            select_count
                .checked_mul(4)
                .ok_or(CodecError::LengthOverflow)?,
        )
        .ok_or(CodecError::LengthOverflow)?;
    position = align_up(position, 8).ok_or(CodecError::LengthOverflow)?;
    let low_offset = position;
    let low_width = width as usize - depth * 2;
    let low_bits = key_count
        .checked_mul(low_width)
        .ok_or(CodecError::LengthOverflow)?;
    let low_bytes = low_bits.div_ceil(8);
    position = position
        .checked_add(low_bytes)
        .ok_or(CodecError::LengthOverflow)?;
    if position > u32::MAX as usize {
        return Err(CodecError::LengthOverflow);
    }
    Ok(ImageOffsets {
        header_len,
        levels,
        buckets: BucketOffsets {
            bits: bucket_offset,
            bit_len: bucket_bits,
            bytes: bucket_bytes,
            select: select_offset,
            select_count,
            low: low_offset,
            low_width,
            low_bytes,
        },
        total_len: position,
    })
}

fn write_header(
    image: &mut [u8],
    width: u8,
    depth: usize,
    key_count: usize,
    leaf_count: usize,
    node_counts: &[usize],
    offsets: &ImageOffsets,
) -> Result<(), CodecError> {
    image[0..4].copy_from_slice(MAGIC);
    put_u16(image, 4, VERSION)?;
    put_u16(
        image,
        6,
        u16::try_from(offsets.header_len).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    image[8] = width;
    image[9] = depth as u8;
    image[10] = complete_levels(node_counts, depth) as u8;
    image[11] = 0;
    put_u64(image, 12, key_count as u64)?;
    put_u32(
        image,
        20,
        u32::try_from(leaf_count).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    put_u32(
        image,
        24,
        u32::try_from(offsets.total_len).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    put_u32(image, 28, FIXED_HEADER as u32)?;
    put_u32(
        image,
        32,
        u32::try_from(FIXED_HEADER + depth * LEVEL_DESCRIPTOR)
            .map_err(|_| CodecError::LengthOverflow)?,
    )?;
    put_u32(image, 36, 0)?;
    for level in 0..depth {
        let at = FIXED_HEADER + level * LEVEL_DESCRIPTOR;
        put_u32(
            image,
            at,
            u32::try_from(node_counts[level]).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        put_u32(
            image,
            at + 4,
            u32::try_from(node_counts[level + 1]).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        put_u32(
            image,
            at + 8,
            u32::try_from(offsets.levels[level].mask).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        put_u32(
            image,
            at + 12,
            u32::try_from(offsets.levels[level].rank).map_err(|_| CodecError::LengthOverflow)?,
        )?;
    }
    let bucket = offsets.buckets;
    let at = FIXED_HEADER + depth * LEVEL_DESCRIPTOR;
    put_u32(
        image,
        at,
        u32::try_from(bucket.bit_len).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    put_u32(
        image,
        at + 4,
        u32::try_from(bucket.bits).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    put_u32(
        image,
        at + 8,
        u32::try_from(bucket.bytes).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    put_u32(
        image,
        at + 12,
        u32::try_from(bucket.select).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    put_u32(
        image,
        at + 16,
        u32::try_from(bucket.select_count).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    put_u32(
        image,
        at + 20,
        u32::try_from(bucket.low).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    image[at + 24] = bucket.low_width as u8;
    image[at + 25..at + 28].fill(0);
    put_u32(
        image,
        at + 28,
        u32::try_from(bucket.low_bytes).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    Ok(())
}

fn complete_levels(node_counts: &[usize], depth: usize) -> usize {
    let mut count = 0usize;
    for level in 0..depth {
        if node_counts[level + 1] != node_counts[level].saturating_mul(4) {
            break;
        }
        count += 1;
    }
    count
}

fn build_level_masks(words: &mut Vec<u64>, values: &[u32], width: u8, level: usize, nodes: usize) {
    words.clear();
    words.resize(nodes.div_ceil(16), 0);
    let parent_shift = width as usize - level * 2;
    let digit_shift = parent_shift - 2;
    let mut previous_parent = None;
    let mut node = 0usize;
    for &value in values {
        let parent = if level == 0 { 0 } else { value >> parent_shift };
        if previous_parent != Some(parent) {
            if previous_parent.is_some() {
                node += 1;
            }
            previous_parent = Some(parent);
        }
        let digit = ((value >> digit_shift) & 3) as usize;
        words[node / 16] |= 1u64 << ((node % 16) * 4 + digit);
    }
}

fn build_rank_directory(masks: &[u64], node_count: usize, output: &mut [u8]) {
    let mut total = 0u32;
    for block in 0..node_count.div_ceil(RANK_STRIDE) {
        let at = block * 4;
        output[at..at + 4].copy_from_slice(&total.to_le_bytes());
        let first_word = block * (RANK_STRIDE / 16);
        let end_word = (first_word + RANK_STRIDE / 16).min(masks.len());
        for word in &masks[first_word..end_word] {
            total += word.count_ones();
        }
    }
}

fn build_bucket_stream(
    image: &mut [u8],
    values: &[u32],
    width: u8,
    depth: usize,
    leaves: usize,
    offsets: BucketOffsets,
) -> Result<(), CodecError> {
    let (before_select, after_select) = image.split_at_mut(offsets.select);
    let bits = &mut before_select[offsets.bits..offsets.bits + offsets.bytes];
    let samples = &mut after_select[..offsets.select_count * 4];
    let mut sample_index = 0usize;
    let mut value_start = 0usize;
    let mut bit_position = 0usize;
    let mut leaf = 0usize;
    while value_start < values.len() {
        let prefix = prefix_at_depth(values[value_start], width, depth);
        let mut value_end = value_start + 1;
        while value_end < values.len() && prefix_at_depth(values[value_end], width, depth) == prefix
        {
            value_end += 1;
        }
        set_bit_range(bits, bit_position, value_end - value_start);
        bit_position += value_end - value_start;
        if leaf.is_multiple_of(SELECT_STRIDE) {
            let sample = u32::try_from(bit_position).map_err(|_| CodecError::LengthOverflow)?;
            let at = sample_index * 4;
            samples[at..at + 4].copy_from_slice(&sample.to_le_bytes());
            sample_index += 1;
        }
        bit_position += 1;
        leaf += 1;
        value_start = value_end;
    }
    if leaf != leaves || bit_position != offsets.bit_len {
        return Err(CodecError::InvalidFormat(
            "bucket sizing disagrees with split",
        ));
    }
    if sample_index != offsets.select_count {
        return Err(CodecError::InvalidFormat(
            "select sample count disagrees with sizing",
        ));
    }
    Ok(())
}

fn set_bit_range(bytes: &mut [u8], mut start: usize, mut len: usize) {
    if len == 0 {
        return;
    }
    let partial = if start & 7 == 0 {
        0
    } else {
        (8 - (start & 7)).min(len)
    };
    if partial != 0 {
        let mask = ((1u16 << partial) - 1) as u8;
        bytes[start / 8] |= mask << (start & 7);
        start += partial;
        len -= partial;
    }
    let full_bytes = len / 8;
    if full_bytes != 0 {
        let first = start / 8;
        bytes[first..first + full_bytes].fill(u8::MAX);
        start += full_bytes * 8;
        len -= full_bytes * 8;
    }
    if len != 0 {
        bytes[start / 8] |= ((1u16 << len) - 1) as u8;
    }
}

fn build_low_stream(low: &mut [u8], values: &[u32], low_width: usize) {
    let mask = low_mask(low_width);
    for (index, value) in values.iter().copied().enumerate() {
        write_packed(low, index * low_width, low_width, value & mask);
    }
}

#[inline]
fn prefix_at_depth(value: u32, width: u8, depth: usize) -> u32 {
    if depth == 0 {
        0
    } else {
        value >> (width as usize - 2 * depth)
    }
}

#[inline]
fn low_mask(width: usize) -> u32 {
    match width {
        0 => 0,
        1..=31 => (1u32 << width) - 1,
        _ => u32::MAX,
    }
}

fn validate_image(image: &[u8]) -> Result<(), CodecError> {
    if image.len() < FIXED_HEADER {
        return Err(CodecError::Truncated);
    }
    if image.get(0..4) != Some(MAGIC.as_slice()) {
        return Err(CodecError::InvalidFormat("bad magic"));
    }
    if read_u16(image, 4)? != VERSION {
        return Err(CodecError::InvalidFormat("unsupported version"));
    }
    let header_len = read_u16(image, 6)? as usize;
    let width = image[8];
    let depth = image[9] as usize;
    let complete = image[10] as usize;
    if !(2..=32).contains(&width) || !width.is_multiple_of(2) || depth > width as usize / 2 {
        return Err(CodecError::InvalidFormat(
            "invalid key width or split depth",
        ));
    }
    if image[11] != 0 || read_u32(image, 28)? as usize != FIXED_HEADER {
        return Err(CodecError::InvalidFormat(
            "unsupported flags or descriptor offset",
        ));
    }
    if read_u32(image, 36)? != 0 {
        return Err(CodecError::InvalidFormat("nonzero reserved header bits"));
    }
    let expected_header = FIXED_HEADER + depth * LEVEL_DESCRIPTOR + TRAILER_LEN;
    let trailer_at = FIXED_HEADER + depth * LEVEL_DESCRIPTOR;
    if header_len != expected_header || read_u32(image, 32)? as usize != trailer_at {
        return Err(CodecError::InvalidFormat(
            "invalid header or trailer length",
        ));
    }
    let key_count =
        usize::try_from(read_u64(image, 12)?).map_err(|_| CodecError::LengthOverflow)?;
    let leaf_count = read_u32(image, 20)? as usize;
    let total_len = read_u32(image, 24)? as usize;
    if total_len != image.len() {
        return Err(if total_len > image.len() {
            CodecError::Truncated
        } else {
            CodecError::InvalidFormat("trailing bytes")
        });
    }
    if key_count > u32::MAX as usize || leaf_count > key_count && key_count != 0 {
        return Err(CodecError::InvalidFormat("impossible key or leaf count"));
    }
    if (key_count == 0 && (depth != 0 || leaf_count != 0))
        || (key_count > 0 && leaf_count == 0)
        || (key_count > 0 && depth == 0 && leaf_count != 1)
    {
        return Err(CodecError::InvalidFormat(
            "empty set and trie metadata disagree",
        ));
    }
    let mut nodes = Vec::with_capacity(depth + 1);
    if depth > 0 {
        nodes.push(read_u32(image, FIXED_HEADER)? as usize);
        if nodes[0] != 1 {
            return Err(CodecError::InvalidFormat("trie root count is not one"));
        }
        for level in 0..depth {
            let at = FIXED_HEADER + level * LEVEL_DESCRIPTOR;
            let current_nodes = read_u32(image, at)? as usize;
            let children = read_u32(image, at + 4)? as usize;
            if current_nodes != nodes[level] || children > current_nodes.saturating_mul(4) {
                return Err(CodecError::InvalidFormat("invalid level population"));
            }
            nodes.push(children);
        }
        if nodes[depth] != leaf_count {
            return Err(CodecError::InvalidFormat(
                "leaf count disagrees with trie levels",
            ));
        }
    } else if key_count > 0 {
        nodes.push(1);
    } else {
        nodes.push(0);
    }

    let offsets = image_offsets(width, depth, key_count, leaf_count, &nodes)?;
    if offsets.header_len != header_len || offsets.total_len != total_len {
        return Err(CodecError::InvalidFormat(
            "image size disagrees with its counts",
        ));
    }
    let mut actual_complete = 0usize;
    for level in 0..depth {
        let at = FIXED_HEADER + level * LEVEL_DESCRIPTOR;
        let descriptor = offsets.levels[level];
        if read_u32(image, at + 8)? as usize != descriptor.mask
            || read_u32(image, at + 12)? as usize != descriptor.rank
        {
            return Err(CodecError::InvalidFormat("noncanonical level offsets"));
        }
        let node_count = nodes[level];
        let word_count = node_count.div_ceil(16);
        let mut child_total = 0u32;
        let rank_count = descriptor.rank_count;
        for block in 0..rank_count {
            let expected_rank = child_total;
            if read_u32(image, descriptor.rank + block * 4)? != expected_rank {
                return Err(CodecError::InvalidFormat("bad rank directory entry"));
            }
            let first_word = block * (RANK_STRIDE / 16);
            let end_word = (first_word + RANK_STRIDE / 16).min(word_count);
            for word_index in first_word..end_word {
                let word = read_u64(image, descriptor.mask + word_index * 8)?;
                child_total = child_total
                    .checked_add(word.count_ones())
                    .ok_or(CodecError::LengthOverflow)?;
            }
        }
        if child_total as usize != nodes[level + 1] {
            return Err(CodecError::InvalidFormat(
                "lane population disagrees with level",
            ));
        }
        if !node_count.is_multiple_of(16) {
            let valid_nibbles = node_count % 16;
            let last = read_u64(image, descriptor.mask + (word_count - 1) * 8)?;
            let used = valid_nibbles * 4;
            if last >> used != 0 {
                return Err(CodecError::InvalidFormat("nonzero mask padding"));
            }
        }
        if nodes[level + 1] == nodes[level].saturating_mul(4) && actual_complete == level {
            actual_complete += 1;
        }
    }
    if complete != actual_complete {
        return Err(CodecError::InvalidFormat("incorrect complete-level count"));
    }

    let bucket_at = trailer_at;
    let bucket_bits = read_u32(image, bucket_at)? as usize;
    let bucket_offset = read_u32(image, bucket_at + 4)? as usize;
    let bucket_bytes = read_u32(image, bucket_at + 8)? as usize;
    let select_offset = read_u32(image, bucket_at + 12)? as usize;
    let select_count = read_u32(image, bucket_at + 16)? as usize;
    let low_offset = read_u32(image, bucket_at + 20)? as usize;
    let low_width = image[bucket_at + 24] as usize;
    let low_bytes = read_u32(image, bucket_at + 28)? as usize;
    if image[bucket_at + 25..bucket_at + 28] != [0, 0, 0] {
        return Err(CodecError::InvalidFormat("nonzero reserved bucket bits"));
    }
    let bucket = offsets.buckets;
    if bucket_bits != bucket.bit_len
        || bucket_offset != bucket.bits
        || bucket_bytes != bucket.bytes
        || select_offset != bucket.select
        || select_count != bucket.select_count
        || low_offset != bucket.low
        || low_width != bucket.low_width
        || low_bytes != bucket.low_bytes
    {
        return Err(CodecError::InvalidFormat("noncanonical bucket offsets"));
    }
    validate_buckets(image, key_count, leaf_count, bucket)?;
    Ok(())
}

fn validate_buckets(
    image: &[u8],
    key_count: usize,
    leaf_count: usize,
    bucket: BucketOffsets,
) -> Result<(), CodecError> {
    let bucket_slice = &image[bucket.bits..bucket.bits + bucket.bytes];
    let low_slice = &image[bucket.low..bucket.low + bucket.low_bytes];
    if !bucket.bit_len.is_multiple_of(8) && !bucket_slice.is_empty() {
        let valid = bucket.bit_len % 8;
        if bucket_slice[bucket.bytes - 1] >> valid != 0 {
            return Err(CodecError::InvalidFormat("nonzero unary padding"));
        }
    }
    let low_bit_len = key_count
        .checked_mul(bucket.low_width)
        .ok_or(CodecError::LengthOverflow)?;
    if low_bit_len % 8 != 0 && !low_slice.is_empty() {
        let valid = low_bit_len % 8;
        if low_slice[low_slice.len() - 1] >> valid != 0 {
            return Err(CodecError::InvalidFormat("nonzero low-bit padding"));
        }
    }
    let mut ones = 0usize;
    let mut zeros = 0usize;
    let mut bucket_start = 0usize;
    let mut previous_low = None;
    let mut sample_index = 0usize;
    for position in 0..bucket.bit_len {
        let one = (bucket_slice[position / 8] >> (position & 7)) & 1 != 0;
        if one {
            let low = read_packed(low_slice, ones * bucket.low_width, bucket.low_width);
            if previous_low.is_some_and(|previous| previous >= low) {
                return Err(CodecError::InvalidFormat(
                    "low suffixes are not strictly ordered",
                ));
            }
            previous_low = Some(low);
            ones += 1;
        } else {
            if ones == bucket_start {
                return Err(CodecError::InvalidFormat("empty trie leaf bucket"));
            }
            if zeros.is_multiple_of(SELECT_STRIDE) {
                let sample = read_u32(image, bucket.select + sample_index * 4)? as usize;
                if sample != position {
                    return Err(CodecError::InvalidFormat("bad select-zero sample"));
                }
                sample_index += 1;
            }
            zeros += 1;
            bucket_start = ones;
            previous_low = None;
        }
    }
    if ones != key_count || zeros != leaf_count || sample_index != bucket.select_count {
        return Err(CodecError::InvalidFormat(
            "unary bucket counts do not match header",
        ));
    }
    Ok(())
}

#[inline]
fn mask_at(image: &[u8], level: usize, node: usize) -> u8 {
    let descriptor = FIXED_HEADER + level * LEVEL_DESCRIPTOR;
    let mask_offset = read_u32_unchecked(image, descriptor + 8) as usize;
    let word_offset = mask_offset + (node / 16) * 8;
    let word = read_u64_unchecked(image, word_offset);
    ((word >> ((node % 16) * 4)) & 0x0f) as u8
}

#[inline]
fn rank_before(image: &[u8], level: usize, node: usize) -> usize {
    let descriptor = FIXED_HEADER + level * LEVEL_DESCRIPTOR;
    let node_count = read_u32_unchecked(image, descriptor) as usize;
    debug_assert!(node < node_count);
    let mask_offset = read_u32_unchecked(image, descriptor + 8) as usize;
    let rank_offset = read_u32_unchecked(image, descriptor + 12) as usize;
    let block = node / RANK_STRIDE;
    let mut total = read_u32_unchecked(image, rank_offset + block * 4) as usize;
    let first_word = block * (RANK_STRIDE / 16);
    let word_index = node / 16;
    for word in first_word..word_index {
        total += read_u64_unchecked(image, mask_offset + word * 8).count_ones() as usize;
    }
    let in_word = node % 16;
    if in_word != 0 {
        let word = read_u64_unchecked(image, mask_offset + word_index * 8);
        let lower = word & ((1u64 << (in_word * 4)) - 1);
        total += lower.count_ones() as usize;
    }
    total
}

#[inline]
fn prefix_mask(width: usize, depth: usize) -> u32 {
    low_mask(width - depth * 2)
}

fn contains_image(image: &[u8], value: u32) -> bool {
    match image[8] {
        20 => contains_image_width::<20>(image, value),
        32 => contains_image_width::<32>(image, value),
        _ => contains_image_width::<0>(image, value),
    }
}

#[inline]
fn contains_image_width<const FIXED_WIDTH: u8>(image: &[u8], value: u32) -> bool {
    let key_count = read_u64_unchecked(image, 12) as usize;
    if key_count == 0 {
        return false;
    }
    let width = if FIXED_WIDTH == 0 {
        image[8] as usize
    } else {
        FIXED_WIDTH as usize
    };
    if width < 32 && value >= (1u32 << width) {
        return false;
    }
    let depth = image[9] as usize;
    let complete = image[10] as usize;
    let leaves = read_u32_unchecked(image, 20) as usize;
    let mut node = prefix_at_depth(value, width as u8, complete) as usize;
    let mut level = complete;
    while level < depth {
        let mask = mask_at(image, level, node);
        let digit = ((value >> (width - (level + 1) * 2)) & 3) as u8;
        let bit = 1u8 << digit;
        if mask & bit == 0 {
            return false;
        }
        node = rank_before(image, level, node) + (mask & (bit - 1)).count_ones() as usize;
        level += 1;
    }
    let leaf = node;
    if leaf >= leaves {
        return false;
    }
    let bucket = bucket_offsets_unchecked(image);
    let (start, end) = bucket_bounds(image, bucket, leaf);
    if start == end {
        return false;
    }
    let target = value & prefix_mask(width, depth);
    if bucket.low_width == 0 {
        return true;
    }
    if let Some(found) = scan_small_bucket(image, bucket, start, end, target) {
        return found;
    }
    let mut low = start;
    let mut high = end;
    while low < high {
        let middle = low + (high - low) / 2;
        let candidate = low_at(image, bucket, middle);
        if candidate < target {
            low = middle + 1;
        } else {
            high = middle;
        }
    }
    low < end && low_at(image, bucket, low) == target
}

fn scan_small_bucket(
    image: &[u8],
    bucket: BucketView,
    start: usize,
    end: usize,
    target: u32,
) -> Option<bool> {
    let len = end - start;
    if !(8..=256).contains(&len) || !bucket.low_width.is_multiple_of(8) {
        return None;
    }
    let byte_width = bucket.low_width / 8;
    if !matches!(byte_width, 1 | 2 | 4) {
        return None;
    }
    let byte_start = start * byte_width;
    let byte_end = end * byte_width;
    let slice = &image[bucket.low + byte_start..bucket.low + byte_end];
    let dispatch = dispatch();
    let scan = match byte_width {
        1 => dispatch.u8,
        2 => dispatch.u16,
        4 => dispatch.u32,
        _ => None,
    }?;
    // The dispatcher only exposes kernels whose required ISA is present.
    Some(unsafe { scan(slice, len, target) })
}

#[derive(Clone, Copy)]
struct BucketView {
    bits: usize,
    bit_len: usize,
    bytes: usize,
    select: usize,
    low: usize,
    low_width: usize,
    low_bytes: usize,
}

fn bucket_offsets_unchecked(image: &[u8]) -> BucketView {
    let trailer = read_u32_unchecked(image, 32) as usize;
    BucketView {
        bit_len: read_u32_unchecked(image, trailer) as usize,
        bits: read_u32_unchecked(image, trailer + 4) as usize,
        bytes: read_u32_unchecked(image, trailer + 8) as usize,
        select: read_u32_unchecked(image, trailer + 12) as usize,
        low: read_u32_unchecked(image, trailer + 20) as usize,
        low_width: image[trailer + 24] as usize,
        low_bytes: read_u32_unchecked(image, trailer + 28) as usize,
    }
}

fn bucket_bounds(image: &[u8], bucket: BucketView, leaf: usize) -> (usize, usize) {
    let end_zero = select_zero(image, bucket, leaf);
    let end = end_zero - leaf;
    let start = if leaf == 0 {
        0
    } else {
        select_zero(image, bucket, leaf - 1) - (leaf - 1)
    };
    (start, end)
}

fn select_zero(image: &[u8], bucket: BucketView, index: usize) -> usize {
    debug_assert!(index < read_u32_unchecked(image, 20) as usize);
    let sample_index = index / SELECT_STRIDE;
    let sample_zero = sample_index * SELECT_STRIDE;
    let sample_position = read_u32_unchecked(image, bucket.select + sample_index * 4) as usize;
    if index == sample_zero {
        return sample_position;
    }
    let mut remaining = index - sample_zero;
    let mut position = sample_position + 1;
    while position < bucket.bit_len {
        let word_start = position & !63;
        let within = position - word_start;
        let word = read_bits_word(&image[bucket.bits..bucket.bits + bucket.bytes], word_start);
        let valid = (64 - within).min(bucket.bit_len - position);
        let valid_mask = if valid == 64 {
            u64::MAX
        } else {
            (1u64 << valid) - 1
        };
        let zeros = !(word >> within) & valid_mask;
        let count = zeros.count_ones() as usize;
        if remaining <= count {
            return position + select_nth_bit(zeros, remaining - 1);
        }
        remaining -= count;
        position = word_start + 64;
    }
    unreachable!("validated unary bucket vector has the requested zero")
}

fn select_nth_bit(mut word: u64, index: usize) -> usize {
    for _ in 0..index {
        word &= word - 1;
    }
    word.trailing_zeros() as usize
}

fn decode_node(
    image: &[u8],
    level: usize,
    node: usize,
    prefix: u32,
    depth: usize,
    next_child: &mut [usize; 17],
    output: &mut Vec<u32>,
) {
    if level == depth {
        let bucket = bucket_offsets_unchecked(image);
        let (start, end) = bucket_bounds(image, bucket, node);
        let low_mask = low_mask(bucket.low_width);
        for index in start..end {
            let low = low_at(image, bucket, index) & low_mask;
            output.push((((prefix as u64) << bucket.low_width) as u32) | low);
        }
        return;
    }
    let mask = mask_at(image, level, node);
    let first_child = next_child[level + 1];
    next_child[level + 1] += mask.count_ones() as usize;
    let mut child_offset = 0usize;
    for digit in 0..4u8 {
        if mask & (1 << digit) == 0 {
            continue;
        }
        let child = first_child + child_offset;
        child_offset += 1;
        decode_node(
            image,
            level + 1,
            child,
            (prefix << 2) | digit as u32,
            depth,
            next_child,
            output,
        );
    }
}

#[inline]
fn low_at(image: &[u8], bucket: BucketView, index: usize) -> u32 {
    read_packed(
        &image[bucket.low..bucket.low + bucket.low_bytes],
        index * bucket.low_width,
        bucket.low_width,
    )
}

fn write_packed(out: &mut [u8], bit_offset: usize, width: usize, value: u32) {
    if width == 0 {
        return;
    }
    let first = bit_offset / 8;
    let shift = bit_offset & 7;
    let mut word = (value as u64) << shift;
    let count = (width + shift).div_ceil(8);
    for byte in 0..count {
        out[first + byte] |= word as u8;
        word >>= 8;
    }
}

fn read_packed(bytes: &[u8], bit_offset: usize, width: usize) -> u32 {
    if width == 0 {
        return 0;
    }
    let first = bit_offset / 8;
    let shift = bit_offset & 7;
    let count = (width + shift).div_ceil(8);
    let mut word = 0u64;
    for byte in 0..count {
        word |= (bytes[first + byte] as u64) << (byte * 8);
    }
    ((word >> shift) as u32) & low_mask(width)
}

#[inline]
fn read_bits_word(bytes: &[u8], bit_offset: usize) -> u64 {
    let byte_offset = bit_offset / 8;
    let mut buffer = [0u8; 8];
    let available = bytes.len().saturating_sub(byte_offset).min(8);
    buffer[..available].copy_from_slice(&bytes[byte_offset..byte_offset + available]);
    u64::from_le_bytes(buffer)
}

fn align_up(value: usize, alignment: usize) -> Option<usize> {
    value
        .checked_add(alignment - 1)
        .map(|value| value & !(alignment - 1))
}

fn put_u16(bytes: &mut [u8], at: usize, value: u16) -> Result<(), CodecError> {
    let target = bytes
        .get_mut(at..at.checked_add(2).ok_or(CodecError::LengthOverflow)?)
        .ok_or(CodecError::LengthOverflow)?;
    target.copy_from_slice(&value.to_le_bytes());
    Ok(())
}

fn put_u32(bytes: &mut [u8], at: usize, value: u32) -> Result<(), CodecError> {
    let target = bytes
        .get_mut(at..at.checked_add(4).ok_or(CodecError::LengthOverflow)?)
        .ok_or(CodecError::LengthOverflow)?;
    target.copy_from_slice(&value.to_le_bytes());
    Ok(())
}

fn put_u64(bytes: &mut [u8], at: usize, value: u64) -> Result<(), CodecError> {
    let target = bytes
        .get_mut(at..at.checked_add(8).ok_or(CodecError::LengthOverflow)?)
        .ok_or(CodecError::LengthOverflow)?;
    target.copy_from_slice(&value.to_le_bytes());
    Ok(())
}

fn read_u16(bytes: &[u8], at: usize) -> Result<u16, CodecError> {
    let end = at.checked_add(2).ok_or(CodecError::LengthOverflow)?;
    let source = bytes.get(at..end).ok_or(CodecError::Truncated)?;
    Ok(u16::from_le_bytes([source[0], source[1]]))
}

fn read_u32(bytes: &[u8], at: usize) -> Result<u32, CodecError> {
    let end = at.checked_add(4).ok_or(CodecError::LengthOverflow)?;
    let source = bytes.get(at..end).ok_or(CodecError::Truncated)?;
    Ok(u32::from_le_bytes([
        source[0], source[1], source[2], source[3],
    ]))
}

fn read_u64(bytes: &[u8], at: usize) -> Result<u64, CodecError> {
    let end = at.checked_add(8).ok_or(CodecError::LengthOverflow)?;
    let source = bytes.get(at..end).ok_or(CodecError::Truncated)?;
    Ok(u64::from_le_bytes([
        source[0], source[1], source[2], source[3], source[4], source[5], source[6], source[7],
    ]))
}

#[inline]
fn read_u32_unchecked(bytes: &[u8], at: usize) -> u32 {
    u32::from_le_bytes([bytes[at], bytes[at + 1], bytes[at + 2], bytes[at + 3]])
}

#[inline]
fn read_u64_unchecked(bytes: &[u8], at: usize) -> u64 {
    u64::from_le_bytes([
        bytes[at],
        bytes[at + 1],
        bytes[at + 2],
        bytes[at + 3],
        bytes[at + 4],
        bytes[at + 5],
        bytes[at + 6],
        bytes[at + 7],
    ])
}

type ScanKernel = unsafe fn(&[u8], usize, u32) -> bool;
type MatchLenKernel = unsafe fn(&[u8], usize, usize) -> usize;

#[derive(Clone, Copy)]
struct Dispatch {
    name: &'static str,
    u8: Option<ScanKernel>,
    u16: Option<ScanKernel>,
    u32: Option<ScanKernel>,
    match_len: MatchLenKernel,
}

static DISPATCH: OnceLock<Dispatch> = OnceLock::new();

fn dispatch() -> &'static Dispatch {
    DISPATCH.get_or_init(detect_dispatch)
}

#[inline]
pub(crate) fn scan_u16_candidates(values: &[u16], needle: u16) -> bool {
    let bytes =
        unsafe { std::slice::from_raw_parts(values.as_ptr().cast::<u8>(), values.len() * 2) };
    match dispatch().u16 {
        Some(kernel) => unsafe { kernel(bytes, values.len(), needle as u32) },
        None => values.contains(&needle),
    }
}

#[inline]
pub(crate) fn extend_match(input: &[u8], candidate: usize, position: usize) -> usize {
    debug_assert!(candidate < position);
    unsafe { (dispatch().match_len)(input, candidate, position) }
}

fn detect_dispatch() -> Dispatch {
    #[cfg(target_arch = "x86_64")]
    {
        if std::is_x86_feature_detected!("avx512f") && std::is_x86_feature_detected!("avx512bw") {
            return Dispatch {
                name: "avx512bw",
                u8: Some(x86_avx512::scan_u8),
                u16: Some(x86_avx512::scan_u16),
                u32: Some(x86_avx512::scan_u32),
                match_len: x86_avx512::match_len,
            };
        }
        if std::is_x86_feature_detected!("avx2") {
            return Dispatch {
                name: "avx2",
                u8: Some(x86_avx2::scan_u8),
                u16: Some(x86_avx2::scan_u16),
                u32: Some(x86_avx2::scan_u32),
                match_len: x86_avx2::match_len,
            };
        }
        if std::is_x86_feature_detected!("sse2") {
            return Dispatch {
                name: "sse2",
                u8: Some(x86_sse2::scan_u8),
                u16: Some(x86_sse2::scan_u16),
                u32: Some(x86_sse2::scan_u32),
                match_len: x86_sse2::match_len,
            };
        }
    }
    #[cfg(target_arch = "aarch64")]
    {
        return Dispatch {
            name: "neon",
            u8: Some(aarch64_neon::scan_u8),
            u16: Some(aarch64_neon::scan_u16),
            u32: Some(aarch64_neon::scan_u32),
            match_len: aarch64_neon::match_len,
        };
    }
    Dispatch {
        name: "scalar",
        u8: None,
        u16: None,
        u32: None,
        match_len: scalar_match_len,
    }
}

fn scalar_match_len(input: &[u8], candidate: usize, position: usize) -> usize {
    let mut length = 0usize;
    while position + length < input.len() && input[candidate + length] == input[position + length] {
        length += 1;
    }
    length
}

#[cfg(target_arch = "x86_64")]
mod x86_avx512 {
    use std::arch::x86_64::*;

    #[target_feature(enable = "avx512f,avx512bw")]
    pub unsafe fn match_len(bytes: &[u8], candidate: usize, position: usize) -> usize {
        let limit = bytes.len() - position;
        let mut length = 0usize;
        while length + 64 <= limit {
            let a = unsafe { _mm512_loadu_si512(bytes.as_ptr().add(candidate + length).cast()) };
            let b = unsafe { _mm512_loadu_si512(bytes.as_ptr().add(position + length).cast()) };
            let different = !_mm512_cmpeq_epi8_mask(a, b);
            if different != 0 {
                return length + different.trailing_zeros() as usize;
            }
            length += 64;
        }
        let remaining = limit - length;
        if remaining != 0 && limit >= 64 {
            let start = limit - 64;
            let a = unsafe { _mm512_loadu_si512(bytes.as_ptr().add(candidate + start).cast()) };
            let b = unsafe { _mm512_loadu_si512(bytes.as_ptr().add(position + start).cast()) };
            let different = !_mm512_cmpeq_epi8_mask(a, b);
            let valid = u64::MAX << (64 - remaining);
            let different = different & valid;
            if different != 0 {
                return start + different.trailing_zeros() as usize;
            }
        } else if remaining != 0 {
            let mut tail = 0usize;
            while tail < remaining
                && bytes[candidate + length + tail] == bytes[position + length + tail]
            {
                tail += 1;
            }
            return length + tail;
        }
        limit
    }

    #[target_feature(enable = "avx512f,avx512bw")]
    pub unsafe fn scan_u8(bytes: &[u8], count: usize, target: u32) -> bool {
        let needle = _mm512_set1_epi8(target as i8);
        let mut index = 0usize;
        while index + 64 <= count {
            let block = unsafe { _mm512_loadu_si512(bytes.as_ptr().add(index).cast()) };
            if _mm512_cmpeq_epi8_mask(block, needle) != 0 {
                return true;
            }
            index += 64;
        }
        bytes[index..count].contains(&(target as u8))
    }

    #[target_feature(enable = "avx512f,avx512bw")]
    pub unsafe fn scan_u16(bytes: &[u8], count: usize, target: u32) -> bool {
        let needle = _mm512_set1_epi16(target as i16);
        let values = bytes.as_ptr().cast::<i16>();
        let mut index = 0usize;
        while index + 32 <= count {
            let block = unsafe { _mm512_loadu_si512(values.add(index).cast()) };
            if _mm512_cmpeq_epi16_mask(block, needle) != 0 {
                return true;
            }
            index += 32;
        }
        let tail = unsafe {
            std::slice::from_raw_parts(values.add(index).cast::<u8>(), (count - index) * 2)
        };
        tail.as_chunks::<2>()
            .0
            .iter()
            .any(|pair| u16::from_le_bytes(*pair) == target as u16)
    }

    #[target_feature(enable = "avx512f")]
    pub unsafe fn scan_u32(bytes: &[u8], count: usize, target: u32) -> bool {
        let needle = _mm512_set1_epi32(target as i32);
        let values = bytes.as_ptr().cast::<i32>();
        let mut index = 0usize;
        while index + 16 <= count {
            let block = unsafe { _mm512_loadu_si512(values.add(index).cast()) };
            if _mm512_cmpeq_epi32_mask(block, needle) != 0 {
                return true;
            }
            index += 16;
        }
        let tail = unsafe {
            std::slice::from_raw_parts(values.add(index).cast::<u8>(), (count - index) * 4)
        };
        tail.as_chunks::<4>()
            .0
            .iter()
            .any(|word| u32::from_le_bytes(*word) == target)
    }
}

/// Plain minimum-relative Elias–Fano reference used by the benchmark suite.
#[derive(Debug)]
pub struct PlainEf {
    len: usize,
    base: u32,
    low_width: usize,
    high_bit_len: usize,
    high: Vec<u8>,
    low: Vec<u8>,
    zero_samples: Vec<u32>,
}

impl PlainEf {
    pub fn encode(sorted_values: &[u32]) -> Result<Self, CodecError> {
        validate_input(sorted_values, 32)?;
        if sorted_values.is_empty() {
            return Ok(Self {
                len: 0,
                base: 0,
                low_width: 0,
                high_bit_len: 0,
                high: Vec::new(),
                low: Vec::new(),
                zero_samples: Vec::new(),
            });
        }
        let base = sorted_values[0];
        let universe =
            u64::from(*sorted_values.last().expect("nonempty input")) - u64::from(base) + 1;
        let ratio = universe / sorted_values.len() as u64;
        let low_width = (u64::BITS - 1 - ratio.leading_zeros()) as usize;
        let bucket_width = 1u64 << low_width;
        let bucket_count = usize::try_from(universe.div_ceil(bucket_width))
            .map_err(|_| CodecError::LengthOverflow)?;
        let high_bit_len = bucket_count
            .checked_add(sorted_values.len())
            .ok_or(CodecError::LengthOverflow)?;
        if high_bit_len > u32::MAX as usize {
            return Err(CodecError::LengthOverflow);
        }
        let mut high = vec![0u8; high_bit_len.div_ceil(8)];
        let low_bit_len = sorted_values
            .len()
            .checked_mul(low_width)
            .ok_or(CodecError::LengthOverflow)?;
        let mut low = vec![0u8; low_bit_len.div_ceil(8)];
        for (index, value) in sorted_values.iter().copied().enumerate() {
            let offset = value - base;
            let high_value = (u64::from(offset) >> low_width) as usize;
            let bit = high_value + index;
            high[bit / 8] |= 1 << (bit & 7);
            write_packed(
                &mut low,
                index * low_width,
                low_width,
                offset & low_mask(low_width),
            );
        }
        let zero_samples = sample_zero_positions(&high, high_bit_len);
        Ok(Self {
            len: sorted_values.len(),
            base,
            low_width,
            high_bit_len,
            high,
            low,
            zero_samples,
        })
    }

    #[inline]
    pub fn len(&self) -> usize {
        self.len
    }

    #[inline]
    pub fn is_empty(&self) -> bool {
        self.len == 0
    }

    pub fn encoded_size(&self) -> usize {
        13 + self.high.len() + self.low.len() + self.zero_samples.len() * 4
    }

    pub fn contains(&self, value: u32) -> bool {
        if self.len == 0 {
            return false;
        }
        let Some(offset) = value.checked_sub(self.base) else {
            return false;
        };
        let high_value = (u64::from(offset) >> self.low_width) as usize;
        let bucket_count = self.high_bit_len - self.len;
        if high_value >= bucket_count {
            return false;
        }
        let end_zero = select_ef_zero(
            &self.high,
            self.high_bit_len,
            &self.zero_samples,
            high_value,
        );
        let start = if high_value == 0 {
            0
        } else {
            select_ef_zero(
                &self.high,
                self.high_bit_len,
                &self.zero_samples,
                high_value - 1,
            ) - (high_value - 1)
        };
        let end = end_zero - high_value;
        let target = offset & low_mask(self.low_width);
        let mut lo = start;
        let mut hi = end;
        while lo < hi {
            let middle = lo + (hi - lo) / 2;
            let candidate = read_packed(&self.low, middle * self.low_width, self.low_width);
            if candidate < target {
                lo = middle + 1;
            } else {
                hi = middle;
            }
        }
        lo < end && read_packed(&self.low, lo * self.low_width, self.low_width) == target
    }

    pub fn decode(&self) -> Vec<u32> {
        let mut output = Vec::with_capacity(self.len);
        let mut index = 0usize;
        for (byte_index, byte) in self.high.iter().copied().enumerate() {
            let mut bits = byte;
            while bits != 0 {
                let bit = bits.trailing_zeros() as usize;
                let position = byte_index * 8 + bit;
                if position < self.high_bit_len {
                    let high_value = position - index;
                    let low = read_packed(&self.low, index * self.low_width, self.low_width);
                    let offset = ((high_value as u64) << self.low_width) | u64::from(low);
                    output.push((u64::from(self.base) + offset) as u32);
                    index += 1;
                }
                bits &= bits - 1;
            }
        }
        output
    }
}

fn sample_zero_positions(bits: &[u8], bit_len: usize) -> Vec<u32> {
    let count = bit_len.saturating_sub(
        (0..bit_len)
            .filter(|position| (bits[position / 8] >> (position & 7)) & 1 != 0)
            .count(),
    );
    let mut samples = Vec::with_capacity(count.div_ceil(SELECT_STRIDE));
    let mut zero = 0usize;
    for position in 0..bit_len {
        if (bits[position / 8] >> (position & 7)) & 1 == 0 {
            if zero.is_multiple_of(SELECT_STRIDE) {
                samples.push(position as u32);
            }
            zero += 1;
        }
    }
    samples
}

fn select_ef_zero(bits: &[u8], bit_len: usize, samples: &[u32], index: usize) -> usize {
    let sample_index = index / SELECT_STRIDE;
    let sample_zero = sample_index * SELECT_STRIDE;
    let sample_position = samples[sample_index] as usize;
    if index == sample_zero {
        return sample_position;
    }
    let mut remaining = index - sample_zero;
    let mut position = sample_position + 1;
    while position < bit_len {
        if (bits[position / 8] >> (position & 7)) & 1 == 0 {
            remaining -= 1;
            if remaining == 0 {
                return position;
            }
        }
        position += 1;
    }
    unreachable!("validated EF vector has the requested zero")
}

#[cfg(test)]
mod tests {
    use super::*;

    fn roundtrip(values: &[u32]) {
        let encoded = Rnbw::encode(values).unwrap();
        assert_eq!(encoded.len(), values.len());
        assert_eq!(encoded.decode(), values);
        assert_eq!(
            Rnbw::from_bytes(encoded.as_bytes()).unwrap().decode(),
            values
        );
        let view = RnbwView::from_bytes(encoded.as_bytes()).unwrap();
        assert_eq!(view.as_bytes().as_ptr(), encoded.as_bytes().as_ptr());
        assert_eq!(view.decode(), values);
        for probe in values.iter().copied().chain([0, u32::MAX, 1, 0x1234_5678]) {
            assert_eq!(
                encoded.contains(probe),
                values.binary_search(&probe).is_ok(),
                "{probe}"
            );
        }
    }

    #[test]
    fn empty_singleton_and_unsorted_inputs() {
        roundtrip(&[]);
        roundtrip(&[u32::MAX]);
        let encoded = Rnbw::encode_unsorted(&[4, 1, 4, 0, 2]).unwrap();
        assert_eq!(encoded.decode(), [0, 1, 2, 4]);
        assert!(matches!(
            Rnbw::encode(&[1, 1]),
            Err(CodecError::UnsortedInput { index: 1 })
        ));
    }

    #[test]
    fn deterministic_random_and_clustered_roundtrips() {
        let mut seed = 0x4d59_5df4_d0f3_3173u64;
        let mut values = Vec::new();
        for _ in 0..100_000 {
            seed ^= seed << 13;
            seed ^= seed >> 7;
            seed ^= seed << 17;
            if seed & 3 == 0 {
                values.push((seed >> 16) as u32);
            }
        }
        values.sort_unstable();
        values.dedup();
        roundtrip(&values);

        let clustered: Vec<u32> = (0..1024)
            .flat_map(|run| (0..32).map(move |v| run * 65536 + v))
            .collect();
        roundtrip(&clustered);
    }

    #[test]
    fn one_pass_prefix_counts_match_direct_prefix_sets() {
        let mut seed = 0x9e37_79b9_7f4a_7c15u64;
        for width in (2..=32).step_by(2) {
            let mask = low_mask(width as usize);
            for _ in 0..24 {
                let mut values = Vec::with_capacity(512);
                for _ in 0..1024 {
                    seed ^= seed << 13;
                    seed ^= seed >> 7;
                    seed ^= seed << 17;
                    values.push((seed as u32) & mask);
                }
                values.sort_unstable();
                values.dedup();
                let max_depth = (width / 2) as usize;
                let mut counts = [0usize; 17];
                count_prefixes_by_level(&values, width, max_depth, &mut counts[..=max_depth]);
                for (depth, &count) in counts.iter().take(max_depth + 1).enumerate() {
                    let shift = width as usize - 2 * depth;
                    let expected = if values.is_empty() {
                        0
                    } else if depth == 0 {
                        1
                    } else {
                        values
                            .iter()
                            .map(|value| value >> shift)
                            .collect::<std::collections::BTreeSet<_>>()
                            .len()
                    };
                    assert_eq!(count, expected, "width={width}, depth={depth}");
                }
            }
        }
    }

    #[test]
    fn width_specific_chunks_roundtrip() {
        let values: Vec<u32> = (0..=u16::MAX as u32).step_by(7).collect();
        let encoded = Rnbw::encode_width(&values, 16).unwrap();
        assert_eq!(encoded.decode(), values);
        for value in 0..=u16::MAX as u32 {
            assert_eq!(
                encoded.contains(value),
                values.binary_search(&value).is_ok()
            );
        }
        assert!(matches!(
            Rnbw::encode_width(&[1 << 16], 16),
            Err(CodecError::ValueOutOfRange { .. })
        ));
    }

    #[test]
    fn rejects_corrupt_headers_directories_and_streams() {
        let encoded = Rnbw::encode(&[1, 2, 7, 11, 1000, u32::MAX]).unwrap();
        for end in 0..encoded.as_bytes().len() {
            assert!(
                Rnbw::from_bytes(&encoded.as_bytes()[..end]).is_err(),
                "prefix {end}"
            );
        }
        let mut bad = encoded.as_bytes().to_vec();
        bad[0] ^= 1;
        assert!(Rnbw::from_bytes(&bad).is_err());
        let mut bad = encoded.as_bytes().to_vec();
        bad[24] ^= 1;
        assert!(Rnbw::from_bytes(&bad).is_err());
        let mut bad = encoded.as_bytes().to_vec();
        let depth = bad[9] as usize;
        if depth > 0 {
            let rank_offset = read_u32_unchecked(&bad, FIXED_HEADER + 12) as usize;
            bad[rank_offset] ^= 1;
            assert!(Rnbw::from_bytes(&bad).is_err());
        }
    }

    #[test]
    fn plain_ef_reference_is_correct() {
        let values = [0, 1, 17, 65_535, 1 << 20, 0x7fff_ffff, u32::MAX];
        let ef = PlainEf::encode(&values).unwrap();
        assert_eq!(ef.decode(), values);
        for value in [0, 2, 17, 18, 65_535, 65_536, u32::MAX] {
            assert_eq!(ef.contains(value), values.binary_search(&value).is_ok());
        }
    }

    #[test]
    fn plain_ef_normalizes_high_offset_dense_ranges() {
        let values: Vec<u32> = (0..100_000).map(|value| 0x6000_0000 + value).collect();
        let ef = PlainEf::encode(&values).unwrap();
        assert_eq!(ef.decode(), values);
        assert!(ef.encoded_size() < values.len());
        for value in [
            0x5fff_ffff,
            values[0],
            values[1],
            values[50_000],
            *values.last().unwrap(),
            0x6001_86a0,
        ] {
            assert_eq!(ef.contains(value), values.binary_search(&value).is_ok());
        }
    }

    #[test]
    fn membership_kernel_is_selected_once() {
        assert_eq!(Rnbw::simd_backend(), Rnbw::simd_backend());
        assert!(matches!(
            Rnbw::simd_backend(),
            "avx512bw" | "avx2" | "sse2" | "neon" | "scalar"
        ));
    }

    #[test]
    fn selected_simd_kernels_match_scalar_for_hits_misses_and_tails() {
        let selected = dispatch();
        let cases = [
            (1usize, 97usize, selected.u8, 255u32),
            (2usize, 47usize, selected.u16, 65_535u32),
            (4usize, 27usize, selected.u32, u32::MAX),
        ];
        for (width, count, kernel, miss) in cases {
            let Some(kernel) = kernel else {
                continue;
            };
            let mut bytes = Vec::with_capacity(width * count);
            for index in 0..count {
                let value = match width {
                    1 => (index * 3) as u32,
                    2 => (index * 73) as u32,
                    _ => (index * 65_537) as u32,
                };
                match width {
                    1 => bytes.push(value as u8),
                    2 => bytes.extend_from_slice(&(value as u16).to_le_bytes()),
                    _ => bytes.extend_from_slice(&value.to_le_bytes()),
                }
            }
            let decode = |index: usize| -> u32 {
                let at = index * width;
                match width {
                    1 => bytes[at] as u32,
                    2 => u16::from_le_bytes([bytes[at], bytes[at + 1]]) as u32,
                    _ => {
                        u32::from_le_bytes([bytes[at], bytes[at + 1], bytes[at + 2], bytes[at + 3]])
                    }
                }
            };
            for target in [decode(0), decode(count / 2), decode(count - 1), miss] {
                let scalar = (0..count).any(|index| decode(index) == target);
                assert_eq!(unsafe { kernel(&bytes, count, target) }, scalar);
            }
        }
    }

    #[test]
    fn selected_match_length_kernel_matches_scalar_for_vector_tails() {
        let mut seed = 0x243f_6a88_85a3_08d3u64;
        for len in 2..=513 {
            let mut bytes = vec![0u8; len];
            for byte in &mut bytes {
                seed ^= seed << 13;
                seed ^= seed >> 7;
                seed ^= seed << 17;
                *byte = (seed >> 24) as u8;
            }

            // Exercise short tails, exact SIMD-width endings, long matches,
            // and mismatches at each possible vector boundary.
            for position in 1..len {
                let candidate = position - 1;
                let expected = scalar_match_len(&bytes, candidate, position);
                assert_eq!(
                    extend_match(&bytes, candidate, position),
                    expected,
                    "len={len}, candidate={candidate}, position={position}"
                );
            }

            if len > 64 {
                for distance in [1usize, 2, 3, 7, 15, 16, 31, 32, 63, 64] {
                    if distance >= len {
                        continue;
                    }
                    let position = distance;
                    let candidate = 0;
                    let expected = scalar_match_len(&bytes, candidate, position);
                    assert_eq!(
                        extend_match(&bytes, candidate, position),
                        expected,
                        "periodic len={len}, distance={distance}"
                    );
                }
            }
        }
    }

//! L2 GenTable + VAR Arena storage with explicitly promoted L3 direct pockets.
//! The 32-bit universe is split into 65,536 16-bit chunks and groups of 16.

use crate::{CodecError, Rnbw, RnbwView, TrnMap, TrnMapError, scan_u16_candidates};
use std::fmt;

pub const CHUNK_COUNT: usize = 1 << 16;
pub const GROUP_COUNT: usize = 1 << 12;
const CHUNKS_PER_GROUP: usize = 16;
const GROUP_HEADER: usize = CHUNKS_PER_GROUP * 3;
const ABSENT: u32 = u32::MAX;
const PAGE_BYTES: usize = 1 << 16;
const MAX_PAGE_BYTES: usize = 1 << 24;
const ARENA_MIN_BLOCK_LOG2: usize = 3;
const ARENA_SIZE_CLASSES: usize =
    MAX_PAGE_BYTES.trailing_zeros() as usize - ARENA_MIN_BLOCK_LOG2 + 1;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum PocketLayout {
    PerChunk32,
    PerPair64,
}

#[derive(Debug)]
pub enum StoreError {
    Unsorted { index: usize },
    ChunkOutOfRange(usize),
    RangeOutOfRange { start: usize, end: usize },
    TooLarge,
    InvalidRecord,
    Codec(CodecError),
    TrnMap(TrnMapError),
}
impl fmt::Display for StoreError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "RNBW store error: {self:?}")
    }
}
impl std::error::Error for StoreError {}
impl From<CodecError> for StoreError {
    fn from(e: CodecError) -> Self {
        Self::Codec(e)
    }
}
impl From<TrnMapError> for StoreError {
    fn from(e: TrnMapError) -> Self {
        Self::TrnMap(e)
    }
}

#[repr(u8)]
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Form {
    Empty = 0,
    Full = 1,
    Inline1 = 2,
    Quot = 3,
    Runs = 4,
    Bitmap = 5,
}
impl Form {
    fn from(v: u8) -> Result<Self, StoreError> {
        match v {
            0 => Ok(Self::Empty),
            1 => Ok(Self::Full),
            2 => Ok(Self::Inline1),
            3 => Ok(Self::Quot),
            4 => Ok(Self::Runs),
            5 => Ok(Self::Bitmap),
            _ => Err(StoreError::InvalidRecord),
        }
    }
}

#[derive(Default)]
struct VarPage {
    words: Box<[u64]>,
    bump: usize,
    generation: u32,
}
impl VarPage {
    fn new(bytes: usize) -> Self {
        Self {
            words: vec![0; bytes.div_ceil(8)].into_boxed_slice(),
            bump: 0,
            generation: 1,
        }
    }
    fn len(&self) -> usize {
        self.words.len() * 8
    }
    fn bytes(&self) -> &[u8] {
        unsafe { std::slice::from_raw_parts(self.words.as_ptr().cast(), self.len()) }
    }
    fn bytes_mut(&mut self) -> &mut [u8] {
        let len = self.len();
        unsafe { std::slice::from_raw_parts_mut(self.words.as_mut_ptr().cast(), len) }
    }
}
#[derive(Default)]
struct VarArena {
    pages: Vec<VarPage>,
    free: Vec<Vec<(u32, u32)>>,
    free_mask: u32,
    live_bytes: usize,
    reserved_bytes: usize,
}
impl VarArena {
    fn alloc(&mut self, record: &[u8]) -> Result<u64, StoreError> {
        let need = (record.len().checked_add(8).ok_or(StoreError::TooLarge)?)
            .max(8)
            .next_power_of_two();
        if need > MAX_PAGE_BYTES {
            return Err(StoreError::TooLarge);
        }
        if self.free.len() != ARENA_SIZE_CLASSES {
            self.free.resize_with(ARENA_SIZE_CLASSES, Vec::new);
        }
        let min_class = need.trailing_zeros() as usize - ARENA_MIN_BLOCK_LOG2;
        let reusable = self.free_mask & (u32::MAX << min_class);
        let (page, off, block) = if reusable != 0 {
            let mut class = reusable.trailing_zeros() as usize;
            let (p, o) = self.free[class].pop().unwrap();
            if self.free[class].is_empty() {
                self.free_mask &= !(1u32 << class);
            }
            let mut block = 1usize << (class + ARENA_MIN_BLOCK_LOG2);
            while block > need {
                block >>= 1;
                class -= 1;
                self.free[class].push((p, o + block as u32));
                self.free_mask |= 1u32 << class;
            }
            (p, o, need)
        } else {
            let p = if self
                .pages
                .last()
                .is_some_and(|page| page.len() - page.bump >= need)
            {
                self.pages.len() - 1
            } else {
                if self.pages.len() >= 1 << 20 {
                    return Err(StoreError::TooLarge);
                }
                let cap = PAGE_BYTES.max(need);
                self.pages.push(VarPage::new(cap));
                self.reserved_bytes += cap;
                self.pages.len() - 1
            };
            let off = self.pages[p].bump;
            self.pages[p].bump += need;
            (p as u32, off as u32, need)
        };
        let p = &mut self.pages[page as usize];
        let raw =
            (u64::from(p.generation & 0xfffff) << 44) | (u64::from(page) << 24) | u64::from(off);
        let b = p.bytes_mut();
        b[off as usize..off as usize + 4].copy_from_slice(&(block as u32).to_le_bytes());
        b[off as usize + 4..off as usize + 8].copy_from_slice(&(record.len() as u32).to_le_bytes());
        b[off as usize + 8..off as usize + 8 + record.len()].copy_from_slice(record);
        self.live_bytes += record.len() + 8;
        Ok(raw)
    }
    fn parts(&self, h: u64) -> Result<(usize, usize, usize), StoreError> {
        let generation = (h >> 44) as u32;
        let page = ((h >> 24) & 0xfffff) as usize;
        let off = (h & 0xffffff) as usize;
        let p = self.pages.get(page).ok_or(StoreError::InvalidRecord)?;
        if p.generation & 0xfffff != generation {
            return Err(StoreError::InvalidRecord);
        }
        let b = p.bytes();
        if off + 8 > b.len() {
            return Err(StoreError::InvalidRecord);
        }
        let block = u32::from_le_bytes(b[off..off + 4].try_into().unwrap()) as usize;
        let len = u32::from_le_bytes(b[off + 4..off + 8].try_into().unwrap()) as usize;
        if off + 8 + len > b.len() || block < 8 + len {
            return Err(StoreError::InvalidRecord);
        }
        Ok((page, off, len))
    }
    fn get(&self, h: u64) -> Result<&[u8], StoreError> {
        let (p, o, n) = self.parts(h)?;
        Ok(&self.pages[p].bytes()[o + 8..o + 8 + n])
    }
    fn retire(&mut self, h: u64) -> Result<(), StoreError> {
        let (p, o, n) = self.parts(h)?;
        let b = self.pages[p].bytes();
        let block = u32::from_le_bytes(b[o..o + 4].try_into().unwrap()) as usize;
        let class = block.trailing_zeros() as usize - ARENA_MIN_BLOCK_LOG2;
        self.free[class].push((p as u32, o as u32));
        self.free_mask |= 1u32 << class;
        self.live_bytes -= n + 8;
        Ok(())
    }
}

struct BitmapSlot {
    words: Box<[u64; 1024]>,
    generation: u32,
    live: bool,
}
#[derive(Default)]
struct BitmapArena {
    slots: Vec<BitmapSlot>,
    handles: Vec<u64>,
    free: Vec<u32>,
}
impl BitmapArena {
    fn alloc(&mut self, values: &[u16]) -> Result<u16, StoreError> {
        let index = if let Some(i) = self.free.pop() {
            i as usize
        } else {
            if self.slots.len() >= u16::MAX as usize {
                return Err(StoreError::TooLarge);
            }
            self.slots.push(BitmapSlot {
                words: Box::new([0; 1024]),
                generation: 1,
                live: false,
            });
            self.slots.len() - 1
        };
        let s = &mut self.slots[index];
        s.words.fill(0);
        for &v in values {
            s.words[v as usize / 64] |= 1u64 << (v & 63)
        }
        s.live = true;
        let generation = s.generation;
        if self.handles.len() <= index {
            self.handles.resize(index + 1, 0);
        }
        self.handles[index] = (u64::from(generation) << 32) | index as u64;
        Ok(index as u16)
    }
    fn contains(&self, index: u16, value: u16) -> Result<bool, StoreError> {
        let s = self
            .slots
            .get(index as usize)
            .ok_or(StoreError::InvalidRecord)?;
        if !s.live
            || self.handles.get(index as usize).copied()
                != Some((u64::from(s.generation) << 32) | u64::from(index))
        {
            return Err(StoreError::InvalidRecord);
        }
        Ok(s.words[value as usize / 64] & (1u64 << (value & 63)) != 0)
    }
    fn decode(&self, index: u16, out: &mut Vec<u16>) -> Result<(), StoreError> {
        let s = self
            .slots
            .get(index as usize)
            .ok_or(StoreError::InvalidRecord)?;
        if !s.live
            || self.handles.get(index as usize).copied()
                != Some((u64::from(s.generation) << 32) | u64::from(index))
        {
            return Err(StoreError::InvalidRecord);
        }
        for (wi, &mut_word) in s.words.iter().enumerate() {
            let mut w = mut_word;
            while w != 0 {
                let b = w.trailing_zeros() as usize;
                out.push((wi * 64 + b) as u16);
                w &= w - 1;
            }
        }
        Ok(())
    }
    fn retire(&mut self, index: u16) -> Result<(), StoreError> {
        let i = index as usize;
        let s = self.slots.get_mut(i).ok_or(StoreError::InvalidRecord)?;
        if !s.live {
            return Err(StoreError::InvalidRecord);
        }
        s.live = false;
        s.generation = s.generation.wrapping_add(1).max(1);
        self.free.push(index as u32);
        Ok(())
    }
}

pub struct RnbwStore {
    tiers: TrnMap<'static>,
    gen_table: Box<[u64]>,
    group_handles: Box<[u64]>,
    arena: VarArena,
    bitmaps: BitmapArena,
    layout: PocketLayout,
    pockets: Option<Box<[u8]>>,
    len: usize,
}

impl RnbwStore {
    pub fn new(layout: PocketLayout) -> Result<Self, StoreError> {
        Ok(Self {
            tiers: TrnMap::new(CHUNK_COUNT)?,
            gen_table: vec![u64::from(ABSENT); GROUP_COUNT].into_boxed_slice(),
            group_handles: vec![0; GROUP_COUNT].into_boxed_slice(),
            arena: VarArena::default(),
            bitmaps: BitmapArena::default(),
            layout,
            pockets: None,
            len: 0,
        })
    }
    pub fn from_sorted(values: &[u32], layout: PocketLayout) -> Result<Self, StoreError> {
        if let Some(i) = values.windows(2).position(|w| w[0] >= w[1]) {
            return Err(StoreError::Unsorted { index: i + 1 });
        }
        let mut s = Self::new(layout)?;
        s.len = values.len();
        let mut at = 0;
        while at < values.len() {
            let gid = (values[at] >> 20) as usize;
            let mut chunks: [Vec<u16>; 16] = std::array::from_fn(|_| Vec::new());
            while at < values.len() && (values[at] >> 20) as usize == gid {
                let v = values[at];
                chunks[((v >> 16) & 15) as usize].push(v as u16);
                at += 1;
            }
            s.replace_group(gid, &chunks)?;
            for (i, c) in chunks.iter().enumerate() {
                if !c.is_empty() {
                    s.set_tier((gid << 4) + i, 0)?;
                }
            }
        }
        Ok(s)
    }
    pub fn len(&self) -> usize {
        self.len
    }
    pub fn is_empty(&self) -> bool {
        self.len == 0
    }
    pub fn simd_backend() -> &'static str {
        Rnbw::simd_backend()
    }
    pub fn l2_bytes(&self) -> usize {
        std::mem::size_of::<Self>()
            + self.arena.reserved_bytes
            + self.arena.pages.capacity() * std::mem::size_of::<VarPage>()
            + self.arena.free.capacity() * std::mem::size_of::<Vec<(u32, u32)>>()
            + self
                .arena
                .free
                .iter()
                .map(|blocks| blocks.capacity() * std::mem::size_of::<(u32, u32)>())
                .sum::<usize>()
            + self.bitmaps.slots.len() * 8192
            + self.bitmaps.slots.capacity() * std::mem::size_of::<BitmapSlot>()
            + self.bitmaps.handles.capacity() * std::mem::size_of::<u64>()
            + self.bitmaps.free.capacity() * std::mem::size_of::<u32>()
            + self.gen_table.len() * 8
            + self.group_handles.len() * 8
            + self.tiers.storage_bytes()
    }
    pub fn l3_bytes(&self) -> usize {
        self.pockets.as_ref().map_or(0, |p| p.len())
    }
    fn group_handle(&self, g: usize) -> Option<u64> {
        let cell = self.gen_table[g];
        if cell as u32 == ABSENT {
            None
        } else {
            Some(self.group_handles[g])
        }
    }
    fn set_tier(&mut self, chunk: usize, tier: i8) -> Result<(), StoreError> {
        if self.tiers.get(chunk)? != tier {
            self.tiers.set(chunk, tier)?;
            self.tiers.mark(chunk)?;
        }
        Ok(())
    }
    fn descriptor(record: &[u8], chunk: usize) -> Result<(Form, u16), StoreError> {
        let p = chunk * 3;
        Ok((
            Form::from(*record.get(p).ok_or(StoreError::InvalidRecord)?)?,
            u16::from_le_bytes(
                record
                    .get(p + 1..p + 3)
                    .ok_or(StoreError::InvalidRecord)?
                    .try_into()
                    .unwrap(),
            ),
        ))
    }
    fn contains_l2(&self, chunk: usize, suffix: u16) -> Result<bool, StoreError> {
        let g = chunk >> 4;
        let Some(h) = self.group_handle(g) else {
            return Ok(false);
        };
        let rec = self.arena.get(h)?;
        let (f, o) = Self::descriptor(rec, chunk & 15)?;
        match f {
            Form::Empty => Ok(false),
            Form::Full => Ok(true),
            Form::Inline1 => Ok(o == suffix),
            Form::Quot => {
                let off = o as usize;
                let n = u16::from_le_bytes(
                    rec.get(off..off + 2)
                        .ok_or(StoreError::InvalidRecord)?
                        .try_into()
                        .unwrap(),
                ) as usize;
                let v = RnbwView::from_validated_bytes(
                    rec.get(off + 2..off + 2 + n)
                        .ok_or(StoreError::InvalidRecord)?,
                );
                Ok(v.contains((((chunk & 15) as u32) << 16) | u32::from(suffix)))
            }
            Form::Runs => {
                let off = o as usize;
                let count = u16::from_le_bytes(
                    rec.get(off..off + 2)
                        .ok_or(StoreError::InvalidRecord)?
                        .try_into()
                        .unwrap(),
                ) as usize;
                let mut lo = 0;
                let mut hi = count;
                while lo < hi {
                    let m = (lo + hi) / 2;
                    let p = off + 2 + m * 4;
                    let a = u16::from_le_bytes(
                        rec.get(p..p + 2)
                            .ok_or(StoreError::InvalidRecord)?
                            .try_into()
                            .unwrap(),
                    );
                    if a <= suffix { lo = m + 1 } else { hi = m }
                }
                if lo == 0 {
                    return Ok(false);
                }
                let p = off + 2 + (lo - 1) * 4;
                let a = u16::from_le_bytes(rec[p..p + 2].try_into().unwrap());
                let n = u16::from_le_bytes(rec[p + 2..p + 4].try_into().unwrap());
                Ok(u32::from(suffix) < u32::from(a) + u32::from(n))
            }
            Form::Bitmap => self.bitmaps.contains(o, suffix),
        }
    }
    pub fn contains(&self, key: u32) -> Result<bool, StoreError> {
        let c = (key >> 16) as usize;
        match self.tiers.get(c)? {
            -1 => Ok(false),
            1 => self.pocket_contains(c, key as u16),
            _ => self.contains_l2(c, key as u16),
        }
    }
    fn decode_group(&self, g: usize) -> Result<[Vec<u16>; 16], StoreError> {
        let mut out: [Vec<u16>; 16] = std::array::from_fn(|_| Vec::new());
        let Some(h) = self.group_handle(g) else {
            return Ok(out);
        };
        let rec = self.arena.get(h)?;
        let mut qdone = false;
        for i in 0..16 {
            let (f, o) = Self::descriptor(rec, i)?;
            match f {
                Form::Empty => {}
                Form::Full => out[i].extend(0..=u16::MAX),
                Form::Inline1 => out[i].push(o),
                Form::Runs => {
                    let p = o as usize;
                    let n = u16::from_le_bytes(
                        rec.get(p..p + 2)
                            .ok_or(StoreError::InvalidRecord)?
                            .try_into()
                            .unwrap(),
                    ) as usize;
                    for j in 0..n {
                        let q = p + 2 + j * 4;
                        let a = u16::from_le_bytes(
                            rec.get(q..q + 2)
                                .ok_or(StoreError::InvalidRecord)?
                                .try_into()
                                .unwrap(),
                        );
                        let l = u16::from_le_bytes(
                            rec.get(q + 2..q + 4)
                                .ok_or(StoreError::InvalidRecord)?
                                .try_into()
                                .unwrap(),
                        );
                        out[i]
                            .extend((u32::from(a)..u32::from(a) + u32::from(l)).map(|x| x as u16));
                    }
                }
                Form::Bitmap => self.bitmaps.decode(o, &mut out[i])?,
                Form::Quot => {
                    if !qdone {
                        qdone = true;
                        let p = o as usize;
                        let n = u16::from_le_bytes(
                            rec.get(p..p + 2)
                                .ok_or(StoreError::InvalidRecord)?
                                .try_into()
                                .unwrap(),
                        ) as usize;
                        let v = RnbwView::from_validated_bytes(
                            rec.get(p + 2..p + 2 + n).ok_or(StoreError::InvalidRecord)?,
                        );
                        for x in v.decode() {
                            let chunk = ((x >> 16) & 15) as usize;
                            out[chunk].push(x as u16);
                        }
                    }
                }
            }
        }
        Ok(out)
    }
    fn chunk_values(&self, chunk: usize) -> Result<Vec<u16>, StoreError> {
        let mut a = self.decode_group(chunk >> 4)?;
        Ok(std::mem::take(&mut a[chunk & 15]))
    }
    fn replace_group(&mut self, g: usize, chunks: &[Vec<u16>; 16]) -> Result<(), StoreError> {
        let old = self.group_handle(g);
        let mut rec = vec![0u8; GROUP_HEADER];
        let mut forms = [Form::Empty; 16];
        let mut bitmap_new = Vec::new();
        for i in 0..16 {
            let v = &chunks[i];
            if v.is_empty() {
                continue;
            }
            if v.len() == 65536 {
                forms[i] = Form::Full;
                put_desc(&mut rec, i, Form::Full, 0)?;
                continue;
            }
            if v.len() == 1 {
                forms[i] = Form::Inline1;
                put_desc(&mut rec, i, Form::Inline1, v[0])?;
                continue;
            }
            let runs = make_runs(v);
            let run_bytes = 2 + runs.len() * 4;
            let qbits = (v.len() as f64 * (18.0 - (v.len() as f64).log2() + 1.0)).ceil() as usize;
            let qbytes = qbits.div_ceil(8);
            if 8192 < qbytes && 8192 < run_bytes {
                forms[i] = Form::Bitmap
            } else if run_bytes < qbytes {
                forms[i] = Form::Runs
            } else {
                forms[i] = Form::Quot
            }
            if forms[i] == Form::Runs {
                let off = append_runs(&mut rec, &runs)?;
                put_desc(&mut rec, i, Form::Runs, off)?;
            }
        }
        let mut keys = Vec::new();
        for i in 0..16 {
            if forms[i] == Form::Quot {
                for &x in &chunks[i] {
                    keys.push(((i as u32) << 16) | u32::from(x));
                }
            }
        }
        if !keys.is_empty() {
            let encoded = Rnbw::encode_width(&keys, 20)?;
            let image = encoded.as_bytes();
            let start = u16::try_from(rec.len()).map_err(|_| StoreError::TooLarge)?;
            let n = u16::try_from(image.len()).map_err(|_| StoreError::TooLarge)?;
            rec.extend_from_slice(&n.to_le_bytes());
            rec.extend_from_slice(image);
            for i in 0..16 {
                if forms[i] == Form::Quot {
                    put_desc(&mut rec, i, Form::Quot, start)?;
                }
            }
        }
        for i in 0..16 {
            if forms[i] == Form::Bitmap {
                let slot = self.bitmaps.alloc(&chunks[i])?;
                bitmap_new.push(slot);
                put_desc(&mut rec, i, Form::Bitmap, slot)?;
            }
        }
        let new = if chunks.iter().all(Vec::is_empty) {
            None
        } else {
            match self.arena.alloc(&rec) {
                Ok(h) => Some(h),
                Err(e) => {
                    for slot in bitmap_new {
                        self.bitmaps.retire(slot)?;
                    }
                    return Err(e);
                }
            }
        };
        let generation = (self.gen_table[g] >> 32) as u32;
        let generation = generation.wrapping_add(1);
        self.gen_table[g] = (u64::from(generation) << 32)
            | u64::from(if new.is_some() { g as u32 } else { ABSENT });
        self.group_handles[g] = new.unwrap_or(0);
        if let Some(h) = old {
            let r = self.arena.get(h)?.to_vec();
            for i in 0..16 {
                let (f, o) = Self::descriptor(&r, i)?;
                if f == Form::Bitmap {
                    self.bitmaps.retire(o)?;
                }
            }
            self.arena.retire(h)?;
        }
        Ok(())
    }
    pub fn insert(&mut self, key: u32) -> Result<bool, StoreError> {
        let c = (key >> 16) as usize;
        let mut v = self.chunk_values(c)?;
        match v.binary_search(&(key as u16)) {
            Ok(_) => return Ok(false),
            Err(i) => v.insert(i, key as u16),
        }
        let g = c >> 4;
        let mut chunks = self.decode_group(g)?;
        chunks[c & 15] = v;
        let hot = (0..16)
            .filter(|&i| self.tiers.get((g << 4) + i).unwrap_or(-1) == 1)
            .collect::<Vec<_>>();
        self.replace_group(g, &chunks)?;
        self.len += 1;
        if chunks[c & 15].len() == 1 {
            self.set_tier(c, 0)?;
        }
        self.tiers.mark(c)?;
        for i in hot {
            self.refresh_hot((g << 4) + i)?;
        }
        if self.tiers.get(c)? == -1 {
            self.set_tier(c, 0)?;
        }
        Ok(true)
    }
    pub fn remove(&mut self, key: u32) -> Result<bool, StoreError> {
        let c = (key >> 16) as usize;
        let mut v = self.chunk_values(c)?;
        let Ok(i) = v.binary_search(&(key as u16)) else {
            return Ok(false);
        };
        v.remove(i);
        let g = c >> 4;
        let mut chunks = self.decode_group(g)?;
        chunks[c & 15] = v;
        let hot = (0..16)
            .filter(|&i| self.tiers.get((g << 4) + i).unwrap_or(-1) == 1)
            .collect::<Vec<_>>();
        self.replace_group(g, &chunks)?;
        self.len -= 1;
        self.tiers.mark(c)?;
        self.set_tier(c, if chunks[c & 15].is_empty() { -1 } else { 0 })?;
        for i in hot {
            let ch = (g << 4) + i;
            if !chunks[i].is_empty() {
                self.refresh_hot(ch)?;
            }
        }
        Ok(true)
    }
    pub fn decode(&self) -> Result<Vec<u32>, StoreError> {
        let mut out = Vec::with_capacity(self.len);
        for g in 0..GROUP_COUNT {
            for (i, v) in self.decode_group(g)?.iter().enumerate() {
                let base = ((g << 20) | (i << 16)) as u32;
                out.extend(v.iter().map(|&x| base | u32::from(x)));
            }
        }
        Ok(out)
    }
    pub fn promote_range(
        &mut self,
        start: usize,
        end: usize,
    ) -> Result<(usize, usize), StoreError> {
        if start > end || end > CHUNK_COUNT {
            return Err(StoreError::RangeOutOfRange { start, end });
        }
        let mut promoted = 0;
        let mut spills = 0;
        match self.layout {
            PocketLayout::PerChunk32 => {
                for c in start..end {
                    let v = self.chunk_values(c)?;
                    if v.is_empty() {
                        continue;
                    }
                    if v.len() <= 18 {
                        let off = c * 32;
                        let pockets = self
                            .pockets
                            .get_or_insert_with(|| vec![0; 2 * 1024 * 1024].into_boxed_slice());
                        pack32(&v, &mut pockets[off..off + 32]);
                        self.set_tier(c, 1)?;
                        promoted += 1
                    } else {
                        self.set_tier(c, 0)?;
                        spills += 1
                    }
                }
            }
            PocketLayout::PerPair64 => {
                let first = start / 2;
                let last = end.div_ceil(2);
                for p in first..last {
                    let a = self.chunk_values(p * 2)?;
                    let b = self.chunk_values(p * 2 + 1)?;
                    if a.is_empty() && b.is_empty() {
                        continue;
                    }
                    if a.len() + b.len() <= 36 {
                        let off = p * 64;
                        let pockets = self
                            .pockets
                            .get_or_insert_with(|| vec![0; 2 * 1024 * 1024].into_boxed_slice());
                        pack64(&a, &b, &mut pockets[off..off + 64]);
                        for c in [p * 2, p * 2 + 1] {
                            if c >= start && c < end && !self.chunk_values(c)?.is_empty() {
                                self.set_tier(c, 1)?;
                                promoted += 1;
                            }
                        }
                    } else {
                        for c in p * 2..p * 2 + 2 {
                            if c >= start && c < end && !self.chunk_values(c)?.is_empty() {
                                self.set_tier(c, 0)?;
                                spills += 1;
                            }
                        }
                    }
                }
            }
        }
        Ok((promoted, spills))
    }
    pub fn demote_range(&mut self, start: usize, end: usize) -> Result<usize, StoreError> {
        if start > end || end > CHUNK_COUNT {
            return Err(StoreError::RangeOutOfRange { start, end });
        }
        let mut n = 0;
        for c in start..end {
            if self.tiers.get(c)? == 1 {
                self.set_tier(
                    c,
                    if self.chunk_values(c)?.is_empty() {
                        -1
                    } else {
                        0
                    },
                )?;
                n += 1;
            }
        }
        Ok(n)
    }
    fn refresh_hot(&mut self, c: usize) -> Result<(), StoreError> {
        match self.layout {
            PocketLayout::PerChunk32 => {
                let v = self.chunk_values(c)?;
                if v.len() <= 18 && !v.is_empty() {
                    let off = c * 32;
                    pack32(
                        &v,
                        &mut self
                            .pockets
                            .get_or_insert_with(|| vec![0; 2 * 1024 * 1024].into_boxed_slice())
                            [off..off + 32],
                    );
                    self.set_tier(c, 1)?
                } else {
                    self.set_tier(c, 0)?
                }
            }
            PocketLayout::PerPair64 => {
                let p = c / 2;
                let a = self.chunk_values(p * 2)?;
                let b = self.chunk_values(p * 2 + 1)?;
                if a.len() + b.len() <= 36 {
                    let off = p * 64;
                    pack64(
                        &a,
                        &b,
                        &mut self
                            .pockets
                            .get_or_insert_with(|| vec![0; 2 * 1024 * 1024].into_boxed_slice())
                            [off..off + 64],
                    );
                } else {
                    for x in p * 2..p * 2 + 2 {
                        if self.tiers.get(x)? == 1 {
                            self.set_tier(x, 0)?;
                        }
                    }
                }
            }
        }
        Ok(())
    }
    fn pocket_contains(&self, c: usize, x: u16) -> Result<bool, StoreError> {
        let p = self.pockets.as_ref().ok_or(StoreError::InvalidRecord)?;
        match self.layout {
            PocketLayout::PerChunk32 => contains32(&p[c * 32..c * 32 + 32], x),
            PocketLayout::PerPair64 => contains64(&p[(c / 2) * 64..(c / 2 + 1) * 64], c & 1, x),
        }
    }
    pub fn dirty_chunks(&mut self) -> Result<Vec<usize>, StoreError> {
        let mut out = Vec::new();
        let mut w = 0;
        loop {
            let next = self.tiers.next_dirty(w);
            if next < 0 {
                break;
            }
            let i = next as usize;
            let bits = self.tiers.take_dirty(i)?;
            for b in 0..64 {
                if bits & (1 << b) != 0 {
                    let c = i * 64 + b;
                    if c < CHUNK_COUNT {
                        out.push(c)
                    }
                }
            }
            w = i + 1;
        }
        Ok(out)
    }
}

fn put_desc(r: &mut Vec<u8>, i: usize, f: Form, o: u16) -> Result<(), StoreError> {
    let p = i * 3;
    r[p] = f as u8;
    r[p + 1..p + 3].copy_from_slice(&o.to_le_bytes());
    Ok(())
}
fn make_runs(v: &[u16]) -> Vec<(u16, u16)> {
    let mut r = Vec::new();
    let mut i = 0;
    while i < v.len() {
        let s = v[i];
        let mut e = i + 1;
        while e < v.len() && u32::from(v[e]) == u32::from(v[e - 1]) + 1 {
            e += 1
        }
        r.push((s, (e - i) as u16));
        i = e;
    }
    r
}
fn append_runs(r: &mut Vec<u8>, runs: &[(u16, u16)]) -> Result<u16, StoreError> {
    let off = u16::try_from(r.len()).map_err(|_| StoreError::TooLarge)?;
    r.extend_from_slice(&(runs.len() as u16).to_le_bytes());
    for &(s, n) in runs {
        r.extend_from_slice(&s.to_le_bytes());
        r.extend_from_slice(&n.to_le_bytes());
    }
    Ok(off)
}

fn write_bits(b: &mut [u8], mut at: usize, n: usize, value: u64) {
    for i in 0..n {
        if value & (1 << i) != 0 {
            b[at / 8] |= 1 << (at & 7)
        }
        at += 1
    }
}
fn read_bits(b: &[u8], at: usize, n: usize) -> u64 {
    let mut v = 0;
    for i in 0..n {
        if b[(at + i) / 8] & (1 << ((at + i) & 7)) != 0 {
            v |= 1 << i
        }
    }
    v
}
#[inline]
fn read_bit_window(b: &[u8], at: usize, n: usize) -> u64 {
    debug_assert!((1..64).contains(&n));
    let byte = at / 8;
    let shift = at & 7;
    let word = u64::from_le_bytes(b[byte..byte + 8].try_into().unwrap());
    (word >> shift) & ((1u64 << n) - 1)
}
#[inline]
fn read_u12(b: &[u8], at: usize) -> u16 {
    let byte = at / 8;
    let shift = at & 7;
    let mut word = u32::from(b[byte]);
    if shift + 12 > 8 {
        word |= u32::from(b[byte + 1]) << 8;
    }
    if shift + 12 > 16 {
        word |= u32::from(b[byte + 2]) << 16;
    }
    ((word >> shift) & 0x0fff) as u16
}
#[inline]
fn select_set64(mut bits: u64, mut nth: usize) -> usize {
    let mut base = 0;
    macro_rules! narrow {
        ($shift:literal) => {{
            let lower = bits & ((1u64 << $shift) - 1);
            let count = lower.count_ones() as usize;
            if nth >= count {
                bits >>= $shift;
                base += $shift;
                nth -= count;
            } else {
                bits = lower;
            }
        }};
    }
    narrow!(32);
    narrow!(16);
    narrow!(8);
    narrow!(4);
    narrow!(2);
    let count = (bits & 1).count_ones() as usize;
    base + usize::from(nth >= count)
}
#[inline]
fn ef_range(bits: u64, bit_len: usize, bucket: usize) -> (usize, usize) {
    let zeros = !bits & ((1u64 << bit_len) - 1);
    let end = select_set64(zeros, bucket) - bucket;
    let start = if bucket == 0 {
        0
    } else {
        select_set64(zeros, bucket - 1) - (bucket - 1)
    };
    (start, end)
}
fn pack32(v: &[u16], b: &mut [u8]) {
    b.fill(0);
    let n = v.len();
    write_bits(b, 0, 5, n as u64);
    let hp = 5;
    for (i, &x) in v.iter().enumerate() {
        let bit = hp + (usize::from(x >> 12)) + i;
        b[bit / 8] |= 1 << (bit & 7);
    }
    let lp = hp + 16 + n;
    for (i, &x) in v.iter().enumerate() {
        write_bits(b, lp + i * 12, 12, u64::from(x & 0xfff));
    }
}
fn contains32(b: &[u8], x: u16) -> Result<bool, StoreError> {
    let n = read_bits(b, 0, 5) as usize;
    let hp = 5;
    let (start, end) = ef_range(read_bit_window(b, hp, 16 + n), 16 + n, usize::from(x >> 12));
    if start == end {
        return Ok(false);
    }
    let mut candidates = [0u16; 18];
    let lp = hp + 16 + n;
    for (index, candidate) in candidates[..end - start].iter_mut().enumerate() {
        *candidate = read_u12(b, lp + (start + index) * 12);
    }
    Ok(scan_u16_candidates(&candidates[..end - start], x & 0xfff))
}
fn pack64(a: &[u16], bval: &[u16], out: &mut [u8]) {
    out.fill(0);
    write_bits(out, 0, 6, a.len() as u64);
    write_bits(out, 6, 6, bval.len() as u64);
    let h0 = 12;
    let h1 = h0 + 16 + a.len();
    for (i, &x) in a.iter().enumerate() {
        let p = h0 + usize::from(x >> 12) + i;
        out[p / 8] |= 1 << (p & 7)
    }
    for (i, &x) in bval.iter().enumerate() {
        let p = h1 + usize::from(x >> 12) + i;
        out[p / 8] |= 1 << (p & 7)
    }
    let lp = h1 + 16 + bval.len();
    for (i, &x) in a.iter().chain(bval).enumerate() {
        write_bits(out, lp + i * 12, 12, u64::from(x & 0xfff));
    }
}
fn contains64(b: &[u8], which: usize, x: u16) -> Result<bool, StoreError> {
    let n0 = read_bits(b, 0, 6) as usize;
    let n1 = read_bits(b, 6, 6) as usize;
    let n = if which == 0 { n0 } else { n1 };
    let hs = if which == 0 { 12 } else { 12 + 16 + n0 };
    let lo = if which == 0 { 0 } else { n0 };
    let lp = 12 + 16 + n0 + 16 + n1;
    let (start, end) = ef_range(read_bit_window(b, hs, 16 + n), 16 + n, usize::from(x >> 12));
    if start == end {
        return Ok(false);
    }
    let mut candidates = [0u16; 36];
    let count = end - start;
    for (index, candidate) in candidates[..count].iter_mut().enumerate() {
        *candidate = read_u12(b, lp + (lo + start + index) * 12);
    }
    Ok(scan_u16_candidates(&candidates[..count], x & 0xfff))
}

        let mut reference = BTreeSet::new();
        for _ in 0..1200 {
            seed ^= seed << 13;
            seed ^= seed >> 7;
            seed ^= seed << 17;

        let initial: Vec<_> = reference.iter().copied().collect();

        store.dirty_chunks().unwrap();
        store.promote_range(0, 4).unwrap();
        for step in 0..5000 {

            seed ^= seed >> 7;
            seed ^= seed << 17;
            let key = (seed as u32) & 0x0003_ffff;
            if step % 3 == 0 {
                assert_eq!(store.insert(key).unwrap(), reference.insert(key));

                    store.decode().unwrap(),
                    reference.iter().copied().collect::<Vec<_>>()
                );
            }
        }
        assert_eq!(store.len(), reference.len());
        assert_eq!(
            store.decode().unwrap(),
            reference.iter().copied().collect::<Vec<_>>()
        );
    }

//! Hierarchical packed ternary map ported from Symbolic std's `trnx64`/`trnmap`.
//!
//! Each leaf word stores 64 trits in thermometer planes (`p = t >= 1`,
//! `u = t >= 0`). Summary planes record whether each 64-way child group has
//! any positive, negative, or zero values. A parallel hierarchy tracks dirty
//! trits. The map payload is one 8-byte-aligned allocation with the Symbolic
//! 384-byte header and level descriptors.

use std::fmt;

const HEADER_BYTES: usize = 384;
const HEADER_WORDS: usize = HEADER_BYTES / 8;
const MAX_LEVELS: usize = (usize::BITS as usize).div_ceil(6) + 1;
const DESCRIPTOR_WORD: usize = 8;

pub const CLASS_NEGATIVE: u8 = 1;
pub const CLASS_ZERO: u8 = 2;
pub const CLASS_POSITIVE: u8 = 4;
const NEGATIVE: u8 = CLASS_NEGATIVE;
const ZERO: u8 = CLASS_ZERO;
const POSITIVE: u8 = CLASS_POSITIVE;
const ALL_CLASSES: u8 = NEGATIVE | ZERO | POSITIVE;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TrnMapError {
    EmptyMap,
    SizeOverflow,
    BufferTooSmall { needed: usize, available: usize },
    IndexOutOfBounds { index: usize, len: usize },
    WordOutOfBounds { word: usize, words: usize },
    InvalidTrit(i8),
    LengthMismatch { left: usize, right: usize },
}

impl fmt::Display for TrnMapError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Self::EmptyMap => f.write_str("trnmap length must be nonzero"),
            Self::SizeOverflow => f.write_str("trnmap allocation size overflows usize"),
            Self::BufferTooSmall { needed, available } => {
                write!(f, "trnmap needs {needed} bytes, buffer has {available}")
            }
            Self::IndexOutOfBounds { index, len } => {
                write!(f, "trit index {index} is outside map length {len}")
            }
            Self::WordOutOfBounds { word, words } => {
                write!(f, "word index {word} is outside {words} words")
            }
            Self::InvalidTrit(value) => write!(f, "trit must be -1, 0, or 1, got {value}"),
            Self::LengthMismatch { left, right } => {
                write!(f, "trnmap lengths differ: {left} and {right}")
            }
        }
    }
}

impl std::error::Error for TrnMapError {}

/// 64 trits packed into the two thermometer planes used by Symbolic std.
#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
pub struct TriWord {
    /// Positive plane: set for trit `+1`.
    p: u64,
    /// Nonnegative plane: set for trits `0` and `+1`.
    u: u64,
}

impl TriWord {
    /// Constructs a word while enforcing the invariant `p` is a subset of `u`.
    #[inline]
    pub const fn from_planes(p: u64, u: u64) -> Self {
        Self { p: p & u, u }
    }

    /// Returns the normalized positive and nonnegative planes.
    #[inline]
    pub const fn planes(self) -> (u64, u64) {
        (self.p, self.u)
    }

    /// A word filled with one trit.
    pub const fn splat(trit: i8) -> Result<Self, TrnMapError> {
        match trit {
            -1 => Ok(Self { p: 0, u: 0 }),
            0 => Ok(Self { p: 0, u: u64::MAX }),
            1 => Ok(Self {
                p: u64::MAX,
                u: u64::MAX,
            }),
            value => Err(TrnMapError::InvalidTrit(value)),
        }
    }

    /// Returns the trit at bit `index`.
    #[inline]
    pub const fn get(self, index: u32) -> Option<i8> {
        if index >= 64 {
            return None;
        }
        let positive = ((self.p >> index) & 1) as i8;
        let nonnegative = ((self.u >> index) & 1) as i8;
        Some(positive + nonnegative - 1)
    }

    /// Replaces one trit.
    pub fn set(&mut self, index: u32, trit: i8) -> Result<(), TrnMapError> {
        if index >= 64 {
            return Err(TrnMapError::IndexOutOfBounds {
                index: index as usize,
                len: 64,
            });
        }
        if !(-1..=1).contains(&trit) {
            return Err(TrnMapError::InvalidTrit(trit));
        }
        let bit = 1u64 << index;
        self.p &= !bit;
        self.u &= !bit;
        match trit {
            0 => self.u |= bit,
            1 => {
                self.p |= bit;
                self.u |= bit;
            }
            _ => {}
        }
        Ok(())
    }

    /// Bit mask of trits in the selected classes (1 = -1, 2 = 0, 4 = +1).
    #[inline]
    pub const fn class_mask(self, classes: u8) -> u64 {
        let mut mask = 0;
        if classes & POSITIVE != 0 {
            mask |= self.p;
        }
        if classes & NEGATIVE != 0 {
            mask |= !self.u;
        }
        if classes & ZERO != 0 {
            mask |= self.u & !self.p;
        }
        mask
    }

    #[inline]
    pub const fn count(self, trit: i8) -> Result<u32, TrnMapError> {
        let class = match trit {
            -1 => !self.u,
            0 => self.u & !self.p,
            1 => self.p,
            value => return Err(TrnMapError::InvalidTrit(value)),
        };
        Ok(class.count_ones())
    }

    /// Kleene AND (`min`).
    #[inline]
    pub const fn kleene_and(self, rhs: Self) -> Self {
        Self::from_planes(self.p & rhs.p, self.u & rhs.u)
    }

    /// Kleene OR (`max`).
    #[inline]
    pub const fn kleene_or(self, rhs: Self) -> Self {
        Self::from_planes(self.p | rhs.p, self.u | rhs.u)
    }

    /// Kleene NOT.
    #[inline]
    pub const fn kleene_not(self) -> Self {
        Self::from_planes(!self.u, !self.p)
    }

    /// Ternary dot product against a bit selection.
    #[inline]
    pub const fn dot(self, selected: u64) -> i32 {
        (self.p & selected).count_ones() as i32 - ((!self.u) & selected).count_ones() as i32
    }
}

#[derive(Clone, Copy, Default)]
struct LevelLayout {
    elements: usize,
    presence_offset: usize,
    presence_records: usize,
    dirty_offset: usize,
    dirty_words: usize,
}

#[derive(Clone, Copy)]
struct Layout {
    levels: [LevelLayout; MAX_LEVELS],
    level_count: usize,
    base_words: usize,
    presence_base: usize,
    dirty_base: usize,
    total_words: usize,
    storage_bytes: usize,
    budget_bytes: usize,
}

impl Layout {
    fn for_len(n: usize) -> Result<Self, TrnMapError> {
        if n == 0 {
            return Err(TrnMapError::EmptyMap);
        }
        let mut levels = [LevelLayout::default(); MAX_LEVELS];
        levels[0].elements = n;
        let mut level_count = 1;
        while levels[level_count - 1].elements != 1 {
            if level_count == MAX_LEVELS {
                return Err(TrnMapError::SizeOverflow);
            }
            levels[level_count].elements = div_ceil_64(levels[level_count - 1].elements);
            level_count += 1;
        }

        let mut presence_records = 0usize;
        let mut dirty_words = 0usize;
        for (level, descriptor) in levels[..level_count].iter_mut().enumerate() {
            descriptor.dirty_offset = dirty_words;
            descriptor.dirty_words = div_ceil_64(descriptor.elements);
            dirty_words = dirty_words
                .checked_add(descriptor.dirty_words)
                .ok_or(TrnMapError::SizeOverflow)?;
            if level != 0 {
                descriptor.presence_offset = presence_records;
                descriptor.presence_records = div_ceil_64(descriptor.elements);
                presence_records = presence_records
                    .checked_add(descriptor.presence_records)
                    .ok_or(TrnMapError::SizeOverflow)?;
            }
        }

        let base_words = div_ceil_64(n);
        let presence_words = presence_records
            .checked_mul(3)
            .ok_or(TrnMapError::SizeOverflow)?;
        let dirty_base = HEADER_WORDS
            .checked_add(base_words.checked_mul(2).ok_or(TrnMapError::SizeOverflow)?)
            .and_then(|v| v.checked_add(presence_words))
            .ok_or(TrnMapError::SizeOverflow)?;
        let total_words = dirty_base
            .checked_add(dirty_words)
            .ok_or(TrnMapError::SizeOverflow)?;
        let storage_bytes = total_words
            .checked_mul(8)
            .ok_or(TrnMapError::SizeOverflow)?;
        if storage_bytes > isize::MAX as usize {
            return Err(TrnMapError::SizeOverflow);
        }
        let budget_bytes = storage_bytes
            .checked_add(7)
            .ok_or(TrnMapError::SizeOverflow)?;
        let presence_base = HEADER_WORDS + base_words * 2;
        Ok(Self {
            levels,
            level_count,
            base_words,
            presence_base,
            dirty_base,
            total_words,
            storage_bytes,
            budget_bytes,
        })
    }
}

enum Backing<'a> {
    Owned(Box<[u64]>),
    Borrowed(&'a mut [u64]),
}

impl Backing<'_> {
    #[inline]
    fn words(&self) -> &[u64] {
        match self {
            Self::Owned(words) => words,
            Self::Borrowed(words) => words,
        }
    }

    #[inline]
    fn words_mut(&mut self) -> &mut [u64] {
        match self {
            Self::Owned(words) => words,
            Self::Borrowed(words) => words,
        }
    }
}

/// A 64-way trit map with exact class summaries and hierarchical dirty bits.
///
/// [`TrnMap::new`] owns one contiguous allocation. [`TrnMap::in_buffer`] carves
/// the same layout from caller-owned bytes, requiring `bytes_for(n)` bytes to
/// account for up to seven bytes of alignment slack. New maps contain only -1
/// and have no dirty trits.
pub struct TrnMap<'a> {
    backing: Backing<'a>,
    len: usize,
    level_count: usize,
    base_words: usize,
    trit_base: usize,
    presence_base: usize,
    dirty_base: usize,
}

impl fmt::Debug for TrnMap<'_> {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("TrnMap")
            .field("len", &self.len)
            .field("levels", &self.level_count)
            .field("base_words", &self.base_words)
            .field("storage_bytes", &self.storage_bytes())
            .field("owned", &matches!(&self.backing, Backing::Owned(_)))
            .finish()
    }
}

impl TrnMap<'static> {
    /// Creates an owned map of `len` trits, initialized to -1.
    pub fn new(len: usize) -> Result<Self, TrnMapError> {
        let layout = Layout::for_len(len)?;
        let mut words = vec![0u64; layout.total_words].into_boxed_slice();
        initialize_storage(&mut words, len, layout, true);
        Ok(Self::from_backing(Backing::Owned(words), len, layout))
    }

    /// Exact in-block allocation size, excluding the caller's alignment slack.
    pub fn storage_bytes_for(len: usize) -> Result<usize, TrnMapError> {
        Ok(Layout::for_len(len)?.storage_bytes)
    }

    /// Caller budget: exact in-block size plus seven bytes for 8-byte alignment.
    pub fn bytes_for(len: usize) -> Result<usize, TrnMapError> {
        Ok(Layout::for_len(len)?.budget_bytes)
    }
}

impl<'a> TrnMap<'a> {
    /// Initializes a map in caller-owned bytes, like Symbolic std's window form.
    /// The map borrows the backing storage until it is dropped.
    pub fn in_buffer(buffer: &'a mut [u8], len: usize) -> Result<Self, TrnMapError> {
        let layout = Layout::for_len(len)?;
        if buffer.len() < layout.budget_bytes {
            return Err(TrnMapError::BufferTooSmall {
                needed: layout.budget_bytes,
                available: buffer.len(),
            });
        }
        let available = buffer.len();
        // SAFETY: u64 accepts every bit pattern; only the aligned middle is used,
        // and the exclusive borrow of `buffer` is retained by the returned map.
        let (_, aligned, _) = unsafe { buffer.align_to_mut::<u64>() };
        if aligned.len() < layout.total_words {
            return Err(TrnMapError::BufferTooSmall {
                needed: layout.budget_bytes,
                available,
            });
        }
        let words = &mut aligned[..layout.total_words];
        initialize_storage(words, len, layout, false);
        Ok(Self::from_backing(Backing::Borrowed(words), len, layout))
    }

    fn from_backing(backing: Backing<'a>, len: usize, layout: Layout) -> Self {
        Self {
            backing,
            len,
            level_count: layout.level_count,
            base_words: layout.base_words,
            trit_base: HEADER_WORDS,
            presence_base: layout.presence_base,
            dirty_base: layout.dirty_base,
        }
    }

    /// Number of trits.
    #[inline]
    pub const fn len(&self) -> usize {
        self.len
    }

    #[inline]
    pub const fn is_empty(&self) -> bool {
        false
    }

    /// Number of 64-trit leaf words.
    #[inline]
    pub const fn word_len(&self) -> usize {
        self.base_words
    }

    /// Number of 64-way hierarchy levels, including the leaf level.
    #[inline]
    pub const fn levels(&self) -> usize {
        self.level_count
    }

    /// Exact size of the contiguous map block.
    #[inline]
    pub fn storage_bytes(&self) -> usize {
        self.backing.words()[2] as usize
    }

    /// Reads trit `index`.
    pub fn get(&self, index: usize) -> Result<i8, TrnMapError> {
        if index >= self.len {
            return Err(TrnMapError::IndexOutOfBounds {
                index,
                len: self.len,
            });
        }
        let word = self.read_leaf(index / 64);
        Ok(word.get((index & 63) as u32).expect("masked trit index"))
    }

    /// Writes trit `index`, updating only summary levels whose state changes.
    pub fn set(&mut self, index: usize, trit: i8) -> Result<(), TrnMapError> {
        if index >= self.len {
            return Err(TrnMapError::IndexOutOfBounds {
                index,
                len: self.len,
            });
        }
        if !(-1..=1).contains(&trit) {
            return Err(TrnMapError::InvalidTrit(trit));
        }
        let word_index = index / 64;
        let mut word = self.read_leaf(word_index);
        let old = word;
        word.set((index & 63) as u32, trit)?;
        if word != old {
            self.write_leaf(word_index, word);
            self.recompute_summaries(word_index);
        }
        Ok(())
    }

    /// Reads one packed group of 64 trits.
    pub fn word(&self, word: usize) -> Result<TriWord, TrnMapError> {
        if word >= self.base_words {
            return Err(TrnMapError::WordOutOfBounds {
                word,
                words: self.base_words,
            });
        }
        Ok(self.read_leaf(word))
    }

    /// Replaces one packed group and updates its summaries.
    pub fn put(&mut self, word: usize, value: TriWord) -> Result<(), TrnMapError> {
        if word >= self.base_words {
            return Err(TrnMapError::WordOutOfBounds {
                word,
                words: self.base_words,
            });
        }
        let valid = valid_bits(self.len.saturating_sub(word * 64).min(64));
        let value = TriWord::from_planes(value.p & valid, value.u & valid);
        if self.read_leaf(word) != value {
            self.write_leaf(word, value);
            self.recompute_summaries(word);
        }
        Ok(())
    }

    /// The map's trit if all entries are +1 or all are -1; otherwise zero.
    pub fn top(&self) -> i8 {
        if self.level_count == 1 {
            let word = self.read_leaf(0);
            let valid = valid_bits(self.len);
            if word.p & valid == valid {
                1
            } else if word.u & valid == 0 {
                -1
            } else {
                0
            }
        } else {
            let level = self.level_count - 1;
            let planes = self.read_presence(level, 0);
            let positive = planes[0] & 1 != 0;
            let negative = planes[1] & 1 != 0;
            let zero = planes[2] & 1 != 0;
            if !zero && !negative {
                1
            } else if !zero && !positive {
                -1
            } else {
                0
            }
        }
    }

    /// First index at or after `start` belonging to `classes`, or `len`.
    /// Class bits are 1 = -1, 2 = 0, 4 = +1. Runs in O(log_64(n)) word loads.
    pub fn find(&self, classes: u8, start: usize) -> usize {
        let classes = classes & ALL_CLASSES;
        if start >= self.len || classes == 0 {
            return self.len;
        }
        let leaf_index = start / 64;
        let leaf_mask = self.leaf_class_mask(leaf_index, classes) & (u64::MAX << (start & 63));
        if leaf_mask != 0 {
            return leaf_index * 64 + leaf_mask.trailing_zeros() as usize;
        }
        if self.level_count == 1 {
            return self.len;
        }

        // Climb one level per miss. Each summary word covers 64 child groups.
        let mut level = 1usize;
        let mut target = leaf_index + 1;
        while level < self.level_count {
            let record = target / 64;
            let record_count = self.level_elements(level).div_ceil(64);
            if record >= record_count {
                return self.len;
            }
            let mask =
                self.presence_class_mask(level, record, classes) & (u64::MAX << (target & 63));
            if mask != 0 {
                let mut child = record * 64 + mask.trailing_zeros() as usize;
                for lower in (1..level).rev() {
                    let lower_mask = self.presence_class_mask(lower, child, classes);
                    debug_assert_ne!(lower_mask, 0);
                    child = child * 64 + lower_mask.trailing_zeros() as usize;
                }
                let leaf = child;
                let leaf_mask = self.leaf_class_mask(leaf, classes);
                debug_assert_ne!(leaf_mask, 0);
                return leaf * 64 + leaf_mask.trailing_zeros() as usize;
            }
            target = record + 1;
            level += 1;
        }
        self.len
    }

    /// Length of the equal-trit run beginning at `index`.
    pub fn run(&self, index: usize) -> Result<usize, TrnMapError> {
        let trit = self.get(index)?;
        let different = ALL_CLASSES ^ trit_class(trit);
        Ok(self.find(different, index) - index)
    }

    /// Count occurrences of one trit, using summary levels to skip empty groups.
    pub fn count(&self, trit: i8) -> Result<usize, TrnMapError> {
        let classes = trit_class_checked(trit)?;
        if self.level_count == 1 {
            return Ok(self.leaf_class_mask(0, classes).count_ones() as usize);
        }

        // Count leaf words, not individual matching trits. The level-one
        // summaries let sparse classes skip 4,096-trit regions in one word;
        // dense classes take a sequential packed-word pass instead of doing a
        // hierarchy descent for every leaf.
        let summary_records = div_ceil_64(self.base_words);
        let plane = match classes {
            NEGATIVE => 1,
            ZERO => 2,
            POSITIVE => 0,
            _ => unreachable!("a checked trit has exactly one class"),
        };
        let words = self.backing.words();
        let mut active_leaf_words = 0usize;
        for record in 0..summary_records {
            active_leaf_words +=
                words[self.presence_index(1, record) + plane].count_ones() as usize;
        }

        let mut count = 0usize;
        // A candidate bitmap is cheaper while at most half its leaf words are
        // active; above that density, the sequential packed-word pass wins.
        if active_leaf_words > self.base_words / 2 {
            for word in 0..self.base_words {
                count += self.leaf_class_mask(word, classes).count_ones() as usize;
            }
        } else {
            for record in 0..summary_records {
                let mut active = words[self.presence_index(1, record) + plane];
                let valid = valid_bits(self.base_words.saturating_sub(record * 64).min(64));
                active &= valid;
                while active != 0 {
                    let child = active.trailing_zeros() as usize;
                    let word = record * 64 + child;
                    count += self.leaf_class_mask(word, classes).count_ones() as usize;
                    active &= active - 1;
                }
            }
        }
        Ok(count)
    }

    /// Whether any trit is dirty. O(1).
    #[inline]
    pub fn dirty(&self) -> bool {
        let top = self.level_count - 1;
        self.read_dirty(top, 0) != 0
    }

    /// Marks one trit dirty and propagates only newly set summary bits.
    pub fn mark(&mut self, index: usize) -> Result<(), TrnMapError> {
        if index >= self.len {
            return Err(TrnMapError::IndexOutOfBounds {
                index,
                len: self.len,
            });
        }
        let dirty_word = index / 64;
        let leaf_bit = 1u64 << (index & 63);
        let old = self.read_dirty(0, dirty_word);
        if old & leaf_bit != 0 {
            return Ok(());
        }
        self.write_dirty(0, dirty_word, old | leaf_bit);
        let mut child = dirty_word;
        for level in 1..self.level_count {
            let record = child / 64;
            let bit = 1u64 << (child & 63);
            let old = self.read_dirty(level, record);
            if old & bit != 0 {
                break;
            }
            self.write_dirty(level, record, old | bit);
            child = record;
        }
        Ok(())
    }

    /// First leaf word at or after `word` containing dirty trits, or -1.
    pub fn next_dirty(&self, word: usize) -> isize {
        if word >= self.base_words {
            return -1;
        }
        if self.level_count == 1 {
            return if self.read_dirty(0, 0) != 0 { 0 } else { -1 };
        }
        let mut level = 1usize;
        let mut target = word;
        while level < self.level_count {
            let record = target / 64;
            let record_count = self.level_elements(level).div_ceil(64);
            if record >= record_count {
                return -1;
            }
            let mask = self.read_dirty(level, record) & (u64::MAX << (target & 63));
            if mask != 0 {
                let mut child = record * 64 + mask.trailing_zeros() as usize;
                for lower in (1..level).rev() {
                    let lower_mask = self.read_dirty(lower, child);
                    debug_assert_ne!(lower_mask, 0);
                    child = child * 64 + lower_mask.trailing_zeros() as usize;
                }
                return child as isize;
            }
            target = record + 1;
            level += 1;
        }
        -1
    }

    /// Returns and clears dirty bits for leaf word `word`, pruning empty ancestors.
    pub fn take_dirty(&mut self, word: usize) -> Result<u64, TrnMapError> {
        if word >= self.base_words {
            return Err(TrnMapError::WordOutOfBounds {
                word,
                words: self.base_words,
            });
        }
        let bits = self.read_dirty(0, word);
        if bits == 0 {
            return Ok(0);
        }
        self.write_dirty(0, word, 0);
        let mut child = word;
        for level in 1..self.level_count {
            let record = child / 64;
            let bit = 1u64 << (child & 63);
            let remaining = self.read_dirty(level, record) & !bit;
            self.write_dirty(level, record, remaining);
            if remaining != 0 {
                break;
            }
            child = record;
        }
        Ok(bits)
    }

    /// Kleene OR (`max`) in place, visiting only right-hand words with 0 or +1.
    pub fn or_assign(&mut self, rhs: &Self) -> Result<(), TrnMapError> {
        self.ensure_same_len(rhs)?;
        self.merge(rhs, true);
        Ok(())
    }

    /// Kleene AND (`min`) in place, visiting only right-hand words with -1 or 0.
    pub fn and_assign(&mut self, rhs: &Self) -> Result<(), TrnMapError> {
        self.ensure_same_len(rhs)?;
        self.merge(rhs, false);
        Ok(())
    }

    /// Kleene NOT in place; swaps positive/negative summaries and preserves dirty bits.
    pub fn not_assign(&mut self) {
        for word_index in 0..self.base_words {
            let old = self.read_leaf(word_index);
            let valid = valid_bits(self.len.saturating_sub(word_index * 64).min(64));
            let inverted = TriWord::from_planes(!old.u & valid, !old.p & valid);
            self.write_leaf(word_index, inverted);
        }
        for level in 1..self.level_count {
            let records = self.level_elements(level).div_ceil(64);
            for record in 0..records {
                let at = self.presence_index(level, record);
                let words = self.backing.words_mut();
                words.swap(at, at + 1);
            }
        }
    }

    fn ensure_same_len(&self, rhs: &Self) -> Result<(), TrnMapError> {
        if self.len == rhs.len {
            Ok(())
        } else {
            Err(TrnMapError::LengthMismatch {
                left: self.len,
                right: rhs.len,
            })
        }
    }

    fn merge(&mut self, rhs: &Self, is_or: bool) {
        if self.level_count == 1 {
            let lhs_word = self.read_leaf(0);
            let rhs_word = rhs.read_leaf(0);
            let merged = if is_or {
                lhs_word.kleene_or(rhs_word)
            } else {
                lhs_word.kleene_and(rhs_word)
            };
            if merged != lhs_word {
                self.write_leaf(0, merged);
            }
            return;
        }
        let summary_records = div_ceil_64(self.base_words);
        for record in 0..summary_records {
            let rhs_at = rhs.presence_index(1, record);
            let rhs_planes = rhs.backing.words();
            let mut candidates = if is_or {
                rhs_planes[rhs_at] | rhs_planes[rhs_at + 2]
            } else {
                rhs_planes[rhs_at + 1] | rhs_planes[rhs_at + 2]
            };
            candidates &= valid_bits(self.base_words.saturating_sub(record * 64).min(64));
            if candidates == 0 {
                continue;
            }

            let lhs_at = self.presence_index(1, record);
            let mut updated_planes = {
                let words = self.backing.words();
                [words[lhs_at], words[lhs_at + 1], words[lhs_at + 2]]
            };
            let mut changed = false;
            while candidates != 0 {
                let child = candidates.trailing_zeros() as usize;
                let word_index = record * 64 + child;
                let lhs_word = self.read_leaf(word_index);
                let rhs_word = rhs.read_leaf(word_index);
                let merged = if is_or {
                    lhs_word.kleene_or(rhs_word)
                } else {
                    lhs_word.kleene_and(rhs_word)
                };
                if merged != lhs_word {
                    self.write_leaf(word_index, merged);
                    let valid = valid_bits(self.len.saturating_sub(word_index * 64).min(64));
                    let bit = 1u64 << child;
                    let present = [
                        merged.p & valid != 0,
                        !merged.u & valid != 0,
                        merged.u & !merged.p & valid != 0,
                    ];
                    for plane in 0..3 {
                        if present[plane] {
                            updated_planes[plane] |= bit;
                        } else {
                            updated_planes[plane] &= !bit;
                        }
                    }
                    changed = true;
                }
                candidates &= candidates - 1;
            }
            if changed {
                let words = self.backing.words_mut();
                words[lhs_at] = updated_planes[0];
                words[lhs_at + 1] = updated_planes[1];
                words[lhs_at + 2] = updated_planes[2];
                if self.level_count > 2 {
                    self.recompute_summary_node(2, record);
                }
            }
        }
    }

    fn level_elements(&self, level: usize) -> usize {
        self.backing.words()[DESCRIPTOR_WORD + 3 * level] as usize
    }

    #[inline]
    fn presence_index(&self, level: usize, record: usize) -> usize {
        let offset = self.backing.words()[DESCRIPTOR_WORD + 3 * level + 1] as usize;
        self.presence_base + (offset + record) * 3
    }

    #[inline]
    fn dirty_index(&self, level: usize, word: usize) -> usize {
        let offset = self.backing.words()[DESCRIPTOR_WORD + 3 * level + 2] as usize;
        self.dirty_base + offset + word
    }

    #[inline]
    fn read_leaf(&self, index: usize) -> TriWord {
        let words = self.backing.words();
        let at = self.trit_base + index * 2;
        TriWord::from_planes(words[at], words[at + 1])
    }

    #[inline]
    fn write_leaf(&mut self, index: usize, value: TriWord) {
        let at = self.trit_base + index * 2;
        let words = self.backing.words_mut();
        words[at] = value.p;
        words[at + 1] = value.u;
    }

    #[inline]
    fn read_presence(&self, level: usize, record: usize) -> [u64; 3] {
        let at = self.presence_index(level, record);
        let words = self.backing.words();
        [words[at], words[at + 1], words[at + 2]]
    }

    #[inline]
    fn presence_class_mask(&self, level: usize, record: usize, classes: u8) -> u64 {
        let planes = self.read_presence(level, record);
        let mut mask = 0;
        if classes & POSITIVE != 0 {
            mask |= planes[0];
        }
        if classes & NEGATIVE != 0 {
            mask |= planes[1];
        }
        if classes & ZERO != 0 {
            mask |= planes[2];
        }
        mask
    }

    #[inline]
    fn leaf_class_mask(&self, word: usize, classes: u8) -> u64 {
        let valid = valid_bits(self.len.saturating_sub(word * 64).min(64));
        self.read_leaf(word).class_mask(classes) & valid
    }

    fn recompute_summaries(&mut self, leaf_word: usize) {
        self.recompute_summary_node(1, leaf_word);
    }

    fn recompute_summary_node(&mut self, mut level: usize, mut child: usize) {
        while level < self.level_count {
            let record = child / 64;
            let bit = 1u64 << (child & 63);
            let present = if level == 1 {
                let valid = valid_bits(self.len.saturating_sub(child * 64).min(64));
                let word = self.read_leaf(child);
                [
                    word.p & valid != 0,
                    !word.u & valid != 0,
                    word.u & !word.p & valid != 0,
                ]
            } else {
                let lower = self.read_presence(level - 1, child);
                [lower[0] != 0, lower[1] != 0, lower[2] != 0]
            };
            let at = self.presence_index(level, record);
            let words = self.backing.words_mut();
            let old = [
                words[at] & bit != 0,
                words[at + 1] & bit != 0,
                words[at + 2] & bit != 0,
            ];
            if old == present {
                break;
            }
            for plane in 0..3 {
                if present[plane] {
                    words[at + plane] |= bit;
                } else {
                    words[at + plane] &= !bit;
                }
            }
            child = record;
            level += 1;
        }
    }

    #[inline]
    fn read_dirty(&self, level: usize, word: usize) -> u64 {
        self.backing.words()[self.dirty_index(level, word)]
    }

    #[inline]
    fn write_dirty(&mut self, level: usize, word: usize, value: u64) {
        let at = self.dirty_index(level, word);
        self.backing.words_mut()[at] = value;
    }
}

fn initialize_storage(words: &mut [u64], len: usize, layout: Layout, owned: bool) {
    words.fill(0);
    let trit_base = HEADER_WORDS;
    let presence_base = layout.presence_base;
    let dirty_base = layout.dirty_base;
    words[0] = len as u64;
    words[1] = layout.level_count as u64;
    words[2] = layout.storage_bytes as u64;
    words[3] = u64::from(owned);
    words[4] = words.as_mut_ptr().wrapping_add(trit_base) as usize as u64;
    words[5] = words.as_mut_ptr().wrapping_add(presence_base) as usize as u64;
    words[6] = words.as_mut_ptr().wrapping_add(dirty_base) as usize as u64;

    for (level, descriptor) in layout.levels[..layout.level_count].iter().enumerate() {
        let at = DESCRIPTOR_WORD + 3 * level;
        words[at] = descriptor.elements as u64;
        words[at + 1] = descriptor.presence_offset as u64;
        words[at + 2] = descriptor.dirty_offset as u64;
    }

    // Maps start all -1. For each level, one bit marks each valid child group.
    for level in 1..layout.level_count {
        let descriptor = layout.levels[level];
        let child_count = descriptor.elements;
        for record in 0..descriptor.presence_records {
            let count = child_count.saturating_sub(record * 64).min(64);
            let valid = valid_bits(count);
            let at = presence_base + (descriptor.presence_offset + record) * 3;
            words[at + 1] = valid;
        }
    }
}

#[inline]
const fn div_ceil_64(value: usize) -> usize {
    (value >> 6) + if value & 63 != 0 { 1 } else { 0 }
}

#[inline]
const fn valid_bits(count: usize) -> u64 {
    if count >= 64 {
        u64::MAX
    } else if count == 0 {
        0
    } else {
        (1u64 << count) - 1
    }
}

#[inline]
const fn trit_class(trit: i8) -> u8 {
    match trit {
        -1 => NEGATIVE,
        0 => ZERO,
        1 => POSITIVE,
        _ => 0,
    }
}

#[inline]
fn trit_class_checked(trit: i8) -> Result<u8, TrnMapError> {
    let class = trit_class(trit);
    if class == 0 {
        Err(TrnMapError::InvalidTrit(trit))
    } else {
        Ok(class)
    }
}

                            .skip(word * 64)

                            *value = false;
                        }
                    }

            let mut rhs = TrnMap::new(len).unwrap();
            let mut a = vec![-1i8; len];
            let mut b = vec![-1i8; len];
            for i in 0..len {
                let av = (i % 3) as i8 - 1;
                let bv = ((i * 7 + 1) % 3) as i8 - 1;
                lhs.set(i, av).unwrap();
                rhs.set(i, bv).unwrap();
                a[i] = av;
                b[i] = bv;
            }
            lhs.or_assign(&rhs).unwrap();
            for (i, value) in a.iter_mut().enumerate() {
                *value = (*value).max(b[i]);
                assert_eq!(lhs.get(i), Ok(*value), "OR len={len}, i={i}");
            }
            lhs.and_assign(&rhs).unwrap();
            for (i, value) in a.iter_mut().enumerate() {
                *value = (*value).min(b[i]);

            }
            lhs.not_assign();
            for (i, value) in a.iter_mut().enumerate() {
                *value = -*value;
                assert_eq!(lhs.get(i), Ok(*value), "NOT len={len}, i={i}");
            }
            assert_eq!(
                lhs.count(-1).unwrap() + lhs.count(0).unwrap() + lhs.count(1).unwrap(),
                len
            );
        }

//! Lane-separated ordered-stream codec with an independent 64 KiB LZ window
//! per lane, followed by one per-lane rANS/Huffman entropy stage.

use super::{CodecError, align_up, put_u16, put_u32, rans, read_u16, read_u32, read_u32_unchecked};
use rayon::prelude::*;
use std::borrow::Cow;
use std::sync::OnceLock;

#[cfg(target_arch = "x86")]
use std::arch::x86::*;
#[cfg(target_arch = "x86_64")]
use std::arch::x86_64::*;

#[cfg(feature = "wgpu")]
pub mod gpu;

const MAGIC: &[u8; 4] = b"RNLZ";
const VERSION: u16 = 2;
const LEGACY_VERSION: u16 = 1;
const HEADER_BYTES: usize = 20;
const LEGACY_DESCRIPTOR_BYTES: usize = 24;
const DESCRIPTOR_BYTES: usize = 32;
const TOKEN_BYTES: usize = 4;
const WINDOW_SIZE: usize = 1 << 16;
const HASH_BITS_COMPACT: usize = 18;
const HASH_BITS_WIDE: usize = 18;
const MAX_MATCH: usize = 130;
const PARALLEL_MIN_BYTES: usize = 16 * 1024;
const LZ_BLOCK_BYTES: usize = 64 * 1024;
const MODE_RAW: u8 = 0;
const MODE_LZ_ENTROPY: u8 = 2;
const METHOD_RAW: u8 = 0;
const METHOD_RANS: u8 = 1;
const METHOD_HUFFMAN: u8 = 2;

type SplitFourFn = fn(&[u8], &mut [Vec<u8>], usize);
static SPLIT_FOUR_DISPATCH: OnceLock<SplitFourFn> = OnceLock::new();
static RAYON_WORKERS: OnceLock<usize> = OnceLock::new();

#[derive(Clone, Debug)]
struct LaneData {
    raw_len: usize,
    token_len: usize,
    token_count: usize,
    literal_len: usize,
    model: Vec<u8>,
    payload: Vec<u8>,
    mode: u8,
    entropy_method: u8,
}

#[derive(Clone, Copy, Debug)]
struct LaneDescriptor {
    raw_len: usize,
    token_len: usize,
    token_count: usize,
    literal_len: usize,
    payload_offset: usize,
    payload_len: usize,
    model_offset: usize,
    model_len: usize,
    mode: u8,
    entropy_method: u8,
}

/// Lane-split ordered data compressed with independent, multi-window LZ history.
#[derive(Clone, Debug)]
pub struct LaneLz {
    image: Vec<u8>,
}

impl LaneLz {
    /// Encodes independent byte lanes. Each lane has a private 64 KiB match window;
    /// the LZ-plus-entropy result is stored when it shrinks the input; otherwise
    /// the lane is stored raw.
    pub fn encode(lanes: &[&[u8]]) -> Result<Self, CodecError> {
        LaneLzEncoder::default().encode(lanes)
    }

    /// Splits ordered little-endian u32 values into four byte-plane lanes.
    pub fn encode_u32_byte_planes(values: &[u32]) -> Result<Self, CodecError> {
        LaneLzEncoder::default().encode_u32_byte_planes(values)
    }

    /// Encodes a fixed-width interleaved stream as one independent byte lane per
    /// field. For example, RGBA pixels use `lane_width = 4`.
    pub fn encode_interleaved(input: &[u8], lane_width: usize) -> Result<Self, CodecError> {
        LaneLzEncoder::default().encode_interleaved(input, lane_width)
    }

    fn from_lanes(lanes: Vec<LaneData>, raw_total: usize) -> Result<Self, CodecError> {
        let descriptor_end = HEADER_BYTES
            .checked_add(
                lanes
                    .len()
                    .checked_mul(DESCRIPTOR_BYTES)
                    .ok_or(CodecError::LengthOverflow)?,
            )
            .ok_or(CodecError::LengthOverflow)?;
        let header_len = lanes.iter().try_fold(descriptor_end, |position, lane| {
            position
                .checked_add(lane.model.len())
                .ok_or(CodecError::LengthOverflow)
        })?;
        let mut position = align_up(header_len, 8).ok_or(CodecError::LengthOverflow)?;
        let mut descriptors = Vec::with_capacity(lanes.len());
        let mut model_position = descriptor_end;
        for lane in &lanes {
            let model_offset = if lane.model.is_empty() {
                0
            } else {
                let at = model_position;
                model_position = model_position
                    .checked_add(lane.model.len())
                    .ok_or(CodecError::LengthOverflow)?;
                at
            };
            let payload_offset = position;
            position = position
                .checked_add(lane.payload.len())
                .ok_or(CodecError::LengthOverflow)?;
            descriptors.push(LaneDescriptor {
                raw_len: lane.raw_len,
                token_len: lane.token_len,
                token_count: lane.token_count,
                literal_len: lane.literal_len,
                payload_offset,
                payload_len: lane.payload.len(),
                model_offset,
                model_len: lane.model.len(),
                mode: lane.mode,
                entropy_method: lane.entropy_method,
            });
            position = align_up(position, 8).ok_or(CodecError::LengthOverflow)?;
        }
        if model_position != header_len
            || header_len > u32::MAX as usize
            || position > u32::MAX as usize
        {
            return Err(CodecError::LengthOverflow);
        }

        let mut image = vec![0u8; position];
        image[..4].copy_from_slice(MAGIC);
        put_u16(&mut image, 4, VERSION)?;
        put_u16(
            &mut image,
            6,
            u16::try_from(lanes.len()).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        put_u32(
            &mut image,
            8,
            u32::try_from(raw_total).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        put_u32(
            &mut image,
            12,
            u32::try_from(header_len).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        put_u32(
            &mut image,
            16,
            u32::try_from(position).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        for (index, (lane, descriptor)) in lanes.iter().zip(&descriptors).enumerate() {
            let at = HEADER_BYTES + index * DESCRIPTOR_BYTES;
            put_u32(
                &mut image,
                at,
                u32::try_from(descriptor.raw_len).map_err(|_| CodecError::LengthOverflow)?,
            )?;
            put_u32(
                &mut image,
                at + 4,
                u32::try_from(descriptor.token_len).map_err(|_| CodecError::LengthOverflow)?,
            )?;
            put_u32(
                &mut image,
                at + 8,
                u32::try_from(descriptor.payload_offset).map_err(|_| CodecError::LengthOverflow)?,
            )?;
            put_u32(
                &mut image,
                at + 12,
                u32::try_from(descriptor.payload_len).map_err(|_| CodecError::LengthOverflow)?,
            )?;
            put_u32(
                &mut image,
                at + 16,
                u32::try_from(descriptor.model_offset).map_err(|_| CodecError::LengthOverflow)?,
            )?;
            put_u16(
                &mut image,
                at + 20,
                u16::try_from(descriptor.model_len).map_err(|_| CodecError::LengthOverflow)?,
            )?;
            image[at + 22] = descriptor.mode;
            image[at + 23] = descriptor.entropy_method;
            put_u32(
                &mut image,
                at + 24,
                u32::try_from(descriptor.token_count).map_err(|_| CodecError::LengthOverflow)?,
            )?;
            put_u32(
                &mut image,
                at + 28,
                u32::try_from(descriptor.literal_len).map_err(|_| CodecError::LengthOverflow)?,
            )?;
            if !lane.model.is_empty() {
                let end = descriptor.model_offset + descriptor.model_len;
                image[descriptor.model_offset..end].copy_from_slice(&lane.model);
            }
            let end = descriptor.payload_offset + descriptor.payload_len;
            image[descriptor.payload_offset..end].copy_from_slice(&lane.payload);
        }
        Ok(Self { image })
    }

    /// Parses and validates the serialized lane container.
    pub fn from_bytes(bytes: &[u8]) -> Result<Self, CodecError> {
        parse_image(bytes)?;
        Ok(Self {
            image: bytes.to_vec(),
        })
    }

    /// Serialized lane container.
    #[inline]
    pub fn as_bytes(&self) -> &[u8] {
        &self.image
    }

    /// Number of lanes stored.
    #[inline]
    pub fn lane_count(&self) -> usize {
        read_u16(&self.image, 6).expect("validated lane header") as usize
    }

    /// Original byte count across all lanes.
    #[inline]
    pub fn raw_bytes(&self) -> usize {
        read_u32(&self.image, 8).expect("validated lane header") as usize
    }

    /// Decodes each lane to its original bytes.
    pub fn decode_lanes(&self) -> Result<Vec<Vec<u8>>, CodecError> {
        // All constructors validate or generate this immutable image. Repeating
        // the full structural validation here would only re-read trusted bytes.
        let count = read_u16(&self.image, 6).expect("validated lane image") as usize;
        let version = read_u16(&self.image, 4).expect("validated lane image");
        let is_legacy = version == LEGACY_VERSION;
        let mut lanes = Vec::with_capacity(count);
        for index in 0..count {
            let descriptor = lane_descriptor_unchecked_version(&self.image, index, version);
            let payload = &self.image
                [descriptor.payload_offset..descriptor.payload_offset + descriptor.payload_len];
            let model = if descriptor.model_len == 0 {
                &[][..]
            } else {
                &self.image[descriptor.model_offset..descriptor.model_offset + descriptor.model_len]
            };
            let decoded = rans::entropy_decode(
                descriptor.entropy_method,
                descriptor.token_len,
                model,
                payload,
            )?;
            let lane = match descriptor.mode {
                MODE_RAW => decoded,
                MODE_LZ_ENTROPY if is_legacy => lz_decode_legacy(&decoded, descriptor.raw_len)?,
                MODE_LZ_ENTROPY => lz_decode(
                    &decoded,
                    descriptor.token_count,
                    descriptor.literal_len,
                    descriptor.raw_len,
                )?,
                _ => return Err(CodecError::InvalidFormat("unknown RNBW lane mode")),
            };
            // Every accepted decoder branch enforces the descriptor length.
            lanes.push(lane);
        }
        Ok(lanes)
    }

    /// Reassembles four little-endian byte planes into their ordered u32 values.
    pub fn decode_u32_byte_planes(&self) -> Result<Vec<u32>, CodecError> {
        let lanes = self.decode_lanes()?;
        if lanes.len() != 4 || lanes.iter().any(|lane| lane.len() != lanes[0].len()) {
            return Err(CodecError::InvalidFormat(
                "expected four equal byte-plane lanes",
            ));
        }
        let mut values = Vec::with_capacity(lanes[0].len());
        for (index, &low_byte) in lanes[0].iter().enumerate() {
            values.push(
                u32::from(low_byte)
                    | (u32::from(lanes[1][index]) << 8)
                    | (u32::from(lanes[2][index]) << 16)
                    | (u32::from(lanes[3][index]) << 24),
            );
        }
        Ok(values)
    }

    /// Decodes lanes to fixed-width interleaved records.
    pub fn decode_interleaved(&self) -> Result<Vec<u8>, CodecError> {
        Ok(self.decode_interleaved_cow()?.into_owned())
    }

    /// Decodes fixed-width interleaved records, borrowing the image payload
    /// when a single raw lane already has the requested byte layout.
    pub fn decode_interleaved_cow(&self) -> Result<Cow<'_, [u8]>, CodecError> {
        // A single raw lane is already the complete interleaved output. Return
        // a view instead of allocating or copying the payload.
        if read_u16(&self.image, 6).expect("validated lane image") == 1 {
            let descriptor = lane_descriptor_unchecked(&self.image, 0);
            if descriptor.mode == MODE_RAW && descriptor.entropy_method == METHOD_RAW {
                if descriptor.payload_len != descriptor.raw_len {
                    return Err(CodecError::InvalidFormat("invalid raw lane length"));
                }
                let end = descriptor
                    .payload_offset
                    .checked_add(descriptor.payload_len)
                    .ok_or(CodecError::LengthOverflow)?;
                return Ok(Cow::Borrowed(&self.image[descriptor.payload_offset..end]));
            }
        }

        let lanes = self.decode_lanes()?;
        if lanes.is_empty() || lanes.iter().any(|lane| lane.len() != lanes[0].len()) {
            return Err(CodecError::InvalidFormat("unequal RNBW interleaved lanes"));
        }
        if lanes.len() == 1 {
            return Ok(Cow::Owned(lanes.into_iter().next().unwrap()));
        }
        let output_len = lanes[0]
            .len()
            .checked_mul(lanes.len())
            .ok_or(CodecError::LengthOverflow)?;
        if lanes.len() == 4 {
            let mut output = Vec::with_capacity(output_len);
            for index in 0..lanes[0].len() {
                output.push(lanes[0][index]);
                output.push(lanes[1][index]);
                output.push(lanes[2][index]);
                output.push(lanes[3][index]);
            }
            return Ok(Cow::Owned(output));
        }
        let mut output = Vec::with_capacity(output_len);
        for (record_index, _) in lanes[0].iter().enumerate() {
            for lane in &lanes {
                output.push(lane[record_index]);
            }
        }
        Ok(Cow::Owned(output))
    }
}

/// Reusable LZ and entropy scratch for repeated encodes.
///
/// The ordinary `LaneLz::encode` API remains convenient for one-shot calls.
/// Callers that encode repeatedly can retain this object to amortize the 64 KiB
/// hash table, 64 KiB window links, and token buffer allocations.
#[derive(Debug, Default)]
pub struct LaneLzEncoder {
    serial: LzWorkspace,
    parallel: Vec<LzWorkspace>,
    interleaved_lanes: Vec<Vec<u8>>,
}

impl LaneLzEncoder {
    /// Encodes independent lanes serially while retaining scratch allocations.
    pub fn encode(&mut self, lanes: &[&[u8]]) -> Result<LaneLz, CodecError> {
        let raw_total = validate_lanes(lanes)?;
        encode_serial_lanes(lanes, raw_total, &mut self.serial)
    }

    /// Encodes independent lanes in parallel, retaining one LZ workspace per
    /// active Rayon worker. Very small lane counts use the serial path.
    pub fn encode_parallel(&mut self, lanes: &[&[u8]]) -> Result<LaneLz, CodecError> {
        let raw_total = validate_lanes(lanes)?;
        encode_parallel_lanes(lanes, raw_total, &mut self.serial, &mut self.parallel)
    }

    /// Splits and encodes interleaved records while retaining the lane buffers.
    ///
    /// This avoids allocating and refilling fresh SoA buffers on repeated image,
    /// shader, and model-data encodes.
    pub fn encode_interleaved(
        &mut self,
        input: &[u8],
        lane_width: usize,
    ) -> Result<LaneLz, CodecError> {
        if lane_width == 1 {
            let raw_total = validate_lanes(&[input])?;
            return encode_serial_lanes(&[input], raw_total, &mut self.serial);
        }
        let LaneLzEncoder {
            serial,
            interleaved_lanes,
            ..
        } = self;
        let raw_total = split_interleaved(input, lane_width, interleaved_lanes)?;
        if lane_width == 4 {
            let lane_refs = [
                interleaved_lanes[0].as_slice(),
                interleaved_lanes[1].as_slice(),
                interleaved_lanes[2].as_slice(),
                interleaved_lanes[3].as_slice(),
            ];
            return encode_serial_lanes(&lane_refs, raw_total, serial);
        }
        let lane_refs: Vec<&[u8]> = interleaved_lanes
            .iter()
            .take(lane_width)
            .map(Vec::as_slice)
            .collect();
        encode_serial_lanes(&lane_refs, raw_total, serial)
    }

    /// Parallel version of [`LaneLzEncoder::encode_interleaved`].
    pub fn encode_interleaved_parallel(
        &mut self,
        input: &[u8],
        lane_width: usize,
    ) -> Result<LaneLz, CodecError> {
        if lane_width == 1 {
            let raw_total = validate_lanes(&[input])?;
            return encode_parallel_lanes(
                &[input],
                raw_total,
                &mut self.serial,
                &mut self.parallel,
            );
        }
        let LaneLzEncoder {
            serial,
            parallel,
            interleaved_lanes,
        } = self;
        let raw_total = split_interleaved(input, lane_width, interleaved_lanes)?;
        if lane_width == 4 {
            let lane_refs = [
                interleaved_lanes[0].as_slice(),
                interleaved_lanes[1].as_slice(),
                interleaved_lanes[2].as_slice(),
                interleaved_lanes[3].as_slice(),
            ];
            return encode_parallel_lanes(&lane_refs, raw_total, serial, parallel);
        }
        let lane_refs: Vec<&[u8]> = interleaved_lanes
            .iter()
            .take(lane_width)
            .map(Vec::as_slice)
            .collect();
        encode_parallel_lanes(&lane_refs, raw_total, serial, parallel)
    }

    /// Splits little-endian u32 values into reusable byte-plane buffers.
    pub fn encode_u32_byte_planes(&mut self, values: &[u32]) -> Result<LaneLz, CodecError> {
        let LaneLzEncoder {
            serial,
            interleaved_lanes,
            ..
        } = self;
        split_u32_byte_planes(values, interleaved_lanes)?;
        let lane_refs = [
            interleaved_lanes[0].as_slice(),
            interleaved_lanes[1].as_slice(),
            interleaved_lanes[2].as_slice(),
            interleaved_lanes[3].as_slice(),
        ];
        let raw_total = values
            .len()
            .checked_mul(4)
            .ok_or(CodecError::LengthOverflow)?;
        encode_serial_lanes(&lane_refs, raw_total, serial)
    }

    /// Parallel version of [`LaneLzEncoder::encode_u32_byte_planes`].
    pub fn encode_u32_byte_planes_parallel(
        &mut self,
        values: &[u32],
    ) -> Result<LaneLz, CodecError> {
        let LaneLzEncoder {
            serial,
            parallel,
            interleaved_lanes,
        } = self;
        split_u32_byte_planes(values, interleaved_lanes)?;
        let lane_refs = [
            interleaved_lanes[0].as_slice(),
            interleaved_lanes[1].as_slice(),
            interleaved_lanes[2].as_slice(),
            interleaved_lanes[3].as_slice(),
        ];
        let raw_total = values
            .len()
            .checked_mul(4)
            .ok_or(CodecError::LengthOverflow)?;
        encode_parallel_lanes(&lane_refs, raw_total, serial, parallel)
    }
}

fn parse_image(image: &[u8]) -> Result<Vec<LaneDescriptor>, CodecError> {
    if image.len() < HEADER_BYTES {
        return Err(CodecError::Truncated);
    }
    let version = read_u16(image, 4)?;
    if image.get(..4) != Some(MAGIC.as_slice()) || !matches!(version, LEGACY_VERSION | VERSION) {
        return Err(CodecError::InvalidFormat("bad RNBW lane signature"));
    }
    let descriptor_bytes = if version == LEGACY_VERSION {
        LEGACY_DESCRIPTOR_BYTES
    } else {
        DESCRIPTOR_BYTES
    };
    let lane_count = read_u16(image, 6)? as usize;
    let raw_total = read_u32(image, 8)? as usize;
    let header_len = read_u32(image, 12)? as usize;
    if read_u32(image, 16)? as usize != image.len() || lane_count == 0 {
        return Err(CodecError::InvalidFormat("invalid RNBW lane header"));
    }
    let descriptor_end = HEADER_BYTES
        .checked_add(
            lane_count
                .checked_mul(descriptor_bytes)
                .ok_or(CodecError::LengthOverflow)?,
        )
        .ok_or(CodecError::LengthOverflow)?;
    if descriptor_end > header_len || header_len > image.len() {
        return Err(CodecError::InvalidFormat("invalid RNBW lane header range"));
    }

    let mut descriptors = Vec::with_capacity(lane_count);
    let mut raw_total_actual = 0usize;
    let mut model_position = descriptor_end;
    let mut payload_position = align_up(header_len, 8).ok_or(CodecError::LengthOverflow)?;
    if payload_position > image.len() {
        return Err(CodecError::InvalidFormat(
            "invalid RNBW lane header padding",
        ));
    }
    if image[header_len..payload_position]
        .iter()
        .any(|&byte| byte != 0)
    {
        return Err(CodecError::InvalidFormat(
            "nonzero RNBW lane header padding",
        ));
    }
    for index in 0..lane_count {
        let at = HEADER_BYTES + index * descriptor_bytes;
        let descriptor = LaneDescriptor {
            raw_len: read_u32(image, at)? as usize,
            token_len: read_u32(image, at + 4)? as usize,
            token_count: if version == LEGACY_VERSION {
                0
            } else {
                read_u32(image, at + 24)? as usize
            },
            literal_len: if version == LEGACY_VERSION {
                0
            } else {
                read_u32(image, at + 28)? as usize
            },
            payload_offset: read_u32(image, at + 8)? as usize,
            payload_len: read_u32(image, at + 12)? as usize,
            model_offset: read_u32(image, at + 16)? as usize,
            model_len: read_u16(image, at + 20)? as usize,
            mode: image[at + 22],
            entropy_method: image[at + 23],
        };
        raw_total_actual = raw_total_actual
            .checked_add(descriptor.raw_len)
            .ok_or(CodecError::LengthOverflow)?;
        let expected_model_len = match descriptor.entropy_method {
            METHOD_RAW => 0,
            METHOD_RANS => 512,
            METHOD_HUFFMAN => 256,
            _ => {
                return Err(CodecError::InvalidFormat(
                    "unknown RNBW lane entropy method",
                ));
            }
        };
        if descriptor.model_len != expected_model_len {
            return Err(CodecError::InvalidFormat("invalid RNBW lane model length"));
        }
        match descriptor.mode {
            MODE_RAW => {
                if descriptor.entropy_method != METHOD_RAW
                    || descriptor.token_len != descriptor.raw_len
                    || descriptor.payload_len != descriptor.raw_len
                    || descriptor.token_count != 0
                    || descriptor.literal_len != 0
                {
                    return Err(CodecError::InvalidFormat("invalid raw RNBW lane"));
                }
            }
            MODE_LZ_ENTROPY => {
                let stored_len = descriptor
                    .payload_len
                    .checked_add(descriptor.model_len)
                    .ok_or(CodecError::LengthOverflow)?;
                let token_stream_len = descriptor
                    .token_count
                    .checked_mul(TOKEN_BYTES)
                    .and_then(|length| length.checked_add(descriptor.literal_len))
                    .ok_or(CodecError::LengthOverflow)?;
                let malformed_v1 = version == LEGACY_VERSION
                    && descriptor.token_len
                        > descriptor
                            .raw_len
                            .checked_mul(2)
                            .and_then(|length| length.checked_add(1))
                            .ok_or(CodecError::LengthOverflow)?;
                let malformed_v2 = version == VERSION
                    && (descriptor.token_count == 0
                        || descriptor.token_count > descriptor.raw_len
                        || descriptor.literal_len > descriptor.raw_len
                        || token_stream_len != descriptor.token_len);
                if stored_len >= descriptor.raw_len || malformed_v1 || malformed_v2 {
                    return Err(CodecError::InvalidFormat("invalid LZ RNBW lane"));
                }
            }
            _ => return Err(CodecError::InvalidFormat("unknown RNBW lane mode")),
        }
        if descriptor.entropy_method == METHOD_RANS && descriptor.payload_len < 32 {
            return Err(CodecError::InvalidFormat("truncated RNBW lane rANS stream"));
        }
        if descriptor.entropy_method == METHOD_HUFFMAN && descriptor.payload_len == 0 {
            return Err(CodecError::InvalidFormat(
                "truncated RNBW lane Huffman stream",
            ));
        }
        if descriptor.model_len == 0 {
            if descriptor.model_offset != 0 {
                return Err(CodecError::InvalidFormat(
                    "unexpected RNBW lane model offset",
                ));
            }
        } else {
            if descriptor.model_offset != model_position {
                return Err(CodecError::InvalidFormat("noncontiguous RNBW lane models"));
            }
            model_position = model_position
                .checked_add(descriptor.model_len)
                .ok_or(CodecError::LengthOverflow)?;
        }
        if descriptor.payload_offset != payload_position {
            return Err(CodecError::InvalidFormat("noncontiguous RNBW lane payload"));
        }
        let payload_end = descriptor
            .payload_offset
            .checked_add(descriptor.payload_len)
            .ok_or(CodecError::LengthOverflow)?;
        if payload_end > image.len() {
            return Err(CodecError::Truncated);
        }
        payload_position = align_up(payload_end, 8).ok_or(CodecError::LengthOverflow)?;
        if payload_position > image.len()
            || image[payload_end..payload_position]
                .iter()
                .any(|&byte| byte != 0)
        {
            return Err(CodecError::InvalidFormat("bad RNBW lane payload padding"));
        }
        descriptors.push(descriptor);
    }
    if raw_total_actual != raw_total
        || model_position != header_len
        || payload_position != image.len()
    {
        return Err(CodecError::InvalidFormat(
            "RNBW lane lengths disagree with header",
        ));
    }
    Ok(descriptors)
}

#[inline]
fn lane_descriptor_unchecked(image: &[u8], index: usize) -> LaneDescriptor {
    let version = u16::from_le_bytes([image[4], image[5]]);
    lane_descriptor_unchecked_version(image, index, version)
}

#[inline]
fn lane_descriptor_unchecked_version(image: &[u8], index: usize, version: u16) -> LaneDescriptor {
    let descriptor_bytes = if version == LEGACY_VERSION {
        LEGACY_DESCRIPTOR_BYTES
    } else {
        DESCRIPTOR_BYTES
    };
    let at = HEADER_BYTES + index * descriptor_bytes;
    LaneDescriptor {
        raw_len: read_u32_unchecked(image, at) as usize,
        token_len: read_u32_unchecked(image, at + 4) as usize,
        token_count: if version == LEGACY_VERSION {
            0
        } else {
            read_u32_unchecked(image, at + 24) as usize
        },
        literal_len: if version == LEGACY_VERSION {
            0
        } else {
            read_u32_unchecked(image, at + 28) as usize
        },
        payload_offset: read_u32_unchecked(image, at + 8) as usize,
        payload_len: read_u32_unchecked(image, at + 12) as usize,
        model_offset: read_u32_unchecked(image, at + 16) as usize,
        model_len: u16::from_le_bytes([image[at + 20], image[at + 21]]) as usize,
        mode: image[at + 22],
        entropy_method: image[at + 23],
    }
}

fn validate_lanes(lanes: &[&[u8]]) -> Result<usize, CodecError> {
    if lanes.is_empty() || lanes.len() > u16::MAX as usize {
        return Err(CodecError::InvalidFormat("invalid RNBW lane count"));
    }
    let raw_total = lanes.iter().try_fold(0usize, |total, lane| {
        if lane.len() > u32::MAX as usize {
            return Err(CodecError::LengthOverflow);
        }
        total
            .checked_add(lane.len())
            .ok_or(CodecError::LengthOverflow)
    })?;
    if raw_total > u32::MAX as usize {
        return Err(CodecError::LengthOverflow);
    }
    Ok(raw_total)
}

fn encode_serial_lanes(
    lanes: &[&[u8]],
    raw_total: usize,
    workspace: &mut LzWorkspace,
) -> Result<LaneLz, CodecError> {
    let hash_bits = if lanes.len() == 1 {
        HASH_BITS_WIDE
    } else {
        HASH_BITS_COMPACT
    };
    let mut stored = Vec::with_capacity(lanes.len());
    for lane in lanes {
        stored.push(encode_lane(lane, workspace, hash_bits)?);
    }
    LaneLz::from_lanes(stored, raw_total)
}

fn encode_parallel_lanes(
    lanes: &[&[u8]],
    raw_total: usize,
    serial: &mut LzWorkspace,
    parallel: &mut Vec<LzWorkspace>,
) -> Result<LaneLz, CodecError> {
    if raw_total < PARALLEL_MIN_BYTES {
        return encode_serial_lanes(lanes, raw_total, serial);
    }
    if lanes.len() == 1 {
        let lane = lanes[0];
        if should_skip_lz_sample(lane) {
            return LaneLz::from_lanes(vec![raw_lane_data(lane)], raw_total);
        }
        if lane.len() <= LZ_BLOCK_BYTES {
            return encode_serial_lanes(lanes, raw_total, serial);
        }
        let worker_count = rayon_worker_count();
        let block_count = lane.len().div_ceil(LZ_BLOCK_BYTES);
        let tasks = worker_count.min(block_count);
        if tasks < 2 {
            return encode_serial_lanes(lanes, raw_total, serial);
        }
        return encode_parallel_single_lane(lane, raw_total, worker_count, serial, parallel);
    }
    if lanes.len() < 2 {
        return encode_serial_lanes(lanes, raw_total, serial);
    }
    let worker_count = rayon_worker_count();
    let workers = worker_count.min(lanes.len());
    if workers < 2 {
        return encode_serial_lanes(lanes, raw_total, serial);
    }
    let lanes_per_task = lanes.len().div_ceil(workers);
    let tasks = lanes.len().div_ceil(lanes_per_task);
    parallel.resize_with(tasks, LzWorkspace::default);
    parallel.truncate(tasks);

    let mut stored: Vec<Option<LaneData>> =
        std::iter::repeat_with(|| None).take(lanes.len()).collect();
    parallel
        .par_iter_mut()
        .zip(stored.par_chunks_mut(lanes_per_task))
        .zip(lanes.par_chunks(lanes_per_task))
        .try_for_each(|((workspace, output), input)| -> Result<(), CodecError> {
            for (slot, lane) in output.iter_mut().zip(input) {
                *slot = Some(encode_lane(lane, workspace, HASH_BITS_COMPACT)?);
            }
            Ok(())
        })?;
    let stored = stored
        .into_iter()
        .map(|lane| lane.expect("parallel lane task filled every output slot"))
        .collect();
    LaneLz::from_lanes(stored, raw_total)
}

fn prepare_split_lanes(lanes: &mut Vec<Vec<u8>>, lane_count: usize, records: usize) {
    if lanes.len() < lane_count {
        lanes.resize_with(lane_count, Vec::new);
    }
    for lane in lanes.iter_mut().take(lane_count) {
        lane.clear();
        lane.reserve(records);
    }
}

fn split_interleaved(
    input: &[u8],
    lane_width: usize,
    lanes: &mut Vec<Vec<u8>>,
) -> Result<usize, CodecError> {
    if lane_width == 0 || lane_width > u16::MAX as usize || !input.len().is_multiple_of(lane_width)
    {
        return Err(CodecError::InvalidFormat(
            "invalid interleaved RNBW lane width",
        ));
    }
    if input.len() > u32::MAX as usize {
        return Err(CodecError::LengthOverflow);
    }
    let records = input.len() / lane_width;
    prepare_split_lanes(lanes, lane_width, records);
    if lane_width == 4 {
        split_four_dispatch(input, &mut lanes[..4], records);
        return Ok(input.len());
    }
    for record in input.chunks_exact(lane_width) {
        for (lane, byte) in lanes.iter_mut().zip(record) {
            lane.push(*byte);
        }
    }
    Ok(input.len())
}

fn split_four_dispatch(input: &[u8], lanes: &mut [Vec<u8>], records: usize) {
    debug_assert_eq!(lanes.len(), 4);
    let split = *SPLIT_FOUR_DISPATCH.get_or_init(|| {
        #[cfg(any(target_arch = "x86", target_arch = "x86_64"))]
        if std::is_x86_feature_detected!("avx2") {
            return split_four_avx2_entry;
        }
        split_four_scalar
    });
    split(input, lanes, records);
}

fn split_four_scalar(input: &[u8], lanes: &mut [Vec<u8>], records: usize) {
    for lane in lanes.iter_mut() {
        lane.resize(records, 0);
    }
    for (record_index, record) in input.chunks_exact(4).enumerate() {
        lanes[0][record_index] = record[0];
        lanes[1][record_index] = record[1];
        lanes[2][record_index] = record[2];
        lanes[3][record_index] = record[3];
    }
}

#[cfg(any(target_arch = "x86", target_arch = "x86_64"))]
fn split_four_avx2_entry(input: &[u8], lanes: &mut [Vec<u8>], records: usize) {
    // The dispatch cache calls this entry only after AVX2 feature detection.
    unsafe { split_four_avx2(input, lanes, records) }
}

#[cfg(any(target_arch = "x86", target_arch = "x86_64"))]
#[target_feature(enable = "avx2")]
unsafe fn split_four_avx2(input: &[u8], lanes: &mut [Vec<u8>], records: usize) {
    const M0: [i8; 32] = [
        0, 4, 8, 12, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, 0, 4,
        8, 12, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128,
    ];
    const M1: [i8; 32] = [
        1, 5, 9, 13, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, 1, 5,
        9, 13, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128,
    ];
    const M2: [i8; 32] = [
        2, 6, 10, 14, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, 2, 6,
        10, 14, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128,
    ];
    const M3: [i8; 32] = [
        3, 7, 11, 15, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, 3, 7,
        11, 15, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128, -128,
    ];

    for lane in lanes.iter_mut() {
        lane.resize(records, 0);
    }
    let masks = [M0, M1, M2, M3].map(|mask| unsafe { _mm256_loadu_si256(mask.as_ptr().cast()) });
    let pack = _mm256_setr_epi32(0, 4, 0, 4, 0, 4, 0, 4);
    let vector_records = records & !7;
    let mut record = 0usize;
    while record < vector_records {
        let bytes = unsafe { input.as_ptr().add(record * 4) };
        let interleaved = unsafe { _mm256_loadu_si256(bytes.cast()) };
        for channel in 0..4 {
            let separated = _mm256_shuffle_epi8(interleaved, masks[channel]);
            let packed = _mm256_permutevar8x32_epi32(separated, pack);
            let lane_out = unsafe { lanes[channel].as_mut_ptr().add(record).cast::<__m128i>() };
            unsafe { _mm_storel_epi64(lane_out, _mm256_castsi256_si128(packed)) };
        }
        record += 8;
    }
    while record < records {
        let source = unsafe { input.as_ptr().add(record * 4) };
        for channel in 0..4 {
            lanes[channel][record] = unsafe { *source.add(channel) };
        }
        record += 1;
    }
}

fn split_u32_byte_planes(values: &[u32], lanes: &mut Vec<Vec<u8>>) -> Result<(), CodecError> {
    let raw_total = values
        .len()
        .checked_mul(4)
        .ok_or(CodecError::LengthOverflow)?;
    if raw_total > u32::MAX as usize {
        return Err(CodecError::LengthOverflow);
    }
    prepare_split_lanes(lanes, 4, values.len());

    // On little-endian targets the in-memory u32 representation is already the
    // byte stream consumed by the four-lane splitter. Reuse its cached AVX2
    // dispatch instead of materializing four scalar `to_le_bytes` pushes per
    // value. The byte view is valid for exactly `raw_total` initialized bytes.
    #[cfg(target_endian = "little")]
    {
        // SAFETY: `u32` has no padding, the checked length is `values.len() * 4`,
        // and `u8` has alignment 1. The view does not outlive `values`.
        let bytes = unsafe { std::slice::from_raw_parts(values.as_ptr().cast::<u8>(), raw_total) };
        split_four_dispatch(bytes, &mut lanes[..4], values.len());
        Ok(())
    }

    #[cfg(target_endian = "big")]
    {
        for &value in values {
            let [x0, x1, x2, x3] = value.to_le_bytes();
            lanes[0].push(x0);
            lanes[1].push(x1);
            lanes[2].push(x2);
            lanes[3].push(x3);
        }
        Ok(())
    }
}

fn encode_lane(
    input: &[u8],
    workspace: &mut LzWorkspace,
    hash_bits: usize,
) -> Result<LaneData, CodecError> {
    if should_skip_lz_sample(input) {
        return Ok(raw_lane_data(input));
    }
    let (token_count, literal_len) = lz_encode(input, workspace, hash_bits);
    choose_lane_data(input, &workspace.tokens, token_count, literal_len)
}

fn encode_parallel_single_lane(
    input: &[u8],
    raw_total: usize,
    worker_count: usize,
    serial: &mut LzWorkspace,
    parallel: &mut Vec<LzWorkspace>,
) -> Result<LaneLz, CodecError> {
    let block_count = input.len().div_ceil(LZ_BLOCK_BYTES);
    let task_count = worker_count.min(block_count);
    let blocks_per_task = block_count.div_ceil(task_count);
    let task_count = block_count.div_ceil(blocks_per_task);
    parallel.resize_with(task_count, LzWorkspace::default);
    parallel.truncate(task_count);

    parallel
        .par_iter_mut()
        .enumerate()
        .for_each(|(task_index, workspace)| {
            let first_block = task_index * blocks_per_task;
            let end_block = (first_block + blocks_per_task).min(block_count);
            let start = first_block * LZ_BLOCK_BYTES;
            let end = (end_block * LZ_BLOCK_BYTES).min(input.len());
            encode_lz_range(input, start, end, workspace, HASH_BITS_WIDE);
        });

    serial.block_tokens.clear();
    serial.block_literals.clear();
    for workspace in parallel.iter() {
        serial
            .block_tokens
            .extend_from_slice(&workspace.block_tokens);
        serial
            .block_literals
            .extend_from_slice(&workspace.block_literals);
    }
    serial.tokens.clear();
    serial
        .tokens
        .reserve(serial.block_tokens.len() + serial.block_literals.len());
    serial.tokens.extend_from_slice(&serial.block_tokens);
    serial.tokens.extend_from_slice(&serial.block_literals);
    let token_count = serial.block_tokens.len() / TOKEN_BYTES;
    let literal_len = serial.block_literals.len();
    let stored = choose_lane_data(input, &serial.tokens, token_count, literal_len)?;
    LaneLz::from_lanes(vec![stored], raw_total)
}

fn encode_lz_range(
    input: &[u8],
    start: usize,
    end: usize,
    workspace: &mut LzWorkspace,
    hash_bits: usize,
) {
    workspace.block_tokens.clear();
    workspace.block_literals.clear();
    let mut block_start = start;
    while block_start < end {
        let block_end = (block_start + LZ_BLOCK_BYTES).min(end);
        let lookahead_end = (block_end + MAX_MATCH).min(input.len());
        let history_start = block_start.saturating_sub(WINDOW_SIZE);
        let context = &input[history_start..lookahead_end];
        let history_len = block_start - history_start;
        let target_len = block_end - block_start;
        let (token_count, literal_len) =
            lz_encode_window(context, history_len, target_len, workspace, hash_bits);
        let token_bytes = token_count * TOKEN_BYTES;
        workspace
            .block_tokens
            .extend_from_slice(&workspace.tokens[..token_bytes]);
        workspace
            .block_literals
            .extend_from_slice(&workspace.tokens[token_bytes..token_bytes + literal_len]);
        block_start = block_end;
    }
}

fn rayon_worker_count() -> usize {
    *RAYON_WORKERS.get_or_init(|| {
        // The process-wide Rayon pool is persistent and uses work stealing.
        // Reuse an embedding application's pool when it initialized Rayon first.
        let _ = rayon::ThreadPoolBuilder::new().build_global();
        rayon::current_num_threads()
    })
}

fn should_skip_lz_sample(input: &[u8]) -> bool {
    const MIN_BYTES: usize = 4096;
    const SAMPLE_BYTES: usize = 32;
    const SAMPLE_COUNT: usize = 64;
    const GRAM_TABLE_SIZE: usize = 4096;
    if input.len() < MIN_BYTES {
        return false;
    }

    let mut byte_counts = [0u16; 256];
    let mut grams = [0u32; GRAM_TABLE_SIZE];
    let mut sample_bytes = 0usize;
    let mut duplicate_grams = 0usize;
    let max_start = input.len() - SAMPLE_BYTES;
    for sample in 0..SAMPLE_COUNT {
        let start = sample * max_start / (SAMPLE_COUNT - 1);
        let bytes = &input[start..start + SAMPLE_BYTES];
        for &byte in bytes {
            byte_counts[byte as usize] += 1;
        }
        sample_bytes += bytes.len();
        for gram in bytes.windows(3) {
            let key = (u32::from(gram[0]) << 16) | (u32::from(gram[1]) << 8) | u32::from(gram[2]);
            let stored_key = key + 1;
            let mut slot = key.wrapping_mul(0x9e37_79b1) as usize & (GRAM_TABLE_SIZE - 1);
            loop {
                match grams[slot] {
                    0 => {
                        grams[slot] = stored_key;
                        break;
                    }
                    existing if existing == stored_key => {
                        duplicate_grams += 1;
                        break;
                    }
                    _ => slot = (slot + 1) & (GRAM_TABLE_SIZE - 1),
                }
            }
        }
    }

    let distinct_bytes = byte_counts.iter().filter(|&&count| count != 0).count();
    let max_frequency = byte_counts.iter().copied().max().unwrap_or(0) as usize;
    distinct_bytes >= 200 && max_frequency <= sample_bytes / 64 && duplicate_grams <= 8
}

fn raw_lane_data(input: &[u8]) -> LaneData {
    LaneData {
        raw_len: input.len(),
        token_len: input.len(),
        token_count: 0,
        literal_len: 0,
        model: Vec::new(),
        payload: input.to_vec(),
        mode: MODE_RAW,
        entropy_method: METHOD_RAW,
    }
}

fn choose_lane_data(
    input: &[u8],
    lz_stream: &[u8],
    token_count: usize,
    literal_len: usize,
) -> Result<LaneData, CodecError> {
    // lz_encode has already counted every literal byte while testing its
    // sliding history window. A very low literal fraction means the 64 KiB
    // window found strong repetition, so give the fixed-capacity Huffman path
    // first refusal and skip rANS when that cheaper code shrinks the tokens.
    let prefer_huffman = input.len() >= 4096 && literal_len <= input.len() / 8;
    let Some((method, model, payload)) =
        rans::entropy_encode_candidate_with_hint(lz_stream, prefer_huffman)?
    else {
        return Ok(raw_lane_data(input));
    };
    if model.len() + payload.len() >= input.len() {
        return Ok(raw_lane_data(input));
    }
    Ok(LaneData {
        raw_len: input.len(),
        token_len: lz_stream.len(),
        token_count,
        literal_len,
        model,
        payload,
        mode: MODE_LZ_ENTROPY,
        entropy_method: method,
    })
}

#[cfg(feature = "wgpu")]
fn encode_lane_from_lz(
    input: &[u8],
    lz_stream: &[u8],
    token_count: usize,
    literal_len: usize,
) -> Result<LaneData, CodecError> {
    choose_lane_data(input, lz_stream, token_count, literal_len)
}

#[derive(Debug, Default)]
struct LzWorkspace {
    // Each hash slot packs its two newest matching positions. Directly
    // testing both candidates avoids the dependent pointer chain.
    heads: Vec<u64>,
    touched_heads: Vec<u32>,
    tokens: Vec<u8>,
    literal_bytes: Vec<u8>,
    block_tokens: Vec<u8>,
    block_literals: Vec<u8>,
}

fn lz_encode(input: &[u8], workspace: &mut LzWorkspace, hash_bits: usize) -> (usize, usize) {
    lz_encode_window(input, 0, input.len(), workspace, hash_bits)
}

fn lz_encode_window(
    input: &[u8],
    history_len: usize,
    target_len: usize,
    workspace: &mut LzWorkspace,
    hash_bits: usize,
) -> (usize, usize) {
    let LzWorkspace {
        heads,
        touched_heads,
        tokens: output,
        literal_bytes,
        ..
    } = workspace;
    let target_end = history_len + target_len;
    debug_assert!(target_end <= input.len());
    output.clear();
    literal_bytes.clear();
    output.reserve(target_len);
    if target_len == 0 {
        return (0, 0);
    }
    if input.len() < 3 {
        emit_literals(output, literal_bytes, &input[history_len..target_end]);
        return finish_token_stream(output, literal_bytes);
    }
    for hash in touched_heads.drain(..) {
        heads[hash as usize] = u64::MAX;
    }
    let hash_size = 1usize << hash_bits;
    if heads.len() != hash_size {
        heads.resize(hash_size, u64::MAX);
    }
    for index in 0..history_len {
        if index + 2 < input.len() {
            let hash = hash_at(input, index, hash_bits);
            insert_position(index, hash, heads, touched_heads);
        }
    }

    let mut index = history_len;
    let mut literal_start = history_len;

    while index < target_end && index + 2 < input.len() {
        let hash = hash_at(input, index, hash_bits);
        let candidates = heads[hash];
        insert_position(index, hash, heads, touched_heads);
        let mut best_len = 0usize;
        let mut best_distance = 0usize;
        let candidate0 = candidates as u32;
        if candidate0 != u32::MAX {
            let candidate_index = candidate0 as usize;
            let distance = index - candidate_index;
            if distance <= WINDOW_SIZE
                && input[candidate_index] == input[index]
                && input[candidate_index + 1] == input[index + 1]
                && input[candidate_index + 2] == input[index + 2]
            {
                let length = 3 + super::extend_match(input, candidate_index + 3, index + 3);
                if length > best_len {
                    best_len = length;
                    best_distance = distance;
                }
            }
        }
        let candidate1 = (candidates >> 32) as u32;
        if candidate1 != u32::MAX {
            let candidate_index = candidate1 as usize;
            let distance = index - candidate_index;
            if distance <= WINDOW_SIZE
                && input[candidate_index] == input[index]
                && input[candidate_index + 1] == input[index + 1]
                && input[candidate_index + 2] == input[index + 2]
            {
                let length = 3 + super::extend_match(input, candidate_index + 3, index + 3);
                if length > best_len {
                    best_len = length;
                    best_distance = distance;
                }
            }
        }
        best_len = best_len.min(target_end - index);
        if best_len >= 3 {
            emit_literals(output, literal_bytes, &input[literal_start..index]);
            emit_matches(output, best_len, best_distance);
            let end = index + best_len;
            // The current position was inserted before candidate matching.
            // Insert the rest of the accepted match so following positions can
            // still match against every byte covered by it.
            index += 1;
            while index < end {
                if index + 2 < input.len() {
                    insert_position(
                        index,
                        hash_at(input, index, hash_bits),
                        heads,
                        touched_heads,
                    );
                }
                index += 1;
            }
            literal_start = index;
        } else {
            index += 1;
        }
    }
    emit_literals(output, literal_bytes, &input[literal_start..target_end]);
    finish_token_stream(output, literal_bytes)
}

fn finish_token_stream(output: &mut Vec<u8>, literal_bytes: &[u8]) -> (usize, usize) {
    let token_count = output.len() / TOKEN_BYTES;
    let literal_len = literal_bytes.len();
    output.extend_from_slice(literal_bytes);
    (token_count, literal_len)
}

fn hash_at(input: &[u8], index: usize, hash_bits: usize) -> usize {
    let value = u32::from_le_bytes([input[index], input[index + 1], input[index + 2], 0]);
    (value.wrapping_mul(0x1e35_a7bd) >> (32 - hash_bits)) as usize
}

fn insert_position(index: usize, hash: usize, heads: &mut [u64], touched_heads: &mut Vec<u32>) {
    let previous = heads[hash];
    if previous == u64::MAX {
        touched_heads.push(hash as u32);
    }
    heads[hash] = (u64::from(previous as u32) << 32) | index as u64;
}

fn emit_literals(output: &mut Vec<u8>, literal_bytes: &mut Vec<u8>, mut literal: &[u8]) {
    while !literal.is_empty() {
        let length = literal.len().min(128);
        let token = ((length - 1) as u32) << 24;
        output.extend_from_slice(&token.to_le_bytes());
        literal_bytes.extend_from_slice(&literal[..length]);
        literal = &literal[length..];
    }
}

fn emit_matches(output: &mut Vec<u8>, mut length: usize, distance: usize) {
    while length != 0 {
        let mut part = length.min(MAX_MATCH);
        if length > MAX_MATCH && length - part < 3 {
            part = length - 3;
        }
        let token = 0x8000_0000 | (((part - 3) as u32) << 24) | (((distance - 1) as u32) << 8);
        output.extend_from_slice(&token.to_le_bytes());
        length -= part;
    }
}

fn lz_decode(
    stream: &[u8],
    token_count: usize,
    literal_len: usize,
    output_len: usize,
) -> Result<Vec<u8>, CodecError> {
    let token_bytes = token_count
        .checked_mul(TOKEN_BYTES)
        .ok_or(CodecError::LengthOverflow)?;
    if token_bytes.checked_add(literal_len) != Some(stream.len()) {
        return Err(CodecError::InvalidFormat(
            "invalid fixed-width RNBW token stream",
        ));
    }
    let mut output = Vec::with_capacity(output_len);
    let mut literal_cursor = token_bytes;
    for token_index in 0..token_count {
        let at = token_index * TOKEN_BYTES;
        let token =
            u32::from_le_bytes([stream[at], stream[at + 1], stream[at + 2], stream[at + 3]]);
        let length = ((token >> 24) & 0x7f) as usize;
        if token & 0x8000_0000 == 0 {
            let length = length + 1;
            let end = literal_cursor
                .checked_add(length)
                .ok_or(CodecError::LengthOverflow)?;
            if token & 0x00ff_ffff != 0
                || end > stream.len()
                || output
                    .len()
                    .checked_add(length)
                    .is_none_or(|n| n > output_len)
            {
                return Err(CodecError::InvalidFormat("invalid RNBW literal token"));
            }
            output.extend_from_slice(&stream[literal_cursor..end]);
            literal_cursor = end;
        } else {
            let length = length + 3;
            let distance = (((token >> 8) & 0xffff) as usize) + 1;
            if token & 0xff != 0
                || distance > output.len()
                || output
                    .len()
                    .checked_add(length)
                    .is_none_or(|n| n > output_len)
            {
                return Err(CodecError::InvalidFormat("invalid RNBW match token"));
            }
            let start = output.len();
            let mut copied = distance.min(length);
            output.extend_from_within(start - distance..start - distance + copied);
            while copied < length {
                let amount = copied.min(length - copied);
                output.extend_from_within(start..start + amount);
                copied += amount;
            }
        }
    }
    if literal_cursor != stream.len() || output.len() != output_len {
        return Err(CodecError::InvalidFormat(
            "RNBW fixed token output length mismatch",
        ));
    }
    Ok(output)
}

fn lz_decode_legacy(tokens: &[u8], output_len: usize) -> Result<Vec<u8>, CodecError> {
    let mut output = Vec::with_capacity(output_len);
    let mut cursor = 0usize;
    while cursor < tokens.len() {
        let control = tokens[cursor];
        cursor += 1;
        if control & 0x80 == 0 {
            let length = (control as usize) + 1;
            let end = cursor
                .checked_add(length)
                .ok_or(CodecError::LengthOverflow)?;
            if end > tokens.len()
                || output
                    .len()
                    .checked_add(length)
                    .is_none_or(|n| n > output_len)
            {
                return Err(CodecError::InvalidFormat("invalid RNBW lane literal"));
            }
            output.extend_from_slice(&tokens[cursor..end]);
            cursor = end;
        } else {
            let length = (control as usize & 0x7f) + 3;
            let distance_bytes = tokens
                .get(cursor..cursor + 2)
                .ok_or(CodecError::Truncated)?;
            let distance = u16::from_le_bytes([distance_bytes[0], distance_bytes[1]]) as usize + 1;
            cursor += 2;
            if distance > output.len()
                || output
                    .len()
                    .checked_add(length)
                    .is_none_or(|n| n > output_len)
            {
                return Err(CodecError::InvalidFormat("invalid RNBW lane match"));
            }
            for _ in 0..length {
                let value = output[output.len() - distance];
                output.push(value);
            }
        }
    }
    if output.len() != output_len {
        return Err(CodecError::InvalidFormat(
            "RNBW lane output length mismatch",
        ));
    }
    Ok(output)
}

#[cfg(test)]
mod tests {

    #[test]
    fn multi_window_lanes_roundtrip_repeated_and_random_data() {
        let mut repeated = Vec::with_capacity(32_000);
        for index in 0..32_000 {
            repeated.push((index % 113) as u8);
        }
        let mut random = Vec::with_capacity(32_000);
        let mut state = 0x9e37_79b9_7f4a_7c15u64;
        for _ in 0..32_000 {
            state ^= state << 13;
            state ^= state >> 7;
            state ^= state << 17;
            random.push(state as u8);
        }
        let encoded = LaneLz::encode(&[&repeated, &random]).unwrap();
        assert_eq!(encoded.lane_count(), 2);
        assert_eq!(encoded.decode_lanes().unwrap(), [repeated, random]);
        let parsed = LaneLz::from_bytes(encoded.as_bytes()).unwrap();
        assert_eq!(
            parsed.decode_lanes().unwrap(),
            encoded.decode_lanes().unwrap()
        );
        let descriptors = parse_image(encoded.as_bytes()).unwrap();
        assert!(descriptors.iter().any(|lane| lane.mode == MODE_LZ_ENTROPY));
        assert!(descriptors.iter().any(|lane| lane.mode == MODE_RAW));
        assert!(
            descriptors
                .iter()
                .filter(|lane| lane.mode == MODE_LZ_ENTROPY)
                .all(|lane| matches!(lane.entropy_method, METHOD_RANS | METHOD_HUFFMAN))
        );
    }

    #[test]
    fn fixed_width_lane_tokens_roundtrip_overlap_and_reject_bad_fields() {
        let mut stream = Vec::new();
        stream.extend_from_slice(&0x0200_0000u32.to_le_bytes()); // 3 literals
        stream.extend_from_slice(&0x8000_0200u32.to_le_bytes()); // length 3, distance 3
        stream.extend_from_slice(b"abc");
        assert_eq!(lz_decode(&stream, 2, 3, 6).unwrap(), b"abcabc");

//! Four-state, 12-bit order-0 rANS wrapper for the raw queryable image.

use super::*;
use arrayvec::ArrayVec;

const ENTROPY_VERSION: u16 = 4;
const LEGACY_RANS_VERSION: u16 = 3;
const RANS_HEADER: usize = 32;
const STREAM_DESCRIPTOR: usize = 24;
const RANS_SCALE_BITS: u32 = 12;
const RANS_SCALE: u32 = 1 << RANS_SCALE_BITS;
const RANS_LOWER_BOUND: u32 = 1 << 23;
const RANS_MODEL_BYTES: usize = 256 * 2;
const HUFFMAN_MODEL_BYTES: usize = 256;
const METHOD_RAW: u8 = 0;
const METHOD_RANS: u8 = 1;
const METHOD_HUFFMAN: u8 = 2;

#[derive(Clone, Copy)]
struct RawStream {
    kind: u8,
    offset: usize,
    len: usize,
}

#[derive(Clone)]
struct StoredStream {
    kind: u8,
    raw_len: usize,
    method: u8,
    model: Vec<u8>,
    bytes: Vec<u8>,
}

#[derive(Clone, Copy)]
struct StoredDescriptor {
    kind: u8,
    method: u8,
    offset: usize,
    stored_len: usize,
    raw_len: usize,
    model_offset: usize,
    model_len: usize,
}

struct RawMetadata {
    key_count: usize,
    leaf_count: usize,
    node_counts: Vec<usize>,
    offsets: ImageOffsets,
    streams: Vec<RawStream>,
}

pub(super) fn entropy_encode(input: &[u8]) -> Result<(u8, Vec<u8>, Vec<u8>), CodecError> {
    if input.is_empty() {
        return Ok((METHOD_RAW, Vec::new(), Vec::new()));
    }
    match entropy_encode_candidate(input)? {
        Some((method, model, payload)) => Ok((method, model, payload)),
        None => Ok((METHOD_RAW, Vec::new(), input.to_vec())),
    }
}

/// Returns an entropy representation only when its complete model+payload is
/// smaller than the source. Callers choosing among larger codec candidates can
/// defer copying the raw source until raw actually wins.
pub(super) fn entropy_encode_candidate(
    input: &[u8],
) -> Result<Option<(u8, Vec<u8>, Vec<u8>)>, CodecError> {
    entropy_encode_candidate_with_hint(input, false)
}

/// A strong match ratio is evidence that the LZ window has already converted
/// repeated source regions into a skewed token stream. In that case, try the
/// cheaper fixed-capacity Huffman pass first and avoid rANS when Huffman wins.
pub(super) fn entropy_encode_candidate_with_hint(
    input: &[u8],
    prefer_huffman: bool,
) -> Result<Option<(u8, Vec<u8>, Vec<u8>)>, CodecError> {
    if input.is_empty() {
        return Ok(None);
    }
    // Both entropy coders need the same symbol histogram. Count it once, then
    // reuse it for rANS normalization and Huffman tree construction.
    let frequencies = symbol_histogram(input);
    if prefer_huffman {
        if let Some((lengths, payload)) = huffman_encode(input, &frequencies, input.len()) {
            return Ok(Some((METHOD_HUFFMAN, lengths.to_vec(), payload)));
        }
    }
    let rans_model = build_model(&frequencies, input.len());
    let rans_payload = rans_encode(input, &rans_model)?;
    let rans_size = rans_payload.len() + RANS_MODEL_BYTES;
    let rans_selected = rans_size < input.len();
    let best_size = if rans_selected {
        rans_size
    } else {
        input.len()
    };
    let best_payload = rans_selected.then_some(rans_payload);

    if !prefer_huffman {
        if let Some((lengths, huffman_payload)) = huffman_encode(input, &frequencies, best_size) {
            let huffman_size = huffman_payload.len() + HUFFMAN_MODEL_BYTES;
            debug_assert!(huffman_size < best_size);
            return Ok(Some((METHOD_HUFFMAN, lengths.to_vec(), huffman_payload)));
        }
    }

    if let Some(payload) = best_payload {
        let mut model = Vec::with_capacity(RANS_MODEL_BYTES);
        for frequency in rans_model {
            model.extend_from_slice(&frequency.to_le_bytes());
        }
        Ok(Some((METHOD_RANS, model, payload)))
    } else {
        Ok(None)
    }
}

pub(super) fn entropy_decode(
    method: u8,
    output_len: usize,
    model_bytes: &[u8],
    payload: &[u8],
) -> Result<Vec<u8>, CodecError> {
    match method {
        METHOD_RAW => {
            if !model_bytes.is_empty() || payload.len() != output_len {
                return Err(CodecError::InvalidFormat("invalid raw entropy stream"));
            }
            Ok(payload.to_vec())
        }
        METHOD_RANS => {
            if model_bytes.len() != RANS_MODEL_BYTES
                || payload.len() + RANS_MODEL_BYTES >= output_len
            {
                return Err(CodecError::InvalidFormat("invalid rANS stream size"));
            }
            let model = read_model(model_bytes)?;
            rans_decode(payload, output_len, &model)
        }
        METHOD_HUFFMAN => {
            if model_bytes.len() != HUFFMAN_MODEL_BYTES
                || payload.len() + HUFFMAN_MODEL_BYTES >= output_len
            {
                return Err(CodecError::InvalidFormat("invalid Huffman stream size"));
            }
            let lengths: &[u8; HUFFMAN_MODEL_BYTES] = model_bytes
                .try_into()
                .map_err(|_| CodecError::InvalidFormat("bad Huffman code lengths"))?;
            huffman_decode(payload, output_len, lengths)
        }
        _ => Err(CodecError::InvalidFormat("unknown entropy method")),
    }
}

pub(super) fn compress_image(image: &[u8]) -> Result<Vec<u8>, CodecError> {
    validate_image(image)?;
    let depth = image[9] as usize;
    let key_width = image[8];
    let raw_header_len = read_u16(image, 6)? as usize;
    let raw_streams = raw_streams(image, depth)?;
    let mut streams = Vec::with_capacity(raw_streams.len());
    for stream in raw_streams {
        let raw = &image[stream.offset..stream.offset + stream.len];
        if raw.is_empty() {
            streams.push(StoredStream {
                kind: stream.kind,
                raw_len: 0,
                method: METHOD_RAW,
                model: Vec::new(),
                bytes: Vec::new(),
            });
            continue;
        }

        let (selected_method, selected_model, selected_bytes) = entropy_encode(raw)?;

        streams.push(StoredStream {
            kind: stream.kind,
            raw_len: raw.len(),
            method: selected_method,
            model: selected_model,
            bytes: selected_bytes,
        });
    }
    let stream_count = streams.len();
    let prefix_offset = RANS_HEADER
        .checked_add(
            stream_count
                .checked_mul(STREAM_DESCRIPTOR)
                .ok_or(CodecError::LengthOverflow)?,
        )
        .ok_or(CodecError::LengthOverflow)?;
    let model_offset_start = prefix_offset
        .checked_add(raw_header_len)
        .ok_or(CodecError::LengthOverflow)?;
    let model_bytes = streams.iter().try_fold(0usize, |total, stream| {
        total
            .checked_add(stream.model.len())
            .ok_or(CodecError::LengthOverflow)
    })?;
    let header_len = model_offset_start
        .checked_add(model_bytes)
        .ok_or(CodecError::LengthOverflow)?;
    let mut position = align_up(header_len, 8).ok_or(CodecError::LengthOverflow)?;
    let mut model_position = model_offset_start;
    let mut descriptors = Vec::with_capacity(stream_count);
    for stream in &streams {
        let method = stream.method;
        let model_len = stream.model.len();
        let model_offset = if method == METHOD_RAW {
            0
        } else {
            model_position
        };
        if method != METHOD_RAW {
            model_position = model_position
                .checked_add(model_len)
                .ok_or(CodecError::LengthOverflow)?;
        }
        let offset = position;
        position = position
            .checked_add(stream.bytes.len())
            .ok_or(CodecError::LengthOverflow)?;
        descriptors.push(StoredDescriptor {
            kind: stream.kind,
            method,
            offset,
            stored_len: stream.bytes.len(),
            raw_len: stream.raw_len,
            model_offset,
            model_len,
        });
        position = align_up(position, 8).ok_or(CodecError::LengthOverflow)?;
    }
    if model_position != header_len
        || header_len > u16::MAX as usize
        || position > u32::MAX as usize
    {
        return Err(CodecError::LengthOverflow);
    }
    let mut output = vec![0u8; position];
    output[0..4].copy_from_slice(MAGIC);
    put_u16(&mut output, 4, ENTROPY_VERSION)?;
    put_u16(&mut output, 6, header_len as u16)?;
    output[8] = key_width;
    output[9] = depth as u8;
    output[10] = stream_count as u8;
    output[11] = 0;
    put_u32(
        &mut output,
        12,
        u32::try_from(image.len()).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    let encoded_len = u32::try_from(output.len()).map_err(|_| CodecError::LengthOverflow)?;
    put_u32(&mut output, 16, encoded_len)?;
    put_u16(
        &mut output,
        20,
        u16::try_from(raw_header_len).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    put_u16(&mut output, 22, 0)?;
    put_u32(&mut output, 24, RANS_HEADER as u32)?;
    put_u32(
        &mut output,
        28,
        u32::try_from(prefix_offset).map_err(|_| CodecError::LengthOverflow)?,
    )?;
    for (index, (stream, descriptor)) in streams.iter().zip(&descriptors).enumerate() {
        let at = RANS_HEADER + index * STREAM_DESCRIPTOR;
        output[at] = descriptor.kind;
        output[at + 1] = descriptor.method;
        put_u16(&mut output, at + 2, 0)?;
        put_u32(
            &mut output,
            at + 4,
            u32::try_from(descriptor.offset).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        put_u32(
            &mut output,
            at + 8,
            u32::try_from(descriptor.stored_len).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        put_u32(
            &mut output,
            at + 12,
            u32::try_from(descriptor.raw_len).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        put_u32(
            &mut output,
            at + 16,
            u32::try_from(descriptor.model_offset).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        put_u32(
            &mut output,
            at + 20,
            u32::try_from(descriptor.model_len).map_err(|_| CodecError::LengthOverflow)?,
        )?;
        if !stream.model.is_empty() {
            let end = descriptor.model_offset + descriptor.model_len;
            output[descriptor.model_offset..end].copy_from_slice(&stream.model);
        }
        output[descriptor.offset..descriptor.offset + descriptor.stored_len]
            .copy_from_slice(&stream.bytes);
    }
    let prefix_end = prefix_offset + raw_header_len;
    output[prefix_offset..prefix_end].copy_from_slice(&image[..raw_header_len]);
    Ok(output)
}

pub(super) fn decompress_image(input: &[u8]) -> Result<Vec<u8>, CodecError> {
    if input.len() < RANS_HEADER {
        return Err(CodecError::Truncated);
    }
    let version = read_u16(input, 4)?;
    if input.get(0..4) != Some(MAGIC.as_slice())
        || ![LEGACY_RANS_VERSION, ENTROPY_VERSION].contains(&version)
    {
        return Err(CodecError::InvalidFormat(
            "bad entropy-coded RNBW signature",
        ));
    }
    let header_len = read_u16(input, 6)? as usize;
    let width = input[8];
    let depth = input[9] as usize;
    let stream_count = input[10] as usize;
    if input[11] != 0
        || !(2..=32).contains(&width)
        || !width.is_multiple_of(2)
        || depth > width as usize / 2
        || stream_count != depth + 2
    {
        return Err(CodecError::InvalidFormat("invalid entropy-coded metadata"));
    }
    let raw_len = read_u32(input, 12)? as usize;
    if read_u32(input, 16)? as usize != input.len() {
        return Err(if read_u32(input, 16)? as usize > input.len() {
            CodecError::Truncated
        } else {
            CodecError::InvalidFormat("trailing entropy-coded bytes")
        });
    }
    let raw_header_len = read_u16(input, 20)? as usize;
    if read_u16(input, 22)? != 0 || read_u32(input, 24)? as usize != RANS_HEADER {
        return Err(CodecError::InvalidFormat("bad entropy descriptor offset"));
    }
    let prefix_offset = read_u32(input, 28)? as usize;
    let expected_prefix_offset = RANS_HEADER
        .checked_add(
            stream_count
                .checked_mul(STREAM_DESCRIPTOR)
                .ok_or(CodecError::LengthOverflow)?,
        )
        .ok_or(CodecError::LengthOverflow)?;
    let prefix_end = prefix_offset
        .checked_add(raw_header_len)
        .ok_or(CodecError::LengthOverflow)?;
    if prefix_offset != expected_prefix_offset
        || prefix_end > input.len()
        || header_len < prefix_end
        || header_len > input.len()
        || raw_header_len != FIXED_HEADER + depth * LEVEL_DESCRIPTOR + TRAILER_LEN
    {
        return Err(CodecError::InvalidFormat("bad raw-header prefix range"));
    }
    let raw_header = &input[prefix_offset..prefix_end];
    let metadata = parse_raw_metadata(raw_header, raw_len, width, depth)?;
    let mut descriptors = Vec::with_capacity(stream_count);
    let mut model_position = prefix_end;
    let mut position = align_up(header_len, 8).ok_or(CodecError::LengthOverflow)?;
    if input[header_len..position].iter().any(|&byte| byte != 0) {
        return Err(CodecError::InvalidFormat("nonzero entropy-header padding"));
    }
    for (index, stream) in metadata.streams.iter().enumerate() {
        let at = RANS_HEADER + index * STREAM_DESCRIPTOR;
        let kind = input[at];
        let method = input[at + 1];
        let max_method = if version == LEGACY_RANS_VERSION {
            METHOD_RANS
        } else {
            METHOD_HUFFMAN
        };
        if kind != stream.kind || method > max_method || read_u16(input, at + 2)? != 0 {
            return Err(CodecError::InvalidFormat(
                "invalid entropy stream descriptor",
            ));
        }
        let descriptor = StoredDescriptor {
            kind,
            method,
            offset: read_u32(input, at + 4)? as usize,
            stored_len: read_u32(input, at + 8)? as usize,
            raw_len: read_u32(input, at + 12)? as usize,
            model_offset: read_u32(input, at + 16)? as usize,
            model_len: read_u32(input, at + 20)? as usize,
        };
        if descriptor.raw_len != stream.len {
            return Err(CodecError::InvalidFormat(
                "entropy stream raw length mismatch",
            ));
        }
        let expected_model_len = match descriptor.method {
            METHOD_RAW => 0,
            METHOD_RANS => RANS_MODEL_BYTES,
            METHOD_HUFFMAN => HUFFMAN_MODEL_BYTES,
            _ => return Err(CodecError::InvalidFormat("unknown entropy method")),
        };
        if descriptor.method == METHOD_RAW {
            if descriptor.model_offset != 0
                || descriptor.model_len != 0
                || descriptor.stored_len != descriptor.raw_len
            {
                return Err(CodecError::InvalidFormat("invalid raw stream descriptor"));
            }
        } else {
            let minimum_payload = if descriptor.method == METHOD_RANS {
                32
            } else {
                1
            };
            if descriptor.model_offset != model_position
                || descriptor.model_len != expected_model_len
                || descriptor.stored_len < minimum_payload
            {
                return Err(CodecError::InvalidFormat("invalid entropy model range"));
            }
            model_position = model_position
                .checked_add(descriptor.model_len)
                .ok_or(CodecError::LengthOverflow)?;
        }
        if descriptor.offset != position {
            return Err(CodecError::InvalidFormat("noncontiguous rANS payload"));
        }
        let data_end = descriptor
            .offset
            .checked_add(descriptor.stored_len)
            .ok_or(CodecError::LengthOverflow)?;
        if data_end > input.len() {
            return Err(CodecError::Truncated);
        }
        position = align_up(data_end, 8).ok_or(CodecError::LengthOverflow)?;
        if position > input.len() || input[data_end..position].iter().any(|&byte| byte != 0) {
            return Err(CodecError::InvalidFormat("bad entropy stream padding"));
        }
        descriptors.push(descriptor);
    }
    if model_position != header_len || position != input.len() {
        return Err(CodecError::InvalidFormat("trailing entropy payload"));
    }

    let mut raw = Vec::new();
    raw.try_reserve_exact(raw_len)
        .map_err(|_| CodecError::LengthOverflow)?;
    raw.resize(raw_len, 0);
    raw[..raw_header_len].copy_from_slice(raw_header);
    for (stream, descriptor) in metadata.streams.iter().zip(&descriptors) {
        let payload = &input[descriptor.offset..descriptor.offset + descriptor.stored_len];
        let decoded = match descriptor.method {
            METHOD_RAW => payload.to_vec(),
            METHOD_RANS => {
                if descriptor.stored_len + RANS_MODEL_BYTES >= descriptor.raw_len {
                    return Err(CodecError::InvalidFormat("rANS stream does not save space"));
                }
                let model = read_model(
                    &input[descriptor.model_offset..descriptor.model_offset + descriptor.model_len],
                )?;
                rans_decode(payload, descriptor.raw_len, &model)?
            }
            METHOD_HUFFMAN => {
                if descriptor.stored_len + HUFFMAN_MODEL_BYTES >= descriptor.raw_len {
                    return Err(CodecError::InvalidFormat(
                        "Huffman stream does not save space",
                    ));
                }
                let lengths: [u8; HUFFMAN_MODEL_BYTES] = input
                    [descriptor.model_offset..descriptor.model_offset + descriptor.model_len]
                    .try_into()
                    .map_err(|_| CodecError::InvalidFormat("bad Huffman code lengths"))?;
                huffman_decode(payload, descriptor.raw_len, &lengths)?
            }
            _ => return Err(CodecError::InvalidFormat("unknown entropy method")),
        };
        if decoded.len() != stream.len {
            return Err(CodecError::InvalidFormat("decoded stream length mismatch"));
        }
        raw[stream.offset..stream.offset + stream.len].copy_from_slice(&decoded);
    }
    rebuild_auxiliary(&mut raw, &metadata)?;
    validate_image(&raw)?;
    Ok(raw)
}

fn raw_streams(image: &[u8], depth: usize) -> Result<Vec<RawStream>, CodecError> {
    let mut streams = Vec::with_capacity(depth + 2);
    for level in 0..depth {
        let at = FIXED_HEADER + level * LEVEL_DESCRIPTOR;
        let nodes = read_u32(image, at)? as usize;
        streams.push(RawStream {
            kind: level as u8,
            offset: read_u32(image, at + 8)? as usize,
            len: nodes.div_ceil(16) * 8,
        });
    }
    let trailer = read_u32(image, 32)? as usize;
    streams.push(RawStream {
        kind: depth as u8,
        offset: read_u32(image, trailer + 4)? as usize,
        len: read_u32(image, trailer + 8)? as usize,
    });
    streams.push(RawStream {
        kind: (depth + 1) as u8,
        offset: read_u32(image, trailer + 20)? as usize,
        len: read_u32(image, trailer + 28)? as usize,
    });
    Ok(streams)
}

fn parse_raw_metadata(
    header: &[u8],
    raw_len: usize,
    width: u8,
    depth: usize,
) -> Result<RawMetadata, CodecError> {
    if header.len() < FIXED_HEADER + TRAILER_LEN
        || header.get(0..4) != Some(MAGIC.as_slice())
        || read_u16(header, 4)? != VERSION
        || header[8] != width
        || header[9] as usize != depth
        || header[11] != 0
        || read_u16(header, 6)? as usize != header.len()
        || read_u32(header, 24)? as usize != raw_len
        || read_u32(header, 28)? as usize != FIXED_HEADER
        || read_u32(header, 32)? as usize != FIXED_HEADER + depth * LEVEL_DESCRIPTOR
        || read_u32(header, 36)? != 0
    {
        return Err(CodecError::InvalidFormat(
            "bad raw header inside entropy stream",
        ));
    }
    let key_count =
        usize::try_from(read_u64(header, 12)?).map_err(|_| CodecError::LengthOverflow)?;
    let leaf_count = read_u32(header, 20)? as usize;
    let mut node_counts = Vec::with_capacity(depth + 1);
    if depth > 0 {
        node_counts.push(read_u32(header, FIXED_HEADER)? as usize);
        if node_counts[0] != 1 {
            return Err(CodecError::InvalidFormat("bad root count in raw header"));
        }
        for level in 0..depth {
            let at = FIXED_HEADER + level * LEVEL_DESCRIPTOR;
            let nodes = read_u32(header, at)? as usize;
            let children = read_u32(header, at + 4)? as usize;
            if nodes != node_counts[level] || children > nodes.saturating_mul(4) {
                return Err(CodecError::InvalidFormat("bad level count in raw header"));
            }
            node_counts.push(children);
        }
        if node_counts[depth] != leaf_count {
            return Err(CodecError::InvalidFormat("bad leaf count in raw header"));
        }
    } else {
        node_counts.push(usize::from(key_count != 0));
        if leaf_count != node_counts[0] {
            return Err(CodecError::InvalidFormat("bad leaf count in raw header"));
        }
    }
    let offsets = image_offsets(width, depth, key_count, leaf_count, &node_counts)?;
    if offsets.total_len != raw_len || offsets.header_len != header.len() {
        return Err(CodecError::InvalidFormat(
            "raw image size disagrees with prefix",
        ));
    }
    for level in 0..depth {
        let at = FIXED_HEADER + level * LEVEL_DESCRIPTOR;
        if read_u32(header, at + 8)? as usize != offsets.levels[level].mask
            || read_u32(header, at + 12)? as usize != offsets.levels[level].rank
        {
            return Err(CodecError::InvalidFormat("bad raw level offsets in prefix"));
        }
    }
    let trailer = FIXED_HEADER + depth * LEVEL_DESCRIPTOR;
    let bucket = offsets.buckets;
    if read_u32(header, trailer)? as usize != bucket.bit_len
        || read_u32(header, trailer + 4)? as usize != bucket.bits
        || read_u32(header, trailer + 8)? as usize != bucket.bytes
        || read_u32(header, trailer + 12)? as usize != bucket.select
        || read_u32(header, trailer + 16)? as usize != bucket.select_count
        || read_u32(header, trailer + 20)? as usize != bucket.low
        || header[trailer + 24] as usize != bucket.low_width
        || read_u32(header, trailer + 28)? as usize != bucket.low_bytes
        || header[trailer + 25..trailer + 28] != [0, 0, 0]
    {
        return Err(CodecError::InvalidFormat(
            "bad raw bucket offsets in prefix",
        ));
    }
    let streams = raw_streams(header, depth)?;
    Ok(RawMetadata {
        key_count,
        leaf_count,
        node_counts,
        offsets,
        streams,
    })
}

fn rebuild_auxiliary(image: &mut [u8], metadata: &RawMetadata) -> Result<(), CodecError> {
    let depth = image[9] as usize;
    for level in 0..depth {
        let at = FIXED_HEADER + level * LEVEL_DESCRIPTOR;
        let node_count = metadata.node_counts[level];
        let mask_offset = read_u32(image, at + 8)? as usize;
        let rank_offset = read_u32(image, at + 12)? as usize;
        let word_count = node_count.div_ceil(16);
        let rank_count = node_count.div_ceil(RANK_STRIDE);
        let mut total = 0u32;
        for block in 0..rank_count {
            put_u32(image, rank_offset + block * 4, total)?;
            let first = block * (RANK_STRIDE / 16);
            let end = (first + RANK_STRIDE / 16).min(word_count);
            for word in first..end {
                total = total
                    .checked_add(read_u64(image, mask_offset + word * 8)?.count_ones())
                    .ok_or(CodecError::LengthOverflow)?;
            }
        }
    }
    let bucket = metadata.offsets.buckets;
    let mut zero = 0usize;
    for position in 0..bucket.bit_len {
        let unary_byte = image[bucket.bits + position / 8];
        if (unary_byte >> (position & 7)) & 1 == 0 {
            if zero.is_multiple_of(SELECT_STRIDE) {
                put_u32(
                    image,
                    bucket.select + (zero / SELECT_STRIDE) * 4,
                    u32::try_from(position).map_err(|_| CodecError::LengthOverflow)?,
                )?;
            }
            zero += 1;
        }
    }
    if zero != metadata.leaf_count || metadata.key_count > bucket.bit_len {
        return Err(CodecError::InvalidFormat("bad decoded bucket counts"));
    }
    Ok(())
}

fn symbol_histogram(input: &[u8]) -> [u64; 256] {
    let mut counts = [0u64; 256];
    for &symbol in input {
        counts[symbol as usize] += 1;
    }
    counts
}

fn build_model(counts: &[u64; 256], input_len: usize) -> [u16; 256] {
    let active = counts.iter().filter(|&&count| count != 0).count();
    let remaining = RANS_SCALE as u64 - active as u64;
    let length = input_len as u64;
    let mut model = [0u16; 256];
    let mut remainders = [(0usize, 0u64); 256];
    let mut remainder_count = 0;
    let mut assigned = 0u32;
    for symbol in 0..256 {
        let count = counts[symbol];
        if count == 0 {
            continue;
        }
        let scaled = count * remaining;
        let frequency = 1 + scaled / length;
        let remainder = scaled % length;
        model[symbol] = frequency as u16;
        assigned += frequency as u32;
        remainders[remainder_count] = (symbol, remainder);
        remainder_count += 1;
    }
    remainders[..remainder_count]
        .sort_unstable_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
    let excess = RANS_SCALE - assigned;
    for &(symbol, _) in remainders[..remainder_count].iter().take(excess as usize) {
        model[symbol] += 1;
    }
    model
}

#[derive(Clone)]
struct HuffmanCodebook {
    codes: [u64; 256],
    counts: [u16; 33],
    first_code: [u64; 33],
    first_symbol: [usize; 33],
    symbols: [u8; 256],
    max_len: usize,
}

#[derive(Clone, Copy)]
struct HuffmanNode {
    left: u16,
    right: u16,
    symbol: u16,
}

#[derive(Clone, Copy)]
struct HuffmanQueueItem {
    frequency: u64,
    minimum_symbol: u16,
    node: u16,
}

impl HuffmanQueueItem {
    #[inline]
    fn precedes(self, other: Self) -> bool {
        (self.frequency, self.minimum_symbol, self.node)
            < (other.frequency, other.minimum_symbol, other.node)
    }
}

fn huffman_encode(
    input: &[u8],
    frequencies: &[u64; 256],
    best_total_size: usize,
) -> Option<([u8; HUFFMAN_MODEL_BYTES], Vec<u8>)> {
    if input.is_empty() {
        return None;
    }
    const NONE: u16 = u16::MAX;
    let mut nodes = ArrayVec::<HuffmanNode, 511>::new();
    let mut queue = ArrayVec::<HuffmanQueueItem, 256>::new();
    for (symbol, &frequency) in frequencies.iter().enumerate() {
        if frequency != 0 {
            let index = u16::try_from(nodes.len()).ok()?;
            nodes
                .try_push(HuffmanNode {
                    left: NONE,
                    right: NONE,
                    symbol: symbol as u16,
                })
                .ok()?;
            huffman_queue_push(
                &mut queue,
                HuffmanQueueItem {
                    frequency,
                    minimum_symbol: symbol as u16,
                    node: index,
                },
            )
            .ok()?;
        }
    }
    while queue.len() > 1 {
        let left = huffman_queue_pop(&mut queue)?;
        let right = huffman_queue_pop(&mut queue)?;
        let frequency = left.frequency.checked_add(right.frequency)?;
        let index = u16::try_from(nodes.len()).ok()?;
        nodes
            .try_push(HuffmanNode {
                left: left.node,
                right: right.node,
                symbol: NONE,
            })
            .ok()?;
        huffman_queue_push(
            &mut queue,
            HuffmanQueueItem {
                frequency,
                minimum_symbol: left.minimum_symbol.min(right.minimum_symbol),
                node: index,
            },
        )
        .ok()?;
    }
    let root = huffman_queue_pop(&mut queue)?.node;

    let mut lengths = [0u8; 256];
    let mut stack = ArrayVec::<(u16, u8), 511>::new();
    stack.try_push((root, 0)).ok()?;
    while let Some((node_index, depth)) = stack.pop() {
        let node = nodes[usize::from(node_index)];
        if node.symbol != NONE {
            if depth > 32 {
                return None;
            }
            lengths[usize::from(node.symbol)] = depth.max(1);
        } else {
            let child_depth = depth.checked_add(1)?;
            if child_depth > 32 {
                return None;
            }
            stack.try_push((node.right, child_depth)).ok()?;
            stack.try_push((node.left, child_depth)).ok()?;
        }
    }
    let codebook = huffman_codebook(&lengths).ok()?;

    let encoded_bits: u128 = frequencies
        .iter()
        .zip(lengths)
        .map(|(&frequency, length)| frequency as u128 * length as u128)
        .sum();
    let encoded_len = usize::try_from(encoded_bits.div_ceil(8)).ok()?;
    if encoded_len.checked_add(HUFFMAN_MODEL_BYTES)? >= best_total_size {
        return None;
    }

    let mut encoded = Vec::with_capacity(encoded_len);
    let mut bits = 0u64;
    let mut bit_count = 0usize;
    for &symbol in input {
        let index = symbol as usize;
        let length = lengths[index] as usize;
        bits = (bits << length) | codebook.codes[index];
        bit_count += length;
        while bit_count >= 8 {
            bit_count -= 8;
            encoded.push((bits >> bit_count) as u8);
            bits &= (1u64 << bit_count) - 1;
        }
    }
    if bit_count != 0 {
        encoded.push((bits << (8 - bit_count)) as u8);
    }
    Some((lengths, encoded))
}

fn huffman_queue_push(
    queue: &mut ArrayVec<HuffmanQueueItem, 256>,
    item: HuffmanQueueItem,
) -> Result<(), ()> {
    queue.try_push(item).map_err(|_| ())?;
    let mut index = queue.len() - 1;
    while index != 0 {
        let parent = (index - 1) / 2;
        if !queue[index].precedes(queue[parent]) {
            break;
        }
        queue.swap(index, parent);
        index = parent;
    }
    Ok(())
}

fn huffman_queue_pop(queue: &mut ArrayVec<HuffmanQueueItem, 256>) -> Option<HuffmanQueueItem> {
    let minimum = *queue.first()?;
    let last = queue.pop()?;
    if !queue.is_empty() {
        queue[0] = last;
        let mut index = 0usize;
        loop {
            let left = index * 2 + 1;
            if left >= queue.len() {
                break;
            }
            let right = left + 1;
            let child = if right < queue.len() && queue[right].precedes(queue[left]) {
                right
            } else {
                left
            };
            if !queue[child].precedes(queue[index]) {
                break;
            }
            queue.swap(index, child);
            index = child;
        }
    }
    Some(minimum)
}

fn huffman_codebook(lengths: &[u8; HUFFMAN_MODEL_BYTES]) -> Result<HuffmanCodebook, CodecError> {
    let mut counts = [0u16; 33];
    let mut max_len = 0usize;
    for &length in lengths {
        if length > 32 {
            return Err(CodecError::InvalidFormat("Huffman code exceeds 32 bits"));
        }
        if length != 0 {
            counts[length as usize] = counts[length as usize]
                .checked_add(1)
                .ok_or(CodecError::LengthOverflow)?;
            max_len = max_len.max(length as usize);
        }
    }
    if max_len == 0 {
        return Err(CodecError::InvalidFormat("empty Huffman model"));
    }

    let mut first_code = [0u64; 33];
    let mut next_code = [0u64; 33];
    let mut first_symbol = [0usize; 33];
    let mut symbols = [0u8; 256];
    let mut code = 0u64;
    let mut symbol_offset = 0usize;
    for length in 1..=max_len {
        code = (code + counts[length - 1] as u64) << 1;
        let limit = 1u64 << length;
        if code + counts[length] as u64 > limit {
            return Err(CodecError::InvalidFormat("oversubscribed Huffman model"));
        }
        first_code[length] = code;
        next_code[length] = code;
        first_symbol[length] = symbol_offset;
        for (symbol, &symbol_length) in lengths.iter().enumerate() {
            if symbol_length as usize == length {
                symbols[symbol_offset] = symbol as u8;
                symbol_offset += 1;
            }
        }
    }

    let mut codes = [0u64; 256];
    for (symbol, &length) in lengths.iter().enumerate() {
        if length != 0 {
            let slot = length as usize;
            codes[symbol] = next_code[slot];
            next_code[slot] += 1;
        }
    }
    Ok(HuffmanCodebook {
        codes,
        counts,
        first_code,
        first_symbol,
        symbols,
        max_len,
    })
}

fn huffman_decode(
    payload: &[u8],
    output_len: usize,
    lengths: &[u8; HUFFMAN_MODEL_BYTES],
) -> Result<Vec<u8>, CodecError> {
    let codebook = huffman_codebook(lengths)?;
    let bit_len = payload
        .len()
        .checked_mul(8)
        .ok_or(CodecError::LengthOverflow)?;
    let mut bit_position = 0usize;
    let mut output = Vec::new();
    output
        .try_reserve_exact(output_len)
        .map_err(|_| CodecError::LengthOverflow)?;
    for _ in 0..output_len {
        let mut code = 0u64;
        let mut found = None;
        for length in 1..=codebook.max_len {
            if bit_position >= bit_len {
                return Err(CodecError::Truncated);
            }
            let byte = payload[bit_position / 8];
            let bit = (byte >> (7 - (bit_position & 7))) & 1;
            bit_position += 1;
            code = (code << 1) | bit as u64;
            let first = codebook.first_code[length];
            let count = codebook.counts[length] as u64;
            if code >= first && code - first < count {
                let index = codebook.first_symbol[length] + (code - first) as usize;
                found = Some(codebook.symbols[index]);
                break;
            }
        }
        output.push(found.ok_or(CodecError::InvalidFormat("invalid Huffman code"))?);
    }

    let expected_bytes = bit_position.div_ceil(8);
    if expected_bytes != payload.len() {
        return Err(CodecError::InvalidFormat("trailing Huffman payload"));
    }
    let padding = expected_bytes * 8 - bit_position;
    if padding != 0 {
        let mask = (1u8 << padding) - 1;
        if payload.last().is_some_and(|byte| byte & mask != 0) {
            return Err(CodecError::InvalidFormat("nonzero Huffman padding"));
        }
    }
    Ok(output)
}

fn rans_encode(input: &[u8], frequencies: &[u16; 256]) -> Result<Vec<u8>, CodecError> {
    if input.is_empty() {
        return Ok(Vec::new());
    }
    let mut cumulative = [0u32; 256];
    let mut total = 0u32;
    for (symbol, frequency) in frequencies.iter().copied().enumerate() {
        cumulative[symbol] = total;
        total += frequency as u32;
    }
    if total != RANS_SCALE {
        return Err(CodecError::InvalidFormat(
            "rANS frequencies do not sum to 4096",
        ));
    }
    let mut states = [RANS_LOWER_BOUND; 4];
    let mut emitted: [Vec<u8>; 4] = std::array::from_fn(|_| Vec::new());
    for index in (0..input.len()).rev() {
        let symbol = input[index] as usize;
        let lane = index & 3;
        let frequency = frequencies[symbol] as u32;
        if frequency == 0 {
            return Err(CodecError::InvalidFormat(
                "rANS model omits an input symbol",
            ));
        }
        let threshold = ((RANS_LOWER_BOUND >> RANS_SCALE_BITS) << 8) * frequency;
        while states[lane] >= threshold {
            emitted[lane].push(states[lane] as u8);
            states[lane] >>= 8;
        }
        states[lane] = ((states[lane] / frequency) << RANS_SCALE_BITS)
            + (states[lane] % frequency)
            + cumulative[symbol];
    }
    for lane in &mut emitted {
        lane.reverse();
    }
    let payload_len = emitted.iter().try_fold(32usize, |sum, lane| {
        sum.checked_add(lane.len())
            .ok_or(CodecError::LengthOverflow)
    })?;
    let mut output = vec![0u8; payload_len];
    for lane in 0..4 {
        put_u32(&mut output, lane * 4, states[lane])?;
        put_u32(
            &mut output,
            16 + lane * 4,
            u32::try_from(emitted[lane].len()).map_err(|_| CodecError::LengthOverflow)?,
        )?;
    }
    let mut cursor = 32usize;
    for lane in emitted {
        output[cursor..cursor + lane.len()].copy_from_slice(&lane);
        cursor += lane.len();
    }
    Ok(output)
}

fn rans_decode(
    payload: &[u8],
    output_len: usize,
    frequencies: &[u16; 256],
) -> Result<Vec<u8>, CodecError> {
    if payload.len() < 32 {
        return Err(CodecError::Truncated);
    }
    let mut cumulative = [0u32; 256];
    let mut decode_table = [0u8; RANS_SCALE as usize];
    let mut total = 0u32;
    for (symbol, frequency) in frequencies.iter().copied().enumerate() {
        cumulative[symbol] = total;
        for slot in total..total + frequency as u32 {
            if slot >= RANS_SCALE {
                return Err(CodecError::InvalidFormat(
                    "rANS model exceeds frequency scale",
                ));
            }
            decode_table[slot as usize] = symbol as u8;
        }
        total = total
            .checked_add(frequency as u32)
            .ok_or(CodecError::LengthOverflow)?;
    }
    if total != RANS_SCALE || frequencies.iter().all(|&frequency| frequency == 0) {
        return Err(CodecError::InvalidFormat("invalid rANS frequency table"));
    }
    let mut states = [0u32; 4];
    let mut lengths = [0usize; 4];
    let mut starts = [0usize; 4];
    for lane in 0..4 {
        states[lane] = read_u32(payload, lane * 4)?;
        lengths[lane] = read_u32(payload, 16 + lane * 4)? as usize;
        if states[lane] < RANS_LOWER_BOUND {
            return Err(CodecError::InvalidFormat("invalid initial rANS state"));
        }
    }
    let mut next_start = 32usize;
    for lane in 0..4 {
        starts[lane] = next_start;
        next_start = next_start
            .checked_add(lengths[lane])
            .ok_or(CodecError::LengthOverflow)?;
        if next_start > payload.len() {
            return Err(CodecError::Truncated);
        }
    }
    if next_start != payload.len() {
        return Err(CodecError::InvalidFormat(
            "rANS lane lengths do not fill payload",
        ));
    }
    let mut cursors = starts;
    let ends = std::array::from_fn::<_, 4, _>(|lane| starts[lane] + lengths[lane]);
    let mut output = Vec::new();
    output
        .try_reserve_exact(output_len)
        .map_err(|_| CodecError::LengthOverflow)?;
    output.resize(output_len, 0);
    for (index, output_byte) in output.iter_mut().enumerate() {
        let lane = index & 3;
        let state = states[lane];
        let slot = state & (RANS_SCALE - 1);
        let symbol = decode_table[slot as usize] as usize;
        let frequency = frequencies[symbol] as u64;
        let offset = (slot - cumulative[symbol]) as u64;
        let mut next = frequency * (state as u64 >> RANS_SCALE_BITS) + offset;
        if next > u32::MAX as u64 {
            return Err(CodecError::InvalidFormat("rANS state overflow"));
        }
        while next < RANS_LOWER_BOUND as u64 {
            if cursors[lane] >= ends[lane] {
                return Err(CodecError::Truncated);
            }
            next = (next << 8) | payload[cursors[lane]] as u64;
            cursors[lane] += 1;
        }
        states[lane] = next as u32;
        *output_byte = symbol as u8;
    }
    for lane in 0..4 {
        if cursors[lane] != ends[lane] || states[lane] != RANS_LOWER_BOUND {
            return Err(CodecError::InvalidFormat(
                "rANS stream has a bad terminal state",
            ));
        }
    }
    Ok(output)
}

fn read_model(bytes: &[u8]) -> Result<[u16; 256], CodecError> {
    if bytes.len() != RANS_MODEL_BYTES {
        return Err(CodecError::InvalidFormat("bad rANS table length"));
    }
    let mut model = [0u16; 256];
    let mut total = 0u32;
    for (index, frequency) in model.iter_mut().enumerate() {
        *frequency = read_u16(bytes, index * 2)?;
        total = total
            .checked_add(*frequency as u32)
            .ok_or(CodecError::LengthOverflow)?;
    }
    if total != RANS_SCALE {
        return Err(CodecError::InvalidFormat("rANS table sum is not 4096"));
    }
    Ok(model)
}

mod tests {

    #[test]
    fn four_state_rans_roundtrips_short_and_all_symbol_streams() {
        let mut seed = 0x9e37_79b9u32;
        for length in [1usize, 2, 3, 4, 7, 31, 32, 33, 255, 4096] {
            let mut input = Vec::with_capacity(length);
            for index in 0..length {
                seed ^= seed << 13;
                seed ^= seed >> 17;
                seed ^= seed << 5;
                input.push(if index % 3 == 0 {
                    (seed >> 16) as u8
                } else {
                    (index % 5) as u8
                });
            }
            let frequencies = symbol_histogram(&input);
            let model = build_model(&frequencies, input.len());
            assert_eq!(
                model.iter().map(|&frequency| frequency as u32).sum::<u32>(),
                RANS_SCALE
            );
            let encoded = rans_encode(&input, &model).unwrap();
            assert_eq!(rans_decode(&encoded, input.len(), &model).unwrap(), input);
        }
    }

    #[test]
    fn canonical_huffman_roundtrips_skewed_and_full_alphabets() {
        let mut skewed = vec![0u8; 20_000];
        skewed.extend((0..256).map(|value| value as u8));
        let mut full = Vec::with_capacity(8192);
        for _ in 0..32 {
            full.extend(0..=u8::MAX);
        }
        for input in [&skewed, &full] {
            let frequencies = symbol_histogram(input);
            let (lengths, encoded) = huffman_encode(input, &frequencies, usize::MAX).unwrap();
            assert_eq!(
                huffman_decode(&encoded, input.len(), &lengths).unwrap(),
                *input
            );
        }
        let mut invalid = [0u8; HUFFMAN_MODEL_BYTES];
        invalid[..3].fill(1);
        assert!(huffman_codebook(&invalid).is_err());
    }

    #[test]
    fn entropy_image_roundtrips_and_rejects_corruption() {
        let values: Vec<u32> = (0..=u16::MAX as u32).collect();
        let source = Rnbw::encode_width(&values, 16).unwrap();
        let compressed = source.to_compressed_bytes().unwrap();
        assert!(compressed.len() < source.as_bytes().len());
        let header_len = read_u16(&compressed, 6).unwrap() as usize;
        let prefix_start = read_u32(&compressed, 28).unwrap() as usize;
        let prefix_end = prefix_start + read_u16(&compressed, 20).unwrap() as usize;
        let stream_count = compressed[10] as usize;
        assert_eq!(read_u16(&compressed, 4).unwrap(), ENTROPY_VERSION);
        let mut model_count = 0;
        let mut method_count = [0usize; 3];
        for index in 0..stream_count {
            let descriptor = RANS_HEADER + index * STREAM_DESCRIPTOR;
            let method = compressed[descriptor + 1] as usize;
            method_count[method] += 1;
            if method == METHOD_RANS as usize || method == METHOD_HUFFMAN as usize {
                let model_start = read_u32(&compressed, descriptor + 16).unwrap() as usize;
                let model_len = read_u32(&compressed, descriptor + 20).unwrap() as usize;
                assert_eq!(
                    model_len,
                    if method == METHOD_RANS as usize {
                        RANS_MODEL_BYTES
                    } else {
                        HUFFMAN_MODEL_BYTES
                    }
                );
                assert!(model_start >= prefix_end);
                assert!(model_start + model_len <= header_len);
                model_count += 1;
            }
        }

        assert!(model_count > 0, "test image must exercise a stored model");
        assert!(
            method_count[METHOD_HUFFMAN as usize] > 0,
            "test should select Huffman"
        );
        let decoded = Rnbw::from_compressed_bytes(&compressed).unwrap();
        assert_eq!(decoded.as_bytes(), source.as_bytes());
        assert_eq!(decoded.decode(), values);
        for value in [0, 1, 17, 32_768, u16::MAX as u32] {
            assert!(decoded.contains(value));
        }
        let mut corrupt = compressed.clone();
        corrupt[0] ^= 1;
        assert!(Rnbw::from_compressed_bytes(&corrupt).is_err());
        let mut corrupt = compressed;
        *corrupt.last_mut().unwrap() ^= 1;
        assert!(Rnbw::from_compressed_bytes(&corrupt).is_err());
    }

    #[test]
    fn version_three_rans_streams_remain_readable() {
        let source = Rnbw::encode(&[3, 0x1234_5678, u32::MAX]).unwrap();
        let mut compressed = source.to_compressed_bytes().unwrap();
        for index in 0..compressed[10] as usize {
            let descriptor = RANS_HEADER + index * STREAM_DESCRIPTOR;
            assert_ne!(compressed[descriptor + 1], METHOD_HUFFMAN);
        }
        put_u16(&mut compressed, 4, LEGACY_RANS_VERSION).unwrap();
        let decoded = Rnbw::from_compressed_bytes(&compressed).unwrap();
        assert_eq!(decoded.as_bytes(), source.as_bytes());
    }

    #[test]
    fn repetition_hint_uses_light_huffman_before_rans() {
        let input = vec![0x5a; 32_768];
        let (method, model, payload) = entropy_encode_candidate_with_hint(&input, true)
            .unwrap()
            .unwrap();
        assert_eq!(method, METHOD_HUFFMAN);
        assert!(model.len() + payload.len() < input.len());
        assert_eq!(
            entropy_decode(method, input.len(), &model, &payload).unwrap(),
            input
        );

        let mixed = (0..4096)
            .map(|n| (n * 73 + n / 7) as u8)
            .collect::<Vec<_>>();
        let candidate = entropy_encode_candidate_with_hint(&mixed, true).unwrap();
        if let Some((method, model, payload)) = candidate {
            assert_eq!(
                entropy_decode(method, mixed.len(), &model, &payload).unwrap(),
                mixed
            );
        }
    }

mod datasets;

use rnbw::{PlainEf, Rnbw};
use std::env;
use std::error::Error;
use std::hint::black_box;

fn main() {
    if let Err(error) = run() {
        eprintln!("error: {error}");
        std::process::exit(2);
    }
}

fn run() -> Result<(), Box<dyn Error>> {
    let mut args = env::args().skip(1);
    let Some(command) = args.next() else {
        return usage();
    };
    if command == "--help" || command == "help" {
        return usage();
    }
    if command != "bench" {
        return Err(format!("unknown command {command:?}; use `rnbw --help`").into());
    }

    let mut workload = "uniform".to_string();
    let mut operation = "build".to_string();
    let mut engine = "all".to_string();
    let mut size = 1_000_000usize;
    while let Some(flag) = args.next() {
        match flag.as_str() {
            "--workload" => workload = args.next().ok_or("--workload needs a value")?,
            "--operation" => operation = args.next().ok_or("--operation needs a value")?,
            "--engine" => engine = args.next().ok_or("--engine needs a value")?,
            "--size" => size = args.next().ok_or("--size needs a value")?.parse()?,
            "--quick" => size = 20_000,
            "-h" | "--help" => return usage(),
            other => return Err(format!("unknown option {other:?}").into()),
        }
    }
    if !["build", "contains", "decode"].contains(&operation.as_str()) {
        return Err("--operation must be build, contains, or decode".into());
    }
    let values = datasets::make_dataset(&workload, size)?;
    if values.is_empty() {
        return Err("workload generated no values".into());
    }
    println!(
        "workload={workload} operation={operation} requested={} actual={}",
        size,
        values.len()
    );

    let engines: Vec<&str> = if engine == "all" {
        vec!["rnbw", "plain-ef", "raw"]
    } else {
        vec![engine.as_str()]
    };
    for name in engines {
        run_engine(name, &operation, &values)?;
    }
    Ok(())
}

fn run_engine(engine: &str, operation: &str, values: &[u32]) -> Result<(), Box<dyn Error>> {
    let probes = if operation == "contains" {
        make_probes(values)
    } else {
        Vec::new()
    };
    let expected_hits = probes
        .iter()
        .filter(|&&p| values.binary_search(&p).is_ok())
        .count();
    let (bytes, runtime_index_bytes, hits, decoded) = match engine {
        "rnbw" => {
            let bitmap = Rnbw::encode(values)?;
            let bytes = bitmap.as_bytes().len();
            let hits = if operation == "contains" {
                Some(
                    probes
                        .iter()
                        .filter(|&&p| black_box(bitmap.contains(black_box(p))))
                        .count(),
                )
            } else {
                None
            };
            let decoded = if operation == "decode" {
                Some(bitmap.decode())
            } else {
                None
            };
            (bytes, bitmap.runtime_index_bytes(), hits, decoded)
        }
        "plain-ef" => {
            let ef = PlainEf::encode(values)?;
            let bytes = ef.encoded_size();
            let hits = if operation == "contains" {
                Some(
                    probes
                        .iter()
                        .filter(|&&p| black_box(ef.contains(black_box(p))))
                        .count(),
                )
            } else {
                None
            };
            let decoded = if operation == "decode" {
                Some(ef.decode())
            } else {
                None
            };
            (bytes, 0, hits, decoded)
        }
        "raw" => {
            let raw = values.to_vec();
            let bytes = raw.len() * std::mem::size_of::<u32>();
            let hits = if operation == "contains" {
                Some(
                    probes
                        .iter()
                        .filter(|&&p| black_box(raw.binary_search(&black_box(p)).is_ok()))
                        .count(),
                )
            } else {
                None
            };
            let decoded = if operation == "decode" {
                Some(raw.clone())
            } else {
                None
            };
            (bytes, 0, hits, decoded)
        }
        other => {
            return Err(format!("unknown engine {other:?}; choose rnbw, plain-ef, or raw").into());
        }
    };

    if let Some(decoded) = decoded
        && decoded.as_slice() != values
    {
        return Err(format!("{engine} decode did not match input").into());
    }
    if let Some(hits) = hits
        && hits != expected_hits
    {
        return Err(format!("{engine} returned {hits} hits; expected {expected_hits}").into());
    }
    println!(
        "engine={engine} encoded_bytes={bytes} runtime_index_bytes={runtime_index_bytes} bytes_per_key={:.4} probes={} hits={}",
        bytes as f64 / values.len() as f64,
        probes.len(),
        expected_hits
    );
    Ok(())
}

fn make_probes(values: &[u32]) -> Vec<u32> {
    let sample_count = values.len().min(4096);
    let mut probes = Vec::with_capacity(sample_count * 2);
    let mut state = 0x9e37_79b9_7f4a_7c15u64;
    for _ in 0..sample_count {
        state ^= state << 13;
        state ^= state >> 7;
        state ^= state << 17;
        probes.push(values[(state as usize) % values.len()]);
    }
    while probes.len() < sample_count * 2 {
        state ^= state << 13;
        state ^= state >> 7;
        state ^= state << 17;
        let candidate = state as u32;
        if values.binary_search(&candidate).is_err() {
            probes.push(candidate);
        }
    }
    probes
}

fn usage() -> Result<(), Box<dyn Error>> {
    println!(
        "RNBW adaptive u32 set codec\n\nUsage:\n  rnbw bench [--workload uniform|clustered|dense|ecs|morton-shell] [--size N] [--engine all|rnbw|plain-ef|raw] [--operation build|contains|decode]\n  rnbw bench --quick\n\nThe CLI reports encoded size and verifies requested contains/decode work. Hyperfine measures full process time; Criterion isolates operations. External codecs are benchmark-only."
    );
    Ok(())
}
